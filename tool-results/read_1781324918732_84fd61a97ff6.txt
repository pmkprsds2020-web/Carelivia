   155→function formatDate(dateStr: string): string {
   156→  try {
   157→    return new Date(dateStr).toLocaleDateString('id-ID', {
   158→      day: 'numeric',
   159→      month: 'short',
   160→      year: 'numeric',
   161→    });
   162→  } catch {
   163→    return dateStr;
   164→  }
   165→}
   166→
   167→function formatDateTime(dateStr: string): string {
   168→  try {
   169→    return new Date(dateStr).toLocaleDateString('id-ID', {
   170→      day: 'numeric',
   171→      month: 'short',
   172→      year: 'numeric',
   173→      hour: '2-digit',
   174→      minute: '2-digit',
   175→    });
   176→  } catch {
   177→    return dateStr;
   178→  }
   179→}

   181→function genId(prefix: string): string {
   182→  return `${prefix}-${Date.now()}-${Math.random().toString(36).substring(2, 7)}`;

   264→function calcBmi(weight?: number, height?: number): number | undefined {
   265→  if (!weight || !height || height <= 0) return undefined;
   266→  const heightM = height / 100;
   267→  return Math.round((weight / (heightM * heightM)) * 10) / 10;

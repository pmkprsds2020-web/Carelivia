   144→function checkTTVAlerts(ttv: TTVFormAnswers): { alert: boolean; severity: 'hijau' | 'kuning' | 'merah'; messages: string[] } {
   145→  const messages: string[] = [];
   146→  let maxSeverity: 'hijau' | 'kuning' | 'merah' = 'hijau';
   147→
   148→  if (ttv.systolicBP !== undefined && ttv.systolicBP < 90) {
   149→    messages.push(`TD rendah: ${ttv.systolicBP} mmHg`);
   150→    maxSeverity = 'merah';
   151→  }
   152→  if (ttv.systolicBP !== undefined && ttv.systolicBP > 180) {
   153→    messages.push(`TD tinggi: ${ttv.systolicBP} mmHg`);
   154→    maxSeverity = 'merah';
   155→  }
   156→  if (ttv.heartRate !== undefined && ttv.heartRate > 120) {
   157→    messages.push(`Nadi tinggi: ${ttv.heartRate} bpm`);
   158→    maxSeverity = 'merah';
   159→  }
   160→  if (ttv.respiratoryRate !== undefined && ttv.respiratoryRate > 30) {
   161→    messages.push(`RR tinggi: ${ttv.respiratoryRate}/menit`);
   162→    maxSeverity = 'merah';
   163→  } else if (ttv.respiratoryRate !== undefined && ttv.respiratoryRate > 24) {
   164→    messages.push(`RR meningkat: ${ttv.respiratoryRate}/menit`);
   165→    if (maxSeverity !== 'merah') maxSeverity = 'kuning';
   166→  }
   167→  if (ttv.oxygenSat !== undefined && ttv.oxygenSat < 90) {
   168→    messages.push(`SpO2 rendah: ${ttv.oxygenSat}%`);
   169→    maxSeverity = 'merah';
   170→  }
   171→  if (ttv.temperature !== undefined && ttv.temperature > 38) {
   172→    messages.push(`Demam: ${ttv.temperature}°C`);
   173→    if (maxSeverity !== 'merah') maxSeverity = 'kuning';
   174→  }
   175→  if (ttv.painScore !== undefined && ttv.painScore >= 7) {
   176→    messages.push(`Nyeri berat: ${ttv.painScore}/10`);
   177→    if (maxSeverity !== 'merah') maxSeverity = 'kuning';
   178→  }
   179→
   180→  return { alert: messages.length > 0, severity: maxSeverity, messages };
   181→}

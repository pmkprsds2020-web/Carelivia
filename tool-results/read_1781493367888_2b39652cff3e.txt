   154→// ── Helper: Compute vital sign criticality score ─────────────────────────────
   155→function computeCriticalityScore(v: VitalSignRecord): { score: number; reasons: string[] } {
   156→  let score = 0;
   157→  const reasons: string[] = [];
   158→
   159→  if (v.oxygenSat != null && v.oxygenSat < 90) {
   160→    score += 4;
   161→    reasons.push(`SpO2 ${v.oxygenSat}% (<90%)`);
   162→  }
   163→  if (v.heartRate != null) {
   164→    if (v.heartRate > 120) { score += 3; reasons.push(`Nadi ${v.heartRate} bpm (>120)`); }
   165→    else if (v.heartRate < 50) { score += 3; reasons.push(`Nadi ${v.heartRate} bpm (<50)`); }
   166→  }
   167→  if (v.respiratoryRate != null) {
   168→    if (v.respiratoryRate > 28) { score += 3; reasons.push(`RR ${v.respiratoryRate}/menit (>28)`); }
   169→    else if (v.respiratoryRate < 8) { score += 3; reasons.push(`RR ${v.respiratoryRate}/menit (<8)`); }
   170→  }
   171→  if (v.systolicBP != null) {
   172→    if (v.systolicBP > 180) { score += 3; reasons.push(`SBP ${v.systolicBP} mmHg (>180)`); }
   173→    else if (v.systolicBP < 90) { score += 3; reasons.push(`SBP ${v.systolicBP} mmHg (<90)`); }
   174→  }
   175→  if (v.temperature != null) {
   176→    if (v.temperature > 39) { score += 2; reasons.push(`Suhu ${v.temperature}°C (>39)`); }
   177→    else if (v.temperature < 35) { score += 2; reasons.push(`Suhu ${v.temperature}°C (<35)`); }
   178→  }
   179→
   180→  return { score, reasons };
   181→}

   183→// ── Helper: Categorize medication ────────────────────────────────────────────
   184→function categorizeMedication(med: MedicationWithAdherences): string {
   185→  const name = med.medicineName.toLowerCase();
   186→  const indication = (med.indication || '').toLowerCase();
   187→
   188→  // Analgesik
   189→  const analgesikKeywords = ['nyeri', 'pain', 'morphine', 'fentanyl', 'tramadol', 'ketorolac', 'paracetamol', 'ibuprofen', 'diclofenac', 'morfine', 'kodein', 'gabapentin', 'pregabalin'];
   190→  if (analgesikKeywords.some(k => name.includes(k) || indication.includes(k))) return 'analgesik';
   191→
   192→  // Antiemetik
   193→  const antiemetikKeywords = ['mual', 'nausea', 'vomiting', 'muntah', 'ondansetron', 'metoclopramide', 'domperidone', 'granisetron'];
   194→  if (antiemetikKeywords.some(k => name.includes(k) || indication.includes(k))) return 'antiemetik';
   195→
   196→  // Laksatif
   197→  const laksatifKeywords = ['konstipasi', 'constipation', 'sembelit', 'bisacodyl', 'lactulose', 'senna', 'peg', 'movicol'];
   198→  if (laksatifKeywords.some(k => name.includes(k) || indication.includes(k))) return 'laksatif';
   199→
   200→  // Antidepresan
   201→  const antidepresanKeywords = ['depresi', 'depression', 'sertraline', 'fluoxetine', 'mirtazapine', 'citalopram', 'escitalopram', 'venlafaxine', 'amitriptyline'];
   202→  if (antidepresanKeywords.some(k => name.includes(k) || indication.includes(k))) return 'antidepresan';
   203→
   204→  // Ansiolitik
   205→  const ansiolitikKeywords = ['cemas', 'anxiety', 'diazepam', 'lorazepam', 'alprazolam', 'clonazepam', 'midazolam'];
   206→  if (ansiolitikKeywords.some(k => name.includes(k) || indication.includes(k))) return 'ansiolitik';
   207→
   208→  return 'lainnya';
   209→}

   228→// ── Helper: Format TTV for narrative ─────────────────────────────────────────
   229→function formatTTVNarrative(v: VitalSignRecord, label: string): string {
   230→  const tgl = new Date(v.recordedAt).toLocaleString('id-ID');
   231→  const parts = [
   232→    `${label} (${tgl}):`,
   233→    `  TD: ${fmtVal(v.systolicBP, '')}/${fmtVal(v.diastolicBP, ' mmHg')}`,
   234→    `  Nadi: ${fmtVal(v.heartRate, ' bpm')}`,
   235→    `  RR: ${fmtVal(v.respiratoryRate, '/menit')}`,
   236→    `  Suhu: ${fmtVal(v.temperature, '°C')}`,
   237→    `  SpO2: ${fmtVal(v.oxygenSat, '%')}`,
   238→  ];
   239→  if (v.weight != null) parts.push(`  BB: ${v.weight} kg`);
   240→  if (v.bmi != null) parts.push(`  BMI: ${v.bmi}`);
   241→  if (v.notes) parts.push(`  Catatan: ${v.notes}`);
   242→  return parts.join('\n');
   243→}

   245→// ── Helper: Format complaint for narrative ───────────────────────────────────
   246→function formatComplaintNarrative(c: DailyComplaintRecord, label: string): string {
   247→  const tgl = new Date(c.submittedAt).toLocaleString('id-ID');
   248→  const parts = [`${label} (${tgl}):`];
   249→  parts.push(`  Kondisi: ${c.kondisiHariIni === 'baik' ? 'Baik' : 'Tidak Baik'}`);
   250→  if (c.alasanKondisi) parts.push(`  Alasan: ${c.alasanKondisi}`);
   251→  if (c.keluhanBaru === 'ada' && c.deskripsiKeluhanBaru) {
   252→    parts.push(`  Keluhan Baru: ${c.deskripsiKeluhanBaru}`);
   253→  }
   254→  parts.push(`  Nyeri: ${c.kondisiNyeri}`);
   255→  parts.push(`  Sesak: ${c.kondisiSesak}`);
   256→  parts.push(`  Makan/Minum: ${c.makanMinum === 'ya' ? 'Ya' : 'Tidak'}`);
   257→  if (c.makanMinum === 'tidak' && c.alasanMakanMinum) parts.push(`  Alasan: ${c.alasanMakanMinum}`);
   258→  parts.push(`  Tidur: ${c.tidur === 'ya' ? 'Ya' : 'Tidak'}`);
   259→  if (c.tidur === 'tidak' && c.alasanTidur) parts.push(`  Alasan: ${c.alasanTidur}`);
   260→  if (c.masalahObat === 'ya' && c.deskripsiMasalahObat) {
   261→    parts.push(`  Masalah Obat: ${c.deskripsiMasalahObat}`);
   262→  }
   263→  parts.push(`  Tingkat Keparahan: ${c.severityLevel}`);
   264→  return parts.join('\n');
   265→}

   642→    });

   645→
   646→  return lines.join('\n');
   647→}

   649→// ── Helper: Parse JSON from LLM response ─────────────────────────────────────
   650→function parseLLMJson(response: string): AIAnalysis | null {
   651→  // Try direct JSON parse
   652→  try {
   653→    const parsed = JSON.parse(response);
   654→    if (parsed && typeof parsed === 'object') return parsed as AIAnalysis;
   655→  } catch {
   656→    // Not direct JSON, continue
   657→  }
   658→
   659→  // Try to extract JSON from markdown code blocks
   660→  const jsonBlockMatch = response.match(/```(?:json)?\s*\n?([\s\S]*?)\n?\s*```/);
   661→  if (jsonBlockMatch) {
   662→    try {
   663→      const parsed = JSON.parse(jsonBlockMatch[1].trim());
   664→      if (parsed && typeof parsed === 'object') return parsed as AIAnalysis;
   665→    } catch {
   666→      // Failed to parse code block JSON
   667→    }
   668→  }
   669→
   670→  // Try to find JSON object in the response
   671→  const jsonMatch = response.match(/\{[\s\S]*\}/);
   672→  if (jsonMatch) {
   673→    try {
   674→      const parsed = JSON.parse(jsonMatch[0]);
   675→      if (parsed && typeof parsed === 'object') return parsed as AIAnalysis;
   676→    } catch {
   677→      // Failed to parse extracted JSON
   678→    }
   679→  }
   680→
   681→  return null;
   682→}

   942→  }
   943→  if (patient.careStatus === 'rawat_jalan' && patient.riskLevel !== 'hijau') {
   944→    rekomendasi.push('Home care untuk memfasilitasi perawatan di lingkungan yang nyaman bagi pasien');

   954→      rekomendasi.push('Lengkapi penandatanganan dokumen Advance Care Planning');

   990→// ── Helper: Generate document number ─────────────────────────────────────────
   991→async function generateDocumentNumber(): Promise<string> {
   992→  const now = new Date();
   993→  const year = now.getFullYear();
   994→  const month = String(now.getMonth() + 1).padStart(2, '0');
   995→
   996→  // Count existing resumes this month to generate sequence
   997→  const startOfMonth = new Date(year, now.getMonth(), 1);
   998→  const startOfNextMonth = new Date(year, now.getMonth() + 1, 1);
   999→
  1000→  const count = await db.palliativeResume.count({
  1001→    where: {
  1002→      createdAt: {
  1003→        gte: startOfMonth,
  1004→        lt: startOfNextMonth,
  1005→      },
  1006→    },
  1007→  });
  1008→
  1009→  const sequence = String(count + 1).padStart(4, '0');
  1010→  return `RPM/${year}/${month}/${sequence}`;
  1011→}

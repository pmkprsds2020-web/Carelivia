    85→function RadioOption<T extends string>({
    86→  options,
    87→  labels,
    88→  value,
    89→  onChange,
    90→  colorMap,
    91→}: {
    92→  options: T[];
    93→  labels: Record<T, string>;
    94→  value: T | '';
    95→  onChange: (val: T) => void;
    96→  colorMap?: Partial<Record<T, string>>;
    97→}) {
    98→  return (
    99→    <div className="flex flex-wrap gap-2">
   100→      {options.map((opt) => {
   101→        const isSelected = value === opt;
   102→        const colorCls = colorMap?.[opt];
   103→        return (
   104→          <button
   105→            key={opt}
   106→            type="button"
   107→            onClick={() => onChange(opt)}
   108→            className={`px-3 py-2 rounded-lg border text-sm transition-all whitespace-nowrap
   109→              ${isSelected
   110→                ? colorCls || 'bg-primary text-primary-foreground border-primary shadow-sm'
   111→                : 'bg-white hover:bg-slate-50 border-slate-200 text-slate-700'}
   112→              `}
   113→          >
   114→            {labels[opt]}
   115→          </button>
   116→        );
   117→      })}
   118→    </div>
   119→  );
   120→}

   124→function ConditionalTextarea({

   132→  label: string;

   153→function DailyComplaintForm({ palliativePatientId }: { palliativePatientId: string }) {
   154→  const { toast } = useToast();

   156→  const addPalliativeClinicalAlert = useStore((s) => s.addPalliativeClinicalAlert);
   157→  const addPalliativeMonitoringNotification = useStore((s) => s.addPalliativeMonitoringNotification);
   158→  const palliativePatients = useStore((s) => s.palliativePatients);
   159→
   160→  const [submitting, setSubmitting] = useState(false);
   161→  const [submitted, setSubmitted] = useState(false);
   162→
   163→  // Form state
   164→  const [kondisiHariIni, setKondisiHariIni] = useState<DailyCondition | ''>('');
   165→  const [alasanKondisi, setAlasanKondisi] = useState('');
   166→  const [keluhanBaru, setKeluhanBaru] = useState<DailyComplaintYesNo | ''>('');
   167→  const [deskripsiKeluhanBaru, setDeskripsiKeluhanBaru] = useState('');
   168→  const [kondisiNyeri, setKondisiNyeri] = useState<DailyPainCondition | ''>('');
   169→  const [kondisiSesak, setKondisiSesak] = useState<DailyDyspneaCondition | ''>('');
   170→  const [makanMinum, setMakanMinum] = useState<DailyYesNo | ''>('');
   171→  const [alasanMakanMinum, setAlasanMakanMinum] = useState('');
   172→  const [tidur, setTidur] = useState<DailyYesNo | ''>('');
   173→  const [alasanTidur, setAlasanTidur] = useState('');
   174→  const [masalahObat, setMasalahObat] = useState<DailyMedicineProblem | ''>('');
   175→  const [deskripsiMasalahObat, setDeskripsiMasalahObat] = useState('');
   176→
   177→  const patient = palliativePatients.find((p) => p.id === palliativePatientId);
   178→  const patientName = patient?.patientName || 'Pasien';
   179→
   180→  const canSubmit = useMemo(() => {
   181→    if (!kondisiHariIni || !keluhanBaru || !kondisiNyeri || !kondisiSesak || !makanMinum || !tidur || !masalahObat) return false;
   182→    if (kondisiHariIni === 'tidak_baik' && !alasanKondisi.trim()) return false;
   183→    if (keluhanBaru === 'ada' && !deskripsiKeluhanBaru.trim()) return false;
   184→    if (makanMinum === 'tidak' && !alasanMakanMinum.trim()) return false;
   185→    if (tidur === 'tidak' && !alasanTidur.trim()) return false;
   186→    if (masalahObat === 'ya' && !deskripsiMasalahObat.trim()) return false;
   187→    return true;
   188→  }, [kondisiHariIni, keluhanBaru, kondisiNyeri, kondisiSesak, makanMinum, tidur, masalahObat,
   189→      alasanKondisi, deskripsiKeluhanBaru, alasanMakanMinum, alasanTidur, deskripsiMasalahObat]);
   190→
   191→  const handleSubmit = async () => {
   192→    if (!canSubmit) return;
   193→    setSubmitting(true);
   194→
   195→    try {
   196→      const formData: DailyComplaintFormInput = {
   197→        palliativePatientId,
   198→        kondisiHariIni: kondisiHariIni as DailyCondition,
   199→        alasanKondisi: kondisiHariIni === 'tidak_baik' ? alasanKondisi : undefined,
   200→        keluhanBaru: keluhanBaru as DailyComplaintYesNo,
   201→        deskripsiKeluhanBaru: keluhanBaru === 'ada' ? deskripsiKeluhanBaru : undefined,
   202→        kondisiNyeri: kondisiNyeri as DailyPainCondition,
   203→        kondisiSesak: kondisiSesak as DailyDyspneaCondition,
   204→        makanMinum: makanMinum as DailyYesNo,
   205→        alasanMakanMinum: makanMinum === 'tidak' ? alasanMakanMinum : undefined,
   206→        tidur: tidur as DailyYesNo,
   207→        alasanTidur: tidur === 'tidak' ? alasanTidur : undefined,
   208→        masalahObat: masalahObat as DailyMedicineProblem,
   209→        deskripsiMasalahObat: masalahObat === 'ya' ? deskripsiMasalahObat : undefined,
   210→      };
   211→
   212→      const res = await fetch('/api/daily-complaints', {
   213→        method: 'POST',
   214→        headers: { 'Content-Type': 'application/json' },
   215→        body: JSON.stringify(formData),
   216→      });
   217→
   218→      if (!res.ok) throw new Error('Gagal menyimpan keluhan harian');
   219→
   220→      const data = await res.json();
   221→      const savedComplaint: DailyComplaintRecord = {
   222→        ...data.complaint,
   223→        submittedAt: data.complaint.submittedAt,
   224→        createdAt: data.complaint.createdAt,
   225→      };
   226→
   227→      // Add to store
   228→      addDailyComplaint(savedComplaint);
   229→
   230→      // Generate clinical alerts in the store
   231→      if (data.alerts && data.alerts.length > 0) {
   232→        for (const alert of data.alerts) {
   233→          addPalliativeClinicalAlert({
   234→            id: `alert-dc-${Date.now()}-${Math.random().toString(36).substring(2, 7)}`,
   235→            patientId: palliativePatientId,
   236→            alertType: alert.severity === 'merah' ? 'perburukan' : 'gejala_berat',
   237→            severity: alert.severity === 'merah' ? 'merah' : 'kuning',
   238→            title: alert.title,
   239→            description: alert.message,
   240→            isRead: false,
   241→            createdAt: new Date().toISOString(),
   242→          });
   243→
   244→          addPalliativeMonitoringNotification({
   245→            id: `notif-dc-${Date.now()}-${Math.random().toString(36).substring(2, 7)}`,
   246→            patientId: palliativePatientId,
   247→            patientName,
   248→            type: alert.severity === 'merah' ? 'pain_increase' : 'screening_completed',
   249→            title: alert.title,
   250→            description: alert.message,
   251→            severity: alert.severity === 'merah' ? 'critical' : 'warning',
   252→            isRead: false,
   253→            referenceId: savedComplaint.id,
   254→            createdAt: new Date().toISOString(),
   255→          });
   256→        }
   257→      }
   258→
   259→      setSubmitted(true);
   260→      toast({
   261→        title: 'Keluhan Harian Terkirim',
   262→        description: 'Terima kasih telah mengisi keluhan harian. Tim paliatif akan memantau kondisi Anda.',
   263→      });
   264→    } catch (error) {
   265→      toast({
   266→        title: 'Gagal Mengirim',
   267→        description: 'Terjadi kesalahan saat menyimpan keluhan. Silakan coba lagi.',
   268→        variant: 'destructive',
   269→      });
   270→    } finally {
   271→      setSubmitting(false);
   272→    }
   273→  };
   274→
   275→  if (submitted) {
   276→    return (
   277→      <Card className="border-emerald-200 bg-emerald-50/50">
   278→        <CardContent className="py-12 flex flex-col items-center gap-4 text-center">
   279→          <div className="h-16 w-16 rounded-full bg-emerald-100 flex items-center justify-center">
   280→            <CheckCircle2 className="h-8 w-8 text-emerald-600" />
   281→          </div>
   282→          <div>
   283→            <h3 className="text-lg font-semibold text-emerald-900">Keluhan Harian Berhasil Dikirim</h3>
   284→            <p className="text-sm text-emerald-700 mt-1">
   285→              Terima kasih telah mengisi keluhan harian. Tim paliatif akan memantau kondisi Anda.
   286→            </p>
   287→          </div>
   288→          <Button
   289→            variant="outline"
   290→            onClick={() => {
   291→              setSubmitted(false);
   292→              setKondisiHariIni('');
   293→              setAlasanKondisi('');
   294→              setKeluhanBaru('');
   295→              setDeskripsiKeluhanBaru('');
   296→              setKondisiNyeri('');
   297→              setKondisiSesak('');
   298→              setMakanMinum('');
   299→              setAlasanMakanMinum('');
   300→              setTidur('');
   301→              setAlasanTidur('');
   302→              setMasalahObat('');
   303→              setDeskripsiMasalahObat('');
   304→            }}
   305→            className="mt-2"
   306→          >
   307→            Isi Keluhan Baru
   308→          </Button>
   309→        </CardContent>
   310→      </Card>
   311→    );
   312→  }
   313→
   314→  return (
   315→    <Card>
   316→      <CardHeader className="pb-4">
   317→        <div className="flex items-center gap-2">
   318→          <ClipboardList className="h-5 w-5 text-primary" />
   319→          <CardTitle className="text-lg">Form Keluhan Harian</CardTitle>
   320→        </div>
   321→        <CardDescription>
   322→          Isilah form berikut untuk melaporkan kondisi Anda hari ini. Data Anda akan dipantau oleh tim paliatif.
   323→        </CardDescription>
   324→      </CardHeader>
   325→      <CardContent className="space-y-6">
   326→        {/* Q1: Kondisi Hari Ini */}
   327→        <div className="space-y-2">
   328→          <Label className="text-sm font-semibold flex items-center gap-1.5">
   329→            <Smile className="h-4 w-4 text-teal-600" />
   330→            1. Bagaimana kondisi Anda hari ini?
   331→          </Label>
   332→          <RadioOption
   333→            options={['baik', 'tidak_baik'] as DailyCondition[]}
   334→            labels={conditionLabels}
   335→            value={kondisiHariIni}
   336→            onChange={setKondisiHariIni}

   338→              baik: 'bg-emerald-600 text-white border-emerald-600 shadow-sm',

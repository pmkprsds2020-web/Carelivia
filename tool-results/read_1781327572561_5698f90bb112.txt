    65→function formatDate(dateStr: string): string {
    66→  try {
    67→    return new Date(dateStr).toLocaleDateString('id-ID', {
    68→      day: 'numeric',
    69→      month: 'long',
    70→      year: 'numeric',
    71→    });
    72→  } catch {
    73→    return dateStr;
    74→  }
    75→}
    76→
    77→function formatDateTime(dateStr: string): string {
    78→  try {
    79→    return new Date(dateStr).toLocaleDateString('id-ID', {
    80→      day: 'numeric',
    81→      month: 'long',
    82→      year: 'numeric',
    83→      hour: '2-digit',
    84→      minute: '2-digit',
    85→    });
    86→  } catch {
    87→    return dateStr;
    88→  }
    89→}
    90→
    91→function formatTime(dateStr: string): string {
    92→  try {
    93→    return new Date(dateStr).toLocaleTimeString('id-ID', {
    94→      hour: '2-digit',
    95→      minute: '2-digit',
    96→    });
    97→  } catch {
    98→    return '';
    99→  }
   100→}

   103→  return `${prefix}-${Date.now()}-${Math.random().toString(36).substring(2, 7)}`;

   472→function SummaryCard({
   473→  icon,
   474→  label,
   475→  value,
   476→  onClick,
   477→  color,
   478→}: {
   479→  icon: React.ReactNode;

   481→  value: string;

   484→}) {
   485→  const colorMap = {
   486→    teal: 'border-l-teal-500 hover:bg-teal-50/50',
   487→    sky: 'border-l-sky-500 hover:bg-sky-50/50',
   488→    amber: 'border-l-amber-500 hover:bg-amber-50/50',

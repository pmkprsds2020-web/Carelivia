   158→// ─── Main Component ─────────────────────────────────────────────────────────
   159→
   160→export default function AISocialAnalysisTab({ palliativePatientId }: AISocialAnalysisTabProps) {
   161→  const { toast } = useToast();
   162→
   163→  const {
   164→    palliativePatients,
   165→    socialAssessments,
   166→    caregivers,
   167→    familyMeetings,
   168→    financialSupportRecords,
   169→    transportRecords,
   170→    palliativeScreeningRecords,
   171→    vitalSignRecords,
   172→    aiSocialAnalysisResult,
   173→    setAiSocialAnalysisResult,
   174→    aiSocialAnalysisLoading,
   175→    setAiSocialAnalysisLoading,
   176→    aiSocialAnalysisRecords,
   177→    addAiSocialAnalysisRecord,
   178→    aiSocialPopulationStats,
   179→    setAiSocialPopulationStats,
   180→    currentUser,
   181→    addSocialAlert,
   182→  } = useStore();
   183→
   184→  const [activeSection, setActiveSection] = useState('overview');
   185→  const [actionDecisions, setActionDecisions] = useState<Record<string, 'accepted' | 'rejected' | 'pending'>>({});
   186→  const [analysisNotes, setAnalysisNotes] = useState('');
   187→
   188→  // ─── Computed Data ─────────────────────────────────────────────────────
   189→
   190→  const patient = useMemo(
   191→    () => palliativePatients.find(p => p.id === palliativePatientId),
   192→    [palliativePatients, palliativePatientId]
   193→  );
   194→
   195→  const latestSocialAssessment = useMemo(() => {
   196→    const assessments = socialAssessments
   197→      .filter(a => a.palliativePatientId === palliativePatientId)
   198→      .sort((a, b) => new Date(b.assessedAt).getTime() - new Date(a.assessedAt).getTime());
   199→    return assessments[0] || null;
   200→  }, [socialAssessments, palliativePatientId]);
   201→
   202→  const patientCaregivers = useMemo(
   203→    () => caregivers.filter(c => c.palliativePatientId === palliativePatientId && c.isActive),
   204→    [caregivers, palliativePatientId]
   205→  );
   206→
   207→  const patientMeetings = useMemo(
   208→    () => familyMeetings.filter(m => m.palliativePatientId === palliativePatientId),
   209→    [familyMeetings, palliativePatientId]
   210→  );
   211→
   212→  const patientFinancials = useMemo(
   213→    () => financialSupportRecords.filter(f => f.palliativePatientId === palliativePatientId),
   214→    [financialSupportRecords, palliativePatientId]
   215→  );
   216→
   217→  const patientTransport = useMemo(
   218→    () => transportRecords.filter(t => t.palliativePatientId === palliativePatientId),
   219→    [transportRecords, palliativePatientId]
   220→  );
   221→
   222→  const patientScreenings = useMemo(
   223→    () => palliativeScreeningRecords.filter(s => s.palliativePatientId === palliativePatientId)
   224→      .sort((a, b) => new Date(b.performedAt).getTime() - new Date(a.performedAt).getTime()),
   225→    [palliativeScreeningRecords, palliativePatientId]
   226→  );
   227→
   228→  const patientVitals = useMemo(
   229→    () => vitalSignRecords.filter(v => v.palliativePatientId === palliativePatientId)
   230→      .sort((a, b) => new Date(b.recordedAt).getTime() - new Date(a.recordedAt).getTime()),
   231→    [vitalSignRecords, palliativePatientId]
   232→  );
   233→
   234→  const previousAnalysis = useMemo(() => {
   235→    const records = aiSocialAnalysisRecords
   236→      .filter(r => r.palliativePatientId === palliativePatientId)
   237→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   238→    return records[0] || null;
   239→  }, [aiSocialAnalysisRecords, palliativePatientId]);
   240→
   241→  // ─── AI Analysis Handler ──────────────────────────────────────────────
   242→
   243→  const handleRunAnalysis = useCallback(async () => {
   244→    if (!palliativePatientId || !patient) return;
   245→
   246→    setAiSocialAnalysisLoading(true);
   247→    setAiSocialAnalysisResult(null);
   248→
   249→    try {
   250→      const requestData = {
   251→        patientData: {
   252→          name: patient.patientName,
   253→          age: patient.dateOfBirth ? new Date().getFullYear() - new Date(patient.dateOfBirth).getFullYear() : undefined,
   254→          gender: patient.gender,
   255→          diagnosis: patient.primaryDiagnosis,
   256→          diseaseStage: patient.diseaseStage,
   257→          careStatus: patient.careStatus,
   258→          riskLevel: patient.riskLevel,
   259→          address: patient.address,
   260→          familyContactName: patient.familyContactName,
   261→          familyContactRelation: patient.familyContactRelation,
   262→          bpjsNumber: patient.bpjsNumber,
   263→        },
   264→        socialData: latestSocialAssessment ? {
   265→          housingCondition: latestSocialAssessment.housingCondition,
   266→          caregiverAvailability: latestSocialAssessment.caregiverAvailability,
   267→          familySupportLevel: latestSocialAssessment.familySupportLevel,
   268→          transportDifficulty: latestSocialAssessment.transportDifficulty,
   269→          economicConstraint: latestSocialAssessment.economicConstraint,
   270→          healthcareAccess: latestSocialAssessment.healthcareAccess,
   271→          medicalEquipmentNeed: latestSocialAssessment.medicalEquipmentNeed,
   272→          socialAssistanceNeed: latestSocialAssessment.socialAssistanceNeed,
   273→          socialIsolationRisk: latestSocialAssessment.socialIsolationRisk,
   274→          priorityLevel: latestSocialAssessment.priorityLevel,
   275→        } : null,
   276→        screeningData: patientScreenings.map(s => ({
   277→          type: s.screeningType,
   278→          score: s.score,
   279→          label: s.scoreLabel,
   280→          interpretation: s.interpretation,
   281→        })),
   282→        caregiverData: patientCaregivers.map(c => ({
   283→          name: c.name,
   284→          role: c.role,
   285→          relation: c.relation,
   286→          zaritScore: c.zaritScore,
   287→          zaritLevel: c.zaritLevel,
   288→          familyApgarScore: c.familyApgarScore,
   289→          familyApgarLevel: c.familyApgarLevel,
   290→          schedule: c.schedule,
   291→          tasks: c.tasks,
   292→        })),
   293→        financialData: patientFinancials.length > 0 ? {
   294→          bpjsStatus: patientFinancials[0].bpjsMembershipStatus,
   295→          insuranceStatus: patientFinancials[0].healthInsuranceStatus,
   296→          assistanceNeeds: patientFinancials[0].socialAssistanceNeeds,
   297→        } : null,
   298→        transportData: patientTransport.length > 0 ? {
   299→          transportNeeds: patientTransport[0].transportNeedType,
   300→          mobilityBarriers: patientTransport[0].notes,
   301→        } : null,
   302→        meetingData: {
   303→          totalMeetings: patientMeetings.length,
   304→          lastMeetingDate: patientMeetings[0]?.scheduledAt,
   305→          followUpActions: patientMeetings[0]?.followUpActions,
   306→        },
   307→      };
   308→
   309→      const response = await fetch('/api/palliative-social-ai', {
   310→        method: 'POST',
   311→        headers: { 'Content-Type': 'application/json' },
   312→        body: JSON.stringify(requestData),
   313→      });
   314→
   315→      if (!response.ok) {
   316→        throw new Error('Gagal menghasilkan analisis AI');
   317→      }
   318→
   319→      const result: AISocialAnalysisResult = await response.json();
   320→      setAiSocialAnalysisResult(result);
   321→
   322→      // Auto-generate early warning alerts
   323→      if (result.earlyWarnings && result.earlyWarnings.length > 0) {
   324→        result.earlyWarnings.forEach((ew) => {
   325→          addSocialAlert({
   326→            id: genId('sal'),
   327→            patientId: palliativePatientId,
   328→            patientName: patient.patientName,
   329→            type: 'dukungan_keluarga',
   330→            severity: ew.severity,
   331→            title: ew.title,
   332→            description: ew.description,
   333→            isRead: false,
   334→            createdAt: new Date().toISOString(),
   335→          });
   336→        });
   337→      }
   338→
   339→      // Save analysis record
   340→      addAiSocialAnalysisRecord({
   341→        id: genId('aiar'),
   342→        palliativePatientId,
   343→        result,
   344→        generatedBy: currentUser?.name || 'Tim Paliatif',
   345→        acceptedActions: [],
   346→        rejectedActions: [],
   347→        modifiedActions: [],
   348→        createdAt: new Date().toISOString(),
   349→      });
   350→
   351→      toast({
   352→        title: 'Analisis AI Selesai',
   353→        description: 'Analisis kebutuhan sosial berhasil dihasilkan',
   354→      });
   355→    } catch (error) {
   356→      console.error('AI Social Analysis error:', error);
   357→      toast({
   358→        title: 'Analisis Gagal',
   359→        description: 'Gagal menghasilkan analisis AI. Silakan coba lagi.',
   360→        variant: 'destructive',
   361→      });
   362→    } finally {
   363→      setAiSocialAnalysisLoading(false);
   364→    }
   365→  }, [palliativePatientId, patient, latestSocialAssessment, patientScreenings, patientCaregivers, patientFinancials, patientTransport, patientMeetings, setAiSocialAnalysisResult, setAiSocialAnalysisLoading, addAiSocialAnalysisRecord, addSocialAlert, currentUser, toast]);
   366→
   367→  // ─── Population Analytics Handler ──────────────────────────────────────
   368→
   369→  const handleLoadPopulationStats = useCallback(() => {
   370→    // Generate population statistics from existing data
   371→    const activePatients = palliativePatients.filter(p => p.patientStatus !== 'program_selesai');
   372→    const highRisk = activePatients.filter(p => p.riskLevel === 'merah').length;
   373→    const medRisk = activePatients.filter(p => p.riskLevel === 'kuning').length;
   374→    const lowRisk = activePatients.filter(p => p.riskLevel === 'hijau').length;
   375→
   376→    // Count caregiver burnout
   377→    const burnoutCaregivers = caregivers.filter(c => c.zaritLevel === 'beban_berat' || c.zaritLevel === 'beban_sedang').length;
   378→
   379→    // Count social needs
   380→    const needCounts: Record<string, number> = {};
   381→    socialAssessments.forEach(a => {
   382→      if (a.economicConstraint !== 'tidak_ada') needCounts['Bantuan Finansial'] = (needCounts['Bantuan Finansial'] || 0) + 1;
   383→      if (a.transportDifficulty !== 'tidak_ada' && a.transportDifficulty !== 'ringan') needCounts['Transportasi'] = (needCounts['Transportasi'] || 0) + 1;
   384→      if (a.medicalEquipmentNeed !== 'tidak_ada') needCounts['Alat Kesehatan'] = (needCounts['Alat Kesehatan'] || 0) + 1;
   385→      if (a.socialIsolationRisk !== 'rendah') needCounts['Pendampingan Sosial'] = (needCounts['Pendampingan Sosial'] || 0) + 1;
   386→      if (a.healthcareAccess !== 'mudah' && a.healthcareAccess !== 'cukup') needCounts['Akses Layanan'] = (needCounts['Akses Layanan'] || 0) + 1;
   387→    });
   388→
   389→    const topNeeds = Object.entries(needCounts)
   390→      .map(([need, count]) => ({ need, count }))
   391→      .sort((a, b) => b.count - a.count);
   392→
   393→    const months = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun'];
   394→    const trendData = months.map(month => ({
   395→      month,
   396→      highRisk: Math.max(1, highRisk + Math.floor(Math.random() * 3 - 1)),
   397→      mediumRisk: Math.max(1, medRisk + Math.floor(Math.random() * 3 - 1)),
   398→      lowRisk: Math.max(1, lowRisk + Math.floor(Math.random() * 3 - 1)),
   399→    }));
   400→
   401→    const stats = {
   402→      totalActivePatients: activePatients.length,
   403→      highSocialRiskCount: highRisk,
   404→      caregiverBurnoutCount: burnoutCaregivers,
   405→      topSocialNeeds: topNeeds,
   406→      familySupportDistribution: [
   407→        { level: 'Kuat', count: activePatients.length > 0 ? Math.floor(activePatients.length * 0.3) : 0 },
   408→        { level: 'Cukup', count: activePatients.length > 0 ? Math.floor(activePatients.length * 0.4) : 0 },
   409→        { level: 'Lemah', count: activePatients.length > 0 ? Math.floor(activePatients.length * 0.2) : 0 },
   410→        { level: 'Tidak Ada', count: activePatients.length > 0 ? Math.floor(activePatients.length * 0.1) : 0 },
   411→      ],
   412→      socialTrendData: trendData,
   413→      predictedNeeds30Days: topNeeds.slice(0, 4).map(n => ({ category: n.need, estimatedCount: Math.ceil(n.count * 1.1) })),
   414→      predictedNeeds90Days: topNeeds.slice(0, 4).map(n => ({ category: n.need, estimatedCount: Math.ceil(n.count * 1.3) })),
   415→      generatedAt: new Date().toISOString(),
   416→    };
   417→
   418→    setAiSocialPopulationStats(stats);
   419→  }, [palliativePatients, caregivers, socialAssessments, setAiSocialPopulationStats]);
   420→
   421→  // ─── Action Decision Handler ──────────────────────────────────────────
   422→
   423→  const handleActionDecision = useCallback((actionId: string, decision: 'accepted' | 'rejected') => {
   424→    setActionDecisions(prev => ({ ...prev, [actionId]: decision }));
   425→    toast({
   426→      title: decision === 'accepted' ? 'Tindakan Diterima' : 'Tindakan Ditolak',
   427→      description: decision === 'accepted' ? 'Tindakan telah ditambahkan ke rencana' : 'Tindakan telah ditolak',
   428→    });
   429→  }, [toast]);
   430→
   431→  // ─── Empty State ──────────────────────────────────────────────────────
   432→
   433→  if (!palliativePatientId) {
   434→    return (
   435→      <div className="flex flex-col items-center justify-center h-64 text-slate-500 gap-3">
   436→        <Brain className="h-12 w-12 opacity-40" />
   437→        <p className="text-lg font-medium">Pilih pasien terlebih dahulu</p>
   438→        <p className="text-sm">Pilih pasien paliatif untuk menjalankan analisis AI</p>
   439→      </div>
   440→    );
   441→  }
   442→
   443→  // ─── Section renderers ────────────────────────────────────────────────
   444→
   445→  // Section 1: Overview / Social Needs Assessment
   446→  const renderOverview = () => (
   447→    <div className="space-y-4">
   448→      {/* Analysis Trigger */}
   449→      <Card className="border-dashed border-2 border-teal-300 bg-teal-50/30">
   450→        <CardContent className="pt-6">
   451→          <div className="flex flex-col sm:flex-row items-center gap-4">
   452→            <div className="h-14 w-14 rounded-full bg-teal-100 flex items-center justify-center">
   453→              <Brain className="h-7 w-7 text-teal-600" />
   454→            </div>
   455→            <div className="flex-1 text-center sm:text-left">
   456→              <h3 className="font-semibold text-lg">Analisis AI Kebutuhan Sosial</h3>
   457→              <p className="text-sm text-slate-600 mt-1">
   458→                AI akan menganalisis data demografi, skrining sosial, ESAS-r, PPS, Distress Thermometer,
   459→                Zarit Caregiver Burden, Family APGAR, monitoring gejala, kepatuhan pengobatan,
   460→                riwayat kunjungan, dan data ekonomi untuk menghasilkan analisis komprehensif.
   461→              </p>
   462→            </div>
   463→            <Button
   464→              onClick={handleRunAnalysis}
   465→              disabled={aiSocialAnalysisLoading}
   466→              className="bg-teal-600 hover:bg-teal-700 text-white gap-2 shrink-0"
   467→            >
   468→              {aiSocialAnalysisLoading ? (
   469→                <><RefreshCw className="h-4 w-4 animate-spin" /> Menganalisis...</>
   470→              ) : (
   471→                <><Sparkles className="h-4 w-4" /> Jalankan Analisis AI</>
   472→              )}
   473→            </Button>
   474→          </div>
   475→          {previousAnalysis && !aiSocialAnalysisResult && (

   239→const calculateSocialScore = (a: SocialAssessmentRecord): number => {
   240→  const fields: string[] = [
   241→    a.housingCondition, a.caregiverAvailability, a.familySupportLevel,
   242→    a.transportDifficulty, a.economicConstraint, a.healthcareAccess,
   243→    a.medicalEquipmentNeed, a.socialAssistanceNeed, a.socialIsolationRisk,
   244→  ];
   245→  return fields.reduce((sum, f) => sum + (scoreMap[f] ?? 0), 0);
   246→};

   248→const calculatePriority = (form: ScreeningFormState): SocialScreeningPriority => {
   249→  const highVals = ['berat', 'tidak_tersedia', 'sangat_sulit', 'tidak_layak', 'tinggi', 'tidak_ada'];
   250→  const medVals = ['sedang', 'terbatas', 'sulit', 'kurang_layak', 'lemah'];
   251→  const vals = Object.values(form).map(v => v.value);
   252→  if (vals.some(v => highVals.includes(v))) return 'tinggi';
   253→  if (vals.some(v => medVals.includes(v))) return 'sedang';
   254→  return 'rendah';
   255→};

   357→
   358→  const [activeTab, setActiveTab] = useState('dashboard');
   359→
   360→  // ─── Horizontal Tab Scroll State ──────────────────────────────────────────
   361→  const tabScrollRef = useRef<HTMLDivElement>(null);
   362→  const [canScrollLeft, setCanScrollLeft] = useState(false);
   363→  const [canScrollRight, setCanScrollRight] = useState(false);
   364→
   365→  const updateScrollIndicators = useCallback(() => {
   366→    const el = tabScrollRef.current;
   367→    if (!el) return;
   368→    const left = el.scrollLeft;
   369→    const maxScroll = el.scrollWidth - el.clientWidth;
   370→    setCanScrollLeft(left > 2);
   371→    setCanScrollRight(left < maxScroll - 2);
   372→  }, []);
   373→
   374→  useEffect(() => {
   375→    const el = tabScrollRef.current;
   376→    if (!el) return;
   377→    updateScrollIndicators();
   378→    el.addEventListener('scroll', updateScrollIndicators, { passive: true });
   379→    const observer = new ResizeObserver(updateScrollIndicators);
   380→    observer.observe(el);
   381→    return () => {
   382→      el.removeEventListener('scroll', updateScrollIndicators);
   383→      observer.disconnect();
   384→    };
   385→  }, [updateScrollIndicators, palliativePatientId]);
   386→
   387→  // Scroll active tab into view
   388→  useEffect(() => {
   389→    const el = tabScrollRef.current;
   390→    if (!el) return;
   391→    const activeTrigger = el.querySelector('[data-state="active"]') as HTMLElement | null;
   392→    if (activeTrigger) {
   393→      activeTrigger.scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'nearest' });
   394→    }
   395→  }, [activeTab]);
   396→
   397→  // ─── Filtered Data ──────────────────────────────────────────────────────
   398→

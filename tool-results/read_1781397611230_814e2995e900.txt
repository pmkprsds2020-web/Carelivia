   157→function formatDate(dateStr: string): string {
   158→  try {
   159→    return new Date(dateStr).toLocaleDateString('id-ID', {
   160→      day: 'numeric',
   161→      month: 'short',
   162→      year: 'numeric',
   163→    });
   164→  } catch {
   165→    return dateStr;
   166→  }
   167→}
   168→
   169→function formatDateTime(dateStr: string): string {
   170→  try {
   171→    return new Date(dateStr).toLocaleDateString('id-ID', {
   172→      day: 'numeric',
   173→      month: 'short',
   174→      year: 'numeric',
   175→      hour: '2-digit',
   176→      minute: '2-digit',
   177→    });
   178→  } catch {
   179→    return dateStr;
   180→  }
   181→}

   266→function calcBmi(weight?: number, height?: number): number | undefined {
   267→  if (!weight || !height || height <= 0) return undefined;
   268→  const heightM = height / 100;
   269→  return Math.round((weight / (heightM * heightM)) * 10) / 10;

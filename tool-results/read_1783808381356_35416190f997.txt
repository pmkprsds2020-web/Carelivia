     1→     1→// ───────────────────────────────────────────────────────────────────────────
     2→     2→// SupabaseSyncProvider — keeps the Zustand store in sync with Supabase.
     3→     3→//
     4→     4→// Responsibilities:
     5→     5→//   1. On mount, load all Monitoring-Paliatif data from Supabase via the 13
     6→     6→//      service files and populate the Zustand store. Local demo data wins on
     7→     7→//      conflict — we only replace the store if Supabase returns non-empty
     8→     8→//      arrays.
     9→     9→//   2. Subscribe to Supabase Realtime for 12 tables so any DB change is
    10→    10→//      mirrored into the store immediately (INSERT/UPDATE/DELETE).
    11→    11→//   3. ALWAYS render children immediately. Supabase loading is fire-and-forget
    12→    12→//      and never blocks the UI. If Supabase is unreachable (missing tables,
    13→    13→//      RLS denied, network down), we log a warning and continue with local
    14→    14→//      Zustand data.
    15→    15→//   4. NEVER throw — every subscription and load is wrapped in try/catch.
    16→    16→// ───────────────────────────────────────────────────────────────────────────

    43→    43→/** Check whether a row id already exists in a store array. */
    44→    44→function hasId<T extends { id: string }>(list: T[], id: string): boolean {
    45→    45→  return list.some((x) => x.id === id);
    46→    46→}
    47→    47→
    48→    48→/** Log a non-fatal sync warning. */
    49→    49→function warn(label: string, err: unknown) {
    50→    50→  console.warn(`[SupabaseSync] ${label}:`, err);
    51→    51→}

    53→    53→// ─── Initial-load handlers ─────────────────────────────────────────────────
    54→    54→//
    55→    55→// Each handler is self-contained: it loads data from Supabase, validates it,
    56→    56→// and dispatches the corresponding Zustand action(s). Wrapped in try/catch by
    57→    57→// the caller. We only REPLACE the store if Supabase returns non-empty data —
    58→    58→// otherwise we keep the local demo data (per the task's "local-wins" rule).
    59→    59→
    60→    60→async function loadPatients(store: ReturnType<typeof useStore.getState>) {
    61→    61→  const patients = await svc.patientService.getAll();
    62→    62→  if (Array.isArray(patients) && patients.length > 0) {
    63→    63→    store.setPalliativePatients(patients as PalliativePatientInfo[]);
    64→    64→  }
    65→    65→  return patients;
    66→    66→}

    68→    68→async function loadPatientScopedData(
    69→    69→  store: ReturnType<typeof useStore.getState>,
    70→    70→  patients: PalliativePatientInfo[]
    71→    71→) {
    72→    72→  for (const patient of patients) {
    73→    73→    const pid = patient.id;
    74→    74→
    75→    75→    // Vitals
    76→    76→    try {
    77→    77→      const rows = await svc.vitalService.getAll(pid);
    78→    78→      if (Array.isArray(rows) && rows.length > 0) {
    79→    79→        // Replace existing vitals for this patient to avoid stale data,
    80→    80→        // but keep vitals from other patients intact.
    81→    81→        const others = store.vitalSignRecords.filter((v) => v.palliativePatientId !== pid);
    82→    82→        store.setVitalSignRecords([
    83→    83→          ...others,
    84→    84→          ...(rows as VitalSignRecordInfo[]),
    85→    85→        ]);
    86→    86→      }
    87→    87→    } catch (e) { warn('vitalService.getAll', e); }
    88→    88→
    89→    89→    // Screenings
    90→    90→    try {
    91→    91→      const rows = await svc.screeningService.getAll(pid);
    92→    92→      if (Array.isArray(rows) && rows.length > 0) {
    93→    93→        const others = store.palliativeScreeningRecords.filter((s) => s.palliativePatientId !== pid);
    94→    94→        store.setPalliativeScreeningRecords([
    95→    95→          ...others,
    96→    96→          ...(rows as PalliativeScreeningRecordInfo[]),
    97→    97→        ]);
    98→    98→      }
    99→    99→    } catch (e) { warn('screeningService.getAll', e); }
   100→   100→
   101→   101→    // Medications
   102→   102→    try {
   103→   103→      const rows = await svc.medicationService.getAll(pid);
   104→   104→      if (Array.isArray(rows) && rows.length > 0) {
   105→   105→        const others = store.palliativeMedications.filter((m) => m.palliativePatientId !== pid);
   106→   106→        store.setPalliativeMedications([
   107→   107→          ...others,
   108→   108→          ...(rows as PalliativeMedicationInfo[]),
   109→   109→        ]);
   110→   110→      }
   111→   111→    } catch (e) { warn('medicationService.getAll', e); }
   112→   112→
   113→   113→    // Nutrition
   114→   114→    try {
   115→   115→      const rows = await svc.nutritionService.getAll(pid);
   116→   116→      if (Array.isArray(rows) && rows.length > 0) {
   117→   117→        const others = store.nutritionRecords.filter((n) => n.palliativePatientId !== pid);
   118→   118→        store.setNutritionRecords([
   119→   119→          ...others,
   120→   120→          ...(rows as NutritionRecordInfo[]),
   121→   121→        ]);
   122→   122→      }
   123→   123→    } catch (e) { warn('nutritionService.getAll', e); }
   124→   124→
   125→   125→    // Daily complaints
   126→   126→    try {
   127→   127→      const rows = await svc.complaintService.getAll(pid);
   128→   128→      if (Array.isArray(rows) && rows.length > 0) {
   129→   129→        const others = store.dailyComplaints.filter((c) => c.palliativePatientId !== pid);
   130→   130→        store.setDailyComplaints([
   131→   131→          ...others,
   132→   132→          ...(rows as DailyComplaintRecord[]),
   133→   133→        ]);
   134→   134→      }
   135→   135→    } catch (e) { warn('complaintService.getAll', e); }
   136→   136→
   137→   137→    // Social assessments
   138→   138→    try {
   139→   139→      const rows = await svc.socialService.getAll(pid);
   140→   140→      if (Array.isArray(rows) && rows.length > 0) {
   141→   141→        const others = store.socialAssessments.filter((s) => s.palliativePatientId !== pid);
   142→   142→        const fresh = store.socialAssessments.filter((s) => s.palliativePatientId === pid);
   143→   143→        // Replace this patient's records
   144→   144→        const next = [...others];
   145→   145→        for (const r of rows as SocialAssessmentRecord[]) {
   146→   146→          if (!hasId(fresh, r.id)) next.push(r);
   147→   147→          else next.push(r); // overwrite in-place
   148→   148→        }
   149→   149→        // Use direct set since addSocialAssessment prepends (we want replace)
   150→   150→        store.socialAssessments = next;
   151→   151→        // Trigger Zustand update via the official setter (addSocialAssessment
   152→   152→        // prepends; we want a clean replace per-patient, so set directly):
   153→   153→        useStore.setState({ socialAssessments: next });
   154→   154→      }
   155→   155→    } catch (e) { warn('socialService.getAll', e); }
   156→   156→
   157→   157→    // ACP
   158→   158→    try {
   159→   159→      const rows = await svc.acpService.getAll(pid);
   160→   160→      if (Array.isArray(rows) && rows.length > 0) {
   161→   161→        const others = store.advanceCarePlans.filter((a) => a.palliativePatientId !== pid);
   162→   162→        store.setAdvanceCarePlans([
   163→   163→          ...others,
   164→   164→          ...(rows as AdvanceCarePlanInfo[]),
   165→   165→        ]);
   166→   166→      }
   167→   167→    } catch (e) { warn('acpService.getAll', e); }
   168→   168→
   169→   169→    // Clinical alerts — loaded via direct Supabase query (the sibling agent's
   170→   170→    // notificationService is for user notifications, not clinical alerts).
   171→   171→    try {
   172→   172→      const rows = await safeQuery(
   173→   173→        supabase
   174→   174→          .from('clinical_alerts')
   175→   175→          .select('*')
   176→   176→          .eq('patient_id', pid)
   177→   177→          .order('created_at', { ascending: false }),
   178→   178→        [] as any[],
   179→   179→        'loadPatientScopedData.clinical_alerts'
   180→   180→      );
   181→   181→      if (Array.isArray(rows) && rows.length > 0) {
   182→   182→        for (const r of rows as any[]) {
   183→   183→          const alert: PalliativeClinicalAlert = {
   184→   184→            id: r.id,
   185→   185→            patientId: r.patient_id ?? pid,
   186→   186→            alertType: r.alert_type ?? 'form_tidak_diisi',
   187→   187→            severity: r.severity ?? 'kuning',
   188→   188→            title: r.title ?? '',
   189→   189→            description: r.description ?? '',
   190→   190→            values: r.values ?? undefined,
   191→   191→            isRead: !!r.is_read,
   192→   192→            createdAt: r.created_at ?? new Date().toISOString(),
   193→   193→          };
   194→   194→          if (!hasId(store.palliativeClinicalAlerts, alert.id)) {
   195→   195→            store.addPalliativeClinicalAlert(alert);
   196→   196→          }
   197→   197→        }
   198→   198→      }
   199→   199→    } catch (e) { warn('clinical_alerts load', e); }
   200→   200→
   201→   201→    // Audit log — same pattern, direct Supabase query.
   202→   202→    try {
   203→   203→      const rows = await safeQuery(
   204→   204→        supabase
   205→   205→          .from('audit_log')
   206→   206→          .select('*')
   207→   207→          .eq('patient_id', pid)
   208→   208→          .order('created_at', { ascending: false }),
   209→   209→        [] as any[],
   210→   210→        'loadPatientScopedData.audit_log'
   211→   211→      );
   212→   212→      if (Array.isArray(rows) && rows.length > 0) {
   213→   213→        for (const r of rows as any[]) {
   214→   214→          const entry: PalliativeAuditEntry = {
   215→   215→            id: r.id,
   216→   216→            patientId: r.patient_id ?? pid,
   217→   217→            action: r.action ?? 'clinical_action',
   218→   218→            performedBy: r.performed_by ?? 'system',
   219→   219→            performedByRole: r.performed_by_role ?? 'system',
   220→   220→            details: r.details ?? undefined,
   221→   221→            ipAddress: r.ip_address ?? undefined,
   222→   222→            device: r.device ?? undefined,
   223→   223→            createdAt: r.created_at ?? new Date().toISOString(),
   224→   224→          };
   225→   225→          if (!hasId(store.palliativeAuditLog, entry.id)) {
   226→   226→            store.addPalliativeAuditEntry(entry);
   227→   227→          }
   228→   228→        }
   229→   229→      }
   230→   230→    } catch (e) { warn('audit_log load', e); }
   231→   231→
   232→   232→    // Chat messages — the sibling agent's chatService.getMessages takes a
   233→   233→    // `roomId` (not a patientId). To list messages per patient we'd need to
   234→   234→    // resolve the room first; the chat panel handles this lazily itself, so
   235→   235→    // we skip preloading here to avoid a costly N+1 lookup.
   236→   236→  }
   237→   237→}

   239→   239→// ─── Realtime handlers ─────────────────────────────────────────────────────
   240→   240→//
   241→   241→// Each handler receives a PostgresChangesPayload and dispatches the matching
   242→   242→// Zustand action. We dedupe by id on INSERT, replace on UPDATE, filter on
   243→   243→// DELETE. Wrapped in try/catch by the caller so a single bad event can never
   244→   244→// break the subscription.
   245→   245→
   246→   246→type RealtimePayload = {
   247→   247→  eventType: 'INSERT' | 'UPDATE' | 'DELETE' | '*';
   248→   248→  new: Record<string, any> | null;
   249→   249→  old: Record<string, any> | null;
   250→   250→  errors: string[] | null;
   251→   251→};
   252→   252→
   253→   253→function handlePatientEvent(store: ReturnType<typeof useStore.getState>, p: RealtimePayload) {
   254→   254→  const row = p.new;
   255→   255→  const id = (row?.id ?? p.old?.id) as string | undefined;
   256→   256→  if (!id) return;
   257→   257→  if (p.eventType === 'DELETE') {
   258→   258→    if (hasId(store.palliativePatients, id)) store.removePalliativePatient(id);
   259→   259→    return;
   260→   260→  }
   261→   261→  // INSERT or UPDATE — try to load the full record via the service so the
   262→   262→  // row goes through the proper snake_case → camelCase mapping.
   263→   263→  svc.patientService
   264→   264→    .getById(id)
   265→   265→    .then((fresh: PalliativePatientInfo | null) => {
   266→   266→      if (!fresh) return;
   267→   267→      const exists = hasId(useStore.getState().palliativePatients, id);
   268→   268→      if (exists) {
   269→   269→        store.updatePalliativePatient(id, fresh);
   270→   270→      } else {
   271→   271→        store.addPalliativePatient(fresh);
   272→   272→      }
   273→   273→    })
   274→   274→    .catch((e: unknown) => warn('patient realtime getById', e));
   275→   275→}

   277→   277→function handleVitalEvent(store: ReturnType<typeof useStore.getState>, p: RealtimePayload) {
   278→   278→  const row = p.new;
   279→   279→  const id = (row?.id ?? p.old?.id) as string | undefined;
   280→   280→  if (!id) return;
   281→   281→  if (p.eventType === 'DELETE') {
   282→   282→    useStore.setState((s) => ({
   283→   283→      vitalSignRecords: s.vitalSignRecords.filter((v) => v.id !== id),
   284→   284→    }));
   285→   285→    return;
   286→   286→  }
   287→   287→  // Reload via service for proper mapping
   288→   288→  svc.vitalService
   289→   289→    .getAll(row?.patient_id as string)
   290→   290→    .then((rows: VitalSignRecordInfo[]) => {
   291→   291→      const fresh = rows.find((r) => r.id === id);
   292→   292→      if (!fresh) return;
   293→   293→      const state = useStore.getState();
   294→   294→      const exists = hasId(state.vitalSignRecords, id);
   295→   295→      if (exists) {
   296→   296→        useStore.setState({
   297→   297→          vitalSignRecords: state.vitalSignRecords.map((v) => (v.id === id ? fresh : v)),
   298→   298→        });
   299→   299→      } else {
   300→   300→        store.addVitalSignRecord(fresh);
   301→   301→      }
   302→   302→    })
   303→   303→    .catch((e: unknown) => warn('vital realtime', e));
   304→   304→}

   609→   609→// ─── Provider ──────────────────────────────────────────────────────────────
   610→   610→
   611→   611→export function SupabaseSyncProvider({ children }: SupabaseSyncProviderProps) {
   612→   612→  const store = useStore();
   613→   613→  const loadedRef = useRef(false);
   614→   614→  const channelRef = useRef<ReturnType<typeof supabase.channel> | null>(null);
   615→   615→
   616→   616→  useEffect(() => {
   617→   617→    if (loadedRef.current) return;
   618→   618→    loadedRef.current = true;
   619→   619→
   620→   620→    // Snapshot the store once — actions on this snapshot remain stable for
   621→   621→    // the lifetime of the provider, even if the component re-renders.
   622→   622→    const s = useStore.getState;
   623→   623→
   624→   624→    // ── 1. Initial load (fire-and-forget, never blocks UI) ────────────────
   625→   625→    (async () => {
   626→   626→      try {
   627→   627→        const patients = await loadPatients(s());
   628→   628→        if (Array.isArray(patients) && patients.length > 0) {
   629→   629→          await loadPatientScopedData(s(), patients as PalliativePatientInfo[]);
   630→   630→        }
   631→   631→        console.log('[SupabaseSync] initial load complete');
   632→   632→      } catch (e) {
   633→   633→        warn('initial load skipped', e);
   634→   634→      }
   635→   635→    })();
   636→   636→
   637→   637→    // ── 2. Realtime subscriptions (best-effort, silently skipped on error) ─
   638→   638→    try {
   639→   639→      const channel = supabase.channel('carelivia-realtime');
   640→   640→      channelRef.current = channel;
   641→   641→
   642→   642→      const tableHandlers: Array<[string, (store: ReturnType<typeof s>, p: RealtimePayload) => void]> = [
   643→   643→        ['patients', handlePatientEvent],
   644→   644→        ['vital_signs', handleVitalEvent],
   645→   645→        ['screenings', handleScreeningEvent],
   646→   646→        ['medications', handleMedicationEvent],
   647→   647→        ['nutrition', handleNutritionEvent],
   648→   648→        ['daily_complaints', handleComplaintEvent],
   649→   649→        ['social_assessments', handleSocialEvent],
   650→   650→        ['acp', handleAcpEvent],
   651→   651→        ['chat_rooms', handleChatRoomEvent],
   652→   652→        ['messages', handleMessageEvent],
   653→   653→        ['clinical_alerts', handleClinicalAlertEvent],
   654→   654→        ['audit_log', handleAuditLogEvent],
   655→   655→        ['notifications', handleNotificationEvent],
   656→   656→      ];
   657→   657→
   658→   658→      let builder = channel;
   659→   659→      for (const [table, handler] of tableHandlers) {
   660→   660→        try {
   661→   661→          builder = builder.on(
   662→   662→            'postgres_changes',
   663→   663→            { event: '*', schema: 'public', table },
   664→   664→            (payload: any) => {
   665→   665→              try {
   666→   666→                handler(s(), {
   667→   667→                  eventType: payload.eventType,
   668→   668→                  new: payload.new,
   669→   669→                  old: payload.old,
   670→   670→                  errors: payload.errors,
   671→   671→                });
   672→   672→              } catch (e) {
   673→   673→                warn(`realtime handler for ${table}`, e);
   674→   674→              }
   675→   675→            }
   676→   676→          );
   677→   677→        } catch (e) {
   678→   678→          warn(`subscribe to ${table} skipped`, e);
   679→   679→        }
   680→   680→      }
   681→   681→
   682→   682→      try {
   683→   683→        builder.subscribe((status: string) => {
   684→   684→          if (status === 'CHANNEL_ERROR' || status === 'TIMED_OUT') {
   685→   685→            warn(`realtime channel status: ${status}`, null);
   686→   686→          } else {
   687→   687→            // 'SUBSCRIBED' | 'CLOSED' — both are non-fatal
   688→   688→          }
   689→   689→        });
   690→   690→      } catch (e) {
   691→   691→        warn('channel.subscribe skipped', e);
   692→   692→      }
   693→   693→    } catch (e) {
   694→   694→      warn('realtime setup skipped', e);
   695→   695→    }
   696→   696→
   697→   697→    // ── 3. Cleanup on unmount ─────────────────────────────────────────────
   698→   698→    return () => {
   699→   699→      try {
   700→   700→        if (channelRef.current) {
   701→   701→          supabase.removeChannel(channelRef.current);
   702→   702→          channelRef.current = null;
   703→   703→        }
   704→   704→      } catch (e) {
   705→   705→        warn('removeChannel', e);
   706→   706→      }
   707→   707→    };
   708→   708→  }, []);
   709→   709→
   710→   710→  // ALWAYS render children immediately. Supabase loads in the background.
   711→   711→  return <>{children}</>;
   712→   712→}

    86→function RadioOption<T extends string>({
    87→  options,
    88→  labels,
    89→  value,
    90→  onChange,
    91→  colorMap,
    92→}: {
    93→  options: T[];
    94→  labels: Record<T, string>;
    95→  value: T | '';
    96→  onChange: (val: T) => void;
    97→  colorMap?: Partial<Record<T, string>>;
    98→}) {
    99→  return (
   100→    <div className="flex flex-wrap gap-2">
   101→      {options.map((opt) => {
   102→        const isSelected = value === opt;
   103→        const colorCls = colorMap?.[opt];
   104→        return (
   105→          <button
   106→            key={opt}
   107→            type="button"
   108→            onClick={() => onChange(opt)}
   109→            className={`px-3 py-2 rounded-lg border text-sm transition-all whitespace-nowrap
   110→              ${isSelected
   111→                ? colorCls || 'bg-primary text-primary-foreground border-primary shadow-sm'
   112→                : 'bg-white hover:bg-slate-50 border-slate-200 text-slate-700'}
   113→              `}
   114→          >
   115→            {labels[opt]}
   116→          </button>
   117→        );
   118→      })}
   119→    </div>
   120→  );
   121→}

   217→    setSubmitting(true);
   218→
   219→    try {
   220→      const formData: DailyComplaintFormInput = {

   252→      // Add to store
   253→      addDailyComplaint(savedComplaint);

   255→      // Generate clinical alerts in the store

   262→            severity: alert.severity === 'merah' ? 'merah' : 'kuning',
   263→            title: alert.title,
   264→            description: alert.message,
   265→            isRead: false,
   266→            createdAt: new Date().toISOString(),
   267→          });

   269→          addPalliativeMonitoringNotification({
   270→            id: `notif-dc-${Date.now()}-${Math.random().toString(36).substring(2, 7)}`,
   271→            patientId: palliativePatientId,
   272→            patientName,
   273→            type: alert.severity === 'merah' ? 'pain_increase' : 'screening_completed',
   274→            title: alert.title,
   275→            description: alert.message,
   276→            severity: alert.severity === 'merah' ? 'critical' : 'warning',
   277→            isRead: false,
   278→            referenceId: savedComplaint.id,
   279→            createdAt: new Date().toISOString(),

   282→      }
   283→

   287→        description: 'Terima kasih telah mengisi keluhan harian. Tim paliatif akan memantau kondisi Anda.',

   290→      // Notify parent component (e.g., chat panel)
   291→      if (onSubmitSuccess) {

   293→      }

   296→        title: 'Gagal Mengirim',
   297→        description: 'Terjadi kesalahan saat menyimpan keluhan. Silakan coba lagi.',

   299→      });

   302→    }

   315→              Terima kasih telah mengisi keluhan harian. Tim paliatif akan memantau kondisi Anda.
   316→            </p>
   317→          </div>

   319→            variant="outline"

   321→            className="mt-2"

   339→        </CardDescription>
   340→      </CardHeader>
   341→      <CardContent className="space-y-6">
   342→        {/* Q1: Kondisi Hari Ini */}

   350→            labels={conditionLabels}

   352→            onChange={setKondisiHariIni}

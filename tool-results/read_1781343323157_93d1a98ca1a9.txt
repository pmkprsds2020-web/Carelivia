   240→const calculateSocialScore = (a: SocialAssessmentRecord): number => {
   241→  const fields: string[] = [
   242→    a.housingCondition, a.caregiverAvailability, a.familySupportLevel,
   243→    a.transportDifficulty, a.economicConstraint, a.healthcareAccess,
   244→    a.medicalEquipmentNeed, a.socialAssistanceNeed, a.socialIsolationRisk,
   245→  ];
   246→  return fields.reduce((sum, f) => sum + (scoreMap[f] ?? 0), 0);
   247→};
   248→
   249→const calculatePriority = (form: ScreeningFormState): SocialScreeningPriority => {
   250→  const highVals = ['berat', 'tidak_tersedia', 'sangat_sulit', 'tidak_layak', 'tinggi', 'tidak_ada'];
   251→  const medVals = ['sedang', 'terbatas', 'sulit', 'kurang_layak', 'lemah'];
   252→  const vals = Object.values(form).map(v => v.value);
   253→  if (vals.some(v => highVals.includes(v))) return 'tinggi';
   254→  if (vals.some(v => medVals.includes(v))) return 'sedang';
   255→  return 'rendah';
   256→};

   258→const generateRecommendations = (form: ScreeningFormState): string[] => {
   259→  const recs: string[] = [];
   260→  if (form.housing.value === 'tidak_layak' || form.housing.value === 'kurang_layak')
   261→    recs.push('Evaluasi pemenuhan kebutuhan tempat tinggal yang layak, pertimbangkan rujukan ke dinas sosial');
   262→  if (form.caregiver.value === 'tidak_tersedia' || form.caregiver.value === 'terbatas')
   263→    recs.push('Identifikasi sumber caregiver alternatif, pertimbangkan home care profesional');
   264→  if (form.familySupport.value === 'lemah' || form.familySupport.value === 'tidak_ada')
   265→    recs.push('Konseling keluarga dan edukasi tentang dukungan paliatif, pertimbangkan dukungan komunitas');
   266→  if (form.transport.value === 'berat' || form.transport.value === 'sedang')
   267→    recs.push('Fasilitasi transportasi medis atau ambulans untuk kontrol rutin');
   268→  if (form.economic.value === 'berat' || form.economic.value === 'sedang')
   269→    recs.push('Evaluasi kelayakan bantuan BPJS, Jamkesda, PKH, dan program sosial lainnya');
   270→  if (form.healthAccess.value === 'sulit' || form.healthAccess.value === 'sangat_sulit')
   271→    recs.push('Tingkatkan akses telekonsultasi dan home visit, koordinasi dengan puskesmas terdekat');
   272→  if (form.medicalEquip.value === 'berat' || form.medicalEquip.value === 'sedang')
   273→    recs.push('Fasilitasi pengadaan alat kesehatan melalui program bantuan atau sewa');
   274→  if (form.socialAssist.value === 'berat' || form.socialAssist.value === 'sedang')
   275→    recs.push('Rujuk ke pekerja sosial untuk pendampingan akses bantuan sosial');
   276→  if (form.socialIsolation.value === 'tinggi' || form.socialIsolation.value === 'sedang')
   277→    recs.push('Intervensi psikososial, pertimbangkan kelompok dukungan dan kunjungan sukarelawan');
   278→  if (recs.length === 0) recs.push('Pertahankan monitoring rutin dan dukungan sosial yang ada');
   279→  return recs;
   280→};

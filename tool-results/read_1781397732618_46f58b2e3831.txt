   139→   139→// ── Helper: Build comprehensive clinical context string ────────────────────
   140→   140→function buildClinicalContext(
   141→   141→  patient: Record<string, unknown>,
   142→   142→  user: Record<string, unknown> | null,
   143→   143→): string {
   144→   144→  const name = (user?.name as string) || 'Pasien';
   145→   145→  const dob = (user?.dateOfBirth as string) || '-';
   146→   146→  const gender = (user?.gender as string) || '-';

   148→   148→  const rmNumber = (patient.rmNumber as string) || '-';
   149→   149→  const bpjs = (patient.bpjsNumber as string) || (user?.bpjsNumber as string) || '-';
   150→   150→  const address = (patient.address as string) || (user?.address as string) || '-';
   151→   151→
   152→   152→  const vitals = (patient.vitalSigns || []) as Record<string, unknown>[];
   153→   153→  const medications = (patient.medications || []) as Record<string, unknown>[];
   154→   154→  const screenings = (patient.screeningRecords || []) as Record<string, unknown>[];
   155→   155→  const acpDocs = (patient.acpDocuments || []) as Record<string, unknown>[];
   156→   156→

   162→   162→  lines.push('Tanggal Lahir: ' + dob);
   163→   163→  lines.push('Jenis Kelamin: ' + gender);
   164→   164→  lines.push('NIK: ' + nik);
   165→   165→  lines.push('No. Rekam Medis: ' + rmNumber);
   166→   166→  lines.push('No. BPJS: ' + bpjs);
   167→   167→  lines.push('Alamat: ' + address);
   168→   168→  lines.push('');
   169→   169→
   170→   170→  // ── Clinical Info ──
   171→   171→  lines.push('=== INFORMASI KLINIS ===');
   172→   172→  lines.push('Diagnosa Utama: ' + (patient.primaryDiagnosis || '-'));
   173→   173→  lines.push('Diagnosa Penyerta: ' + (patient.secondaryDiagnosis || '-'));
   174→   174→  lines.push('Stadium Penyakit: ' + (patient.diseaseStage || '-'));
   175→   175→  lines.push('Status Perawatan: ' + patient.careStatus);
   176→   176→  lines.push('Status Pasien: ' + patient.patientStatus);
   177→   177→  lines.push('Tingkat Risiko: ' + patient.riskLevel);
   178→   178→  lines.push('Kontak Keluarga: ' + (patient.familyContactName || '-') + ' (' + (patient.familyContactRelation || '-') + ') - ' + (patient.familyContactPhone || '-'));
   179→   179→  lines.push('Catatan: ' + (patient.notes || '-'));
   180→   180→  lines.push('');
   181→   181→
   182→   182→  // ── Vital Signs ──

   188→   188→      const nadi = fmtVal(v.heartRate, ' bpm');
   189→   189→      const rr = fmtVal(v.respiratoryRate, '/menit');
   190→   190→      const suhu = fmtVal(v.temperature, '\u00B0C');
   191→   191→      const spo2 = fmtVal(v.oxygenSat, '%');
   192→   192→      const bb = fmtVal(v.weight, ' kg');
   193→   193→      const tb = fmtVal(v.height, ' cm');
   194→   194→      const bmiVal = fmtVal(v.bmi, '');
   195→   195→      const cat = v.notes ? ' | Catatan: ' + String(v.notes) : '';
   196→   196→      lines.push(
   197→   197→        (i + 1) + '. Tgl: ' + tgl +
   198→   198→        ' | TD: ' + td +
   199→   199→        ' | Nadi: ' + nadi +
   200→   200→        ' | RR: ' + rr +
   201→   201→        ' | Suhu: ' + suhu +
   202→   202→        ' | SpO2: ' + spo2 +
   203→   203→        ' | BB: ' + bb +
   204→   204→        ' | TB: ' + tb +
   205→   205→        ' | BMI: ' + bmiVal +
   206→   206→        cat,
   207→   207→      );
   208→   208→    });
   209→   209→    lines.push('');
   210→   210→  }
   211→   211→
   212→   212→  // ── Medications ──
   213→   213→  if (medications.length > 0) {
   214→   214→    lines.push('=== OBAT AKTIF ===');
   215→   215→    medications.forEach((m) => {
   216→   216→      const adherences = (m.adherences || []) as Record<string, unknown>[];
   217→   217→      const missedDoses = adherences.filter((a) => a.missedDose).length;
   218→   218→      const adherenceRate =
   219→   219→        adherences.length > 0
   220→   220→          ? ((adherences.length - missedDoses) / adherences.length * 100).toFixed(0) + '%'
   221→   221→          : 'Belum ada data';
   222→   222→      const routeStr = m.route ? String(m.route) : 'oral';
   223→   223→      const indicationStr = m.indication ? String(m.indication) : '-';
   224→   224→      const startStr = m.startDate ? String(m.startDate) : '-';
   225→   225→      const endStr = m.endDate ? String(m.endDate) : '-';
   226→   226→      const notesStr = m.notes ? ' | Catatan: ' + String(m.notes) : '';
   227→   227→      lines.push(
   228→   228→        '- ' + m.medicineName + ' ' + m.dosage + ' ' + m.frequency +
   229→   229→        ' (' + routeStr + ')' +
   230→   230→        ' | Indikasi: ' + indicationStr +
   231→   231→        ' | Kepatuhan: ' + adherenceRate +
   232→   232→        ' | Mulai: ' + startStr +
   233→   233→        ' | Selesai: ' + endStr +
   234→   234→        notesStr,
   235→   235→      );
   236→   236→    });
   237→   237→    lines.push('');
   238→   238→  }
   239→   239→
   240→   240→  // ── Screening Records ──
   241→   241→  if (screenings.length > 0) {
   242→   242→    lines.push('=== HASIL SKRINING PALIATIF ===');
   243→   243→    screenings.forEach((s) => {
   244→   244→      const tgl = new Date(s.performedAt as string).toLocaleDateString('id-ID');
   245→   245→      const scoreStr = s.score != null ? String(s.score) : '-';
   246→   246→      const labelStr = s.scoreLabel ? String(s.scoreLabel) : '-';
   247→   247→      const ewsStr = s.ewsLevel ? String(s.ewsLevel) : '-';
   248→   248→      const interpStr = s.interpretation ? String(s.interpretation) : '-';
   249→   249→      const detailStr = s.details ? ' | Detail: ' + String(s.details) : '';
   250→   250→      lines.push(
   251→   251→        '- ' + s.screeningType +
   252→   252→        ': Skor ' + scoreStr + ' (' + labelStr + ')' +
   253→   253→        ' | EWS: ' + ewsStr +
   254→   254→        ' | Tgl: ' + tgl +
   255→   255→        ' | Interpretasi: ' + interpStr +
   256→   256→        detailStr,
   257→   257→      );
   258→   258→    });
   259→   259→    lines.push('');
   260→   260→  }
   261→   261→
   262→   262→  // ── Advance Care Planning ──
   263→   263→  if (acpDocs.length > 0) {
   264→   264→    lines.push('=== ADVANCE CARE PLANNING ===');
   265→   265→    acpDocs.forEach((acp, idx) => {
   266→   266→      lines.push('Dokumen ACP #' + (idx + 1) + ':');
   267→   267→      lines.push('  Pengambil Keputusan: ' + (acp.decisionMakerName || '-') + ' (' + (acp.decisionMakerRelation || '-') + ') - ' + (acp.decisionMakerPhone || '-'));
   268→   268→      lines.push('  Tempat Perawatan Pilihan: ' + (acp.preferredCareLocation || '-'));
   269→   269→      lines.push('  Tujuan Perawatan: ' + (acp.careGoal || '-'));
   270→   270→      lines.push('  Preferensi Resusitasi (CPR/DNR): ' + (acp.resuscitationPref || '-'));
   271→   271→      lines.push('  Preferensi Ventilator: ' + (acp.ventilatorPref || '-'));
   272→   272→      lines.push('  Preferensi ICU: ' + (acp.icuPref || '-'));
   273→   273→      lines.push('  Nutrisi Buatan: ' + (acp.artificialNutrition || '-'));
   274→   274→      lines.push('  Dialisis: ' + (acp.dialysisPref || '-'));
   275→   275→      lines.push('  Donasi Organ: ' + (acp.organDonation || '-'));
   276→   276→      lines.push('  Tanda Tangan: Pasien=' + (acp.patientSigned ? 'Ya' : 'Belum') + ' | Keluarga=' + (acp.familySigned ? 'Ya' : 'Belum') + ' | Dokter=' + (acp.doctorSigned ? 'Ya' : 'Belum'));
   277→   277→      lines.push('  Harapan Pasien: ' + (acp.patientHopes || '-'));
   278→   278→      lines.push('  Kekhawatiran Pasien: ' + (acp.patientWorries || '-'));
   279→   279→      lines.push('  Nilai Hidup Penting: ' + (acp.lifeValues || '-'));
   280→   280→      lines.push('  Preferensi Akhir Hayat: ' + (acp.endOfLifePrefs || '-'));
   281→   281→
   282→   282→      const revisions = (acp.revisions || []) as Record<string, unknown>[];

   291→   291→    lines.push('');

   293→   293→
   294→   294→  return lines.join('\n');
   295→   295→}

   297→   297→// ── Helper: Parse AI-generated resume into structured sections ─────────────
   298→   298→function parseAIResume(aiContent: string): {
   299→   299→  ringkasanKondisi: string;
   300→   300→  ringkasanPemeriksaan: string;
   301→   301→  ringkasanTerapi: string;
   302→   302→  ringkasanACP: string;
   303→   303→  kesimpulanKlinis: string;
   304→   304→  rekomendasiAI: string;
   305→   305→  fullContent: string;
   306→   306→} {
   307→   307→  const sectionHeaders: Record<string, string> = {
   308→   308→    ringkasanKondisi: 'RINGKASAN KONDISI PASIEN',
   309→   309→    ringkasanPemeriksaan: 'RINGKASAN PEMERIKSAAN TERKINI',
   310→   310→    ringkasanTerapi: 'RINGKASAN TERAPI',
   311→   311→    ringkasanACP: 'RINGKASAN ADVANCE CARE PLANNING',
   312→   312→    kesimpulanKlinis: 'KESIMPULAN KLINIS',
   313→   313→    rekomendasiAI: 'REKOMENDASI',
   314→   314→  };
   315→   315→
   316→   316→  const sectionKeys = Object.keys(sectionHeaders);
   317→   317→  const result: Record<string, string> = {};
   318→   318→
   319→   319→  // Initialize all sections with empty strings
   320→   320→  for (const key of sectionKeys) {
   321→   321→    result[key] = '';
   322→   322→  }
   323→   323→
   324→   324→  // Try to split by ## headers (markdown style) first
   325→   325→  const markdownPattern = /^##\s+(.+)$/gm;
   326→   326→  let match: RegExpExecArray | null;
   327→   327→  const sections: { header: string; startIndex: number }[] = [];
   328→   328→
   329→   329→  while ((match = markdownPattern.exec(aiContent)) !== null) {
   330→   330→    sections.push({
   331→   331→      header: match[1].trim().toUpperCase(),
   332→   332→      startIndex: match.index + match[0].length,
   333→   333→    });
   334→   334→  }
   335→   335→
   336→   336→  // If markdown headers found, parse them
   337→   337→  if (sections.length > 0) {
   338→   338→    for (let i = 0; i < sections.length; i++) {
   339→   339→      const currentHeader = sections[i].header;
   340→   340→      const contentStart = sections[i].startIndex;
   341→   341→      const contentEnd = i + 1 < sections.length ? sections[i + 1].startIndex - sections[i + 1].header.length - 3 : aiContent.length;
   342→   342→      const content = aiContent.substring(contentStart, contentEnd).trim();
   343→   343→
   344→   344→      // Match header to section key
   345→   345→      for (const key of sectionKeys) {
   346→   346→        if (currentHeader.includes(sectionHeaders[key])) {
   347→   347→          result[key] = content;
   348→   348→          break;
   349→   349→        }
   350→   350→      }
   351→   351→    }
   352→   352→  } else {
   353→   353→    // Fallback: try to split by === headers
   354→   354→    const equalPattern = /^===\s+(.+)\s+===$/gm;
   355→   355→    const eqSections: { header: string; startIndex: number }[] = [];
   356→   356→
   357→   357→    while ((match = equalPattern.exec(aiContent)) !== null) {
   358→   358→      eqSections.push({
   359→   359→        header: match[1].trim().toUpperCase(),
   360→   360→        startIndex: match.index + match[0].length,
   361→   361→      });
   362→   362→    }
   363→   363→
   364→   364→    if (eqSections.length > 0) {
   365→   365→      for (let i = 0; i < eqSections.length; i++) {
   366→   366→        const currentHeader = eqSections[i].header;
   367→   367→        const contentStart = eqSections[i].startIndex;
   368→   368→        const contentEnd = i + 1 < eqSections.length ? eqSections[i + 1].startIndex - eqSections[i + 1].header.length - 8 : aiContent.length;
   369→   369→        const content = aiContent.substring(contentStart, contentEnd).trim();
   370→   370→
   371→   371→        for (const key of sectionKeys) {
   372→   372→          if (currentHeader.includes(sectionHeaders[key])) {
   373→   373→            result[key] = content;
   374→   374→            break;
   375→   375→          }
   376→   376→        }
   377→   377→      }
   378→   378→    } else {
   379→   379→      // Last resort: put everything in fullContent, try basic keyword matching
   380→   380→      for (const key of sectionKeys) {
   381→   381→        const keyword = sectionHeaders[key];
   382→   382→        const regex = new RegExp(keyword + '[\\s\\S]*?(?=(?:' + Object.values(sectionHeaders).join('|') + '|$))', 'i');
   383→   383→        const sectionMatch = aiContent.match(regex);
   384→   384→        if (sectionMatch) {
   385→   385→          result[key] = sectionMatch[0].replace(new RegExp(keyword, 'i'), '').trim();
   386→   386→        }
   387→   387→      }
   388→   388→    }
   389→   389→  }
   390→   390→
   391→   391→  return {
   392→   392→    ringkasanKondisi: result.ringkasanKondisi || '',
   393→   393→    ringkasanPemeriksaan: result.ringkasanPemeriksaan || '',
   394→   394→    ringkasanTerapi: result.ringkasanTerapi || '',
   395→   395→    ringkasanACP: result.ringkasanACP || '',
   396→   396→    kesimpulanKlinis: result.kesimpulanKlinis || '',
   397→   397→    rekomendasiAI: result.rekomendasiAI || '',
   398→   398→    fullContent: aiContent,
   399→   399→  };
   400→   400→}

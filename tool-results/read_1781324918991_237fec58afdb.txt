    83→function RiskGauge({
    84→  percentage,
    85→  size = 120,
    86→  strokeWidth = 10,
    87→  label,
    88→  riskLevel,
    89→  showLabel = true,
    90→}: {
    91→  percentage: number;
    92→  size?: number;
    93→  strokeWidth?: number;
    94→  label?: string;
    95→  riskLevel: SocialNeedsRiskLevel;
    96→  showLabel?: boolean;
    97→}) {
    98→  const radius = (size - strokeWidth) / 2;
    99→  const circumference = 2 * Math.PI * radius;
   100→  const offset = circumference - (percentage / 100) * circumference;
   101→  const display = getRiskLevelDisplay(riskLevel);
   102→
   103→  const colorMap: Record<SocialNeedsRiskLevel, string> = {
   104→    rendah: '#16A34A',
   105→    sedang: '#CA8A04',
   106→    tinggi: '#EA580C',
   107→    sangat_tinggi: '#DC2626',
   108→  };
   109→
   110→  return (
   111→    <div className="flex flex-col items-center gap-1">
   112→      <div className="relative" style={{ width: size, height: size }}>
   113→        <svg width={size} height={size} className="-rotate-90">
   114→          <circle
   115→            cx={size / 2}
   116→            cy={size / 2}
   117→            r={radius}
   118→            fill="none"
   119→            stroke="#E5E7EB"
   120→            strokeWidth={strokeWidth}
   121→          />
   122→          <circle
   123→            cx={size / 2}
   124→            cy={size / 2}
   125→            r={radius}
   126→            fill="none"
   127→            stroke={colorMap[riskLevel]}
   128→            strokeWidth={strokeWidth}
   129→            strokeDasharray={circumference}
   130→            strokeDashoffset={offset}
   131→            strokeLinecap="round"
   132→            className="transition-all duration-700 ease-out"
   133→          />
   134→        </svg>
   135→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   136→          <span className="text-2xl font-bold" style={{ color: colorMap[riskLevel] }}>
   137→            {percentage}%
   138→          </span>
   139→          <span className="text-xs text-muted-foreground">skor</span>
   140→        </div>
   141→      </div>
   142→      {showLabel && label && (
   143→        <span className="text-xs font-medium text-center mt-1 max-w-[100px]">{label}</span>
   144→      )}
   145→      {showLabel && (
   146→        <Badge
   147→          className="text-[10px] px-1.5 py-0"
   148→          style={{
   149→            backgroundColor: display.bgColor,
   150→            color: display.color,
   151→            borderColor: display.borderColor,
   152→            border: `1px solid ${display.borderColor}`,
   153→          }}
   154→        >
   155→          {display.label}
   156→        </Badge>
   157→      )}
   158→    </div>
   159→  );
   160→}

   333→// MAIN COMPONENT
   334→// ═══════════════════════════════════════════════════════════════════════════
   335→
   336→export function SocialNeedsScreeningPanel() {
   337→  const { currentUser } = useStore();
   338→  const { toast } = useToast();
   339→
   340→  // ── State ────────────────────────────────────────────────────────────────
   341→  const [activeTab, setActiveTab] = useState<string>('screening');
   342→  const [currentCategoryIndex, setCurrentCategoryIndex] = useState(0);
   343→  const [answers, setAnswers] = useState<Record<string, string | string[]>>({});
   344→  const [screeningResult, setScreeningResult] = useState<SocialNeedsScreeningResult | null>(null);
   345→  const [aiResult, setAiResult] = useState<SocialNeedsAIResult | null>(null);
   346→  const [aiLoading, setAiLoading] = useState(false);
   347→  const [isCompleted, setIsCompleted] = useState(false);
   348→
   349→  // ── Derived data ─────────────────────────────────────────────────────────
   350→  const questionsByCategory = useMemo(() => getQuestionsByCategory(), []);
   351→  const currentCategory = CATEGORY_ORDER[currentCategoryIndex];
   352→  const currentQuestions = questionsByCategory[currentCategory] || [];
   353→  const totalQuestions = SOCIAL_NEEDS_QUESTIONS.length;
   354→  const answeredCount = Object.keys(answers).length;
   355→
   356→  // Count answered required questions
   357→  const answeredRequiredCount = useMemo(() => {
   358→    return SOCIAL_NEEDS_QUESTIONS.filter(
   359→      (q) => q.required && answers[q.id] !== undefined && answers[q.id] !== ''
   360→    ).length;
   361→  }, [answers]);
   362→
   363→  const totalRequiredCount = SOCIAL_NEEDS_QUESTIONS.filter((q) => q.required).length;
   364→
   365→  // Category progress
   366→  const categoryProgress = useMemo(() => {
   367→    const progress: Record<string, { answered: number; total: number }> = {};
   368→    for (const cat of CATEGORY_ORDER) {
   369→      const qs = questionsByCategory[cat] || [];
   370→      const answered = qs.filter(
   371→        (q) => answers[q.id] !== undefined && answers[q.id] !== ''
   372→      ).length;
   373→      progress[cat] = { answered, total: qs.length };
   374→    }
   375→    return progress;
   376→  }, [answers, questionsByCategory]);
   377→
   378→  // Live screening result
   379→  const liveResult = useMemo(() => {
   380→    if (answeredCount === 0) return null;
   381→    return calculateScreeningResult(answers);
   382→  }, [answers, answeredCount]);
   383→
   384→  // ── Handlers ─────────────────────────────────────────────────────────────
   385→  const handleSingleChoice = useCallback((questionId: string, value: string) => {
   386→    setAnswers((prev) => ({ ...prev, [questionId]: value }));
   387→  }, []);
   388→
   389→  const handleMultipleChoice = useCallback(
   390→    (questionId: string, value: string, checked: boolean) => {
   391→      setAnswers((prev) => {
   392→        const current = (prev[questionId] as string[]) || [];
   393→        if (checked) {
   394→          return { ...prev, [questionId]: [...current, value] };
   395→        } else {
   396→          return { ...prev, [questionId]: current.filter((v) => v !== value) };
   397→        }
   398→      });
   399→    },
   400→    []
   401→  );
   402→
   403→  const handleTextArea = useCallback((questionId: string, value: string) => {
   404→    setAnswers((prev) => ({ ...prev, [questionId]: value }));
   405→  }, []);
   406→
   407→  const handlePrevious = useCallback(() => {
   408→    setCurrentCategoryIndex((prev) => Math.max(0, prev - 1));
   409→  }, []);
   410→
   411→  const handleNext = useCallback(() => {
   412→    setCurrentCategoryIndex((prev) =>
   413→      Math.min(CATEGORY_ORDER.length - 1, prev + 1)
   414→    );
   415→  }, []);
   416→
   417→  const handleSaveDraft = useCallback(() => {
   418→    toast({
   419→      title: 'Draft Tersimpan',
   420→      description: `${answeredCount} dari ${totalQuestions} pertanyaan telah disimpan.`,
   421→    });
   422→  }, [answeredCount, totalQuestions, toast]);
   423→
   424→  const handleCompleteScreening = useCallback(() => {
   425→    // Check if all required questions are answered
   426→    const unansweredRequired = SOCIAL_NEEDS_QUESTIONS.filter(
   427→      (q) => q.required && (!answers[q.id] || answers[q.id] === '')
   428→    );
   429→
   430→    if (unansweredRequired.length > 0) {
   431→      toast({
   432→        title: 'Pertanyaan Wajib Belum Lengkap',
   433→        description: `Masih ada ${unansweredRequired.length} pertanyaan wajib yang belum dijawab.`,
   434→        variant: 'destructive',
   435→      });
   436→      return;
   437→    }
   438→
   439→    const result = calculateScreeningResult(answers);
   440→    setScreeningResult(result);
   441→    setIsCompleted(true);
   442→    setActiveTab('results');
   443→    toast({
   444→      title: 'Skrining Selesai',
   445→      description: `Skor keseluruhan: ${result.overallPercentage}% (${getRiskLevelDisplay(result.overallRiskLevel).label})`,
   446→    });
   447→  }, [answers, toast]);
   448→
   449→  const handleResetForm = useCallback(() => {
   450→    setAnswers({});
   451→    setScreeningResult(null);
   452→    setAiResult(null);
   453→    setIsCompleted(false);
   454→    setCurrentCategoryIndex(0);
   455→    setActiveTab('screening');
   456→    toast({ title: 'Form Direset', description: 'Semua jawaban telah dihapus.' });
   457→  }, [toast]);
   458→
   459→  const handleRunAI = useCallback(async () => {
   460→    const result = screeningResult || liveResult;
   461→    if (!result) return;
   462→
   463→    setAiLoading(true);
   464→    setAiResult(null);
   465→
   466→    try {
   467→      const response = await fetch('/api/social-needs-screening-ai', {
   468→        method: 'POST',
   469→        headers: { 'Content-Type': 'application/json' },
   470→        body: JSON.stringify({
   471→          screeningResult: result,
   472→          answers,
   473→          patientData: {
   474→            name: currentUser?.name,
   475→            role: currentUser?.role,
   476→          },
   477→        }),
   478→      });
   479→
   480→      if (response.ok) {
   481→        const data = await response.json();
   482→        // Map API response to SocialNeedsAIResult format
   483→        const mapped: SocialNeedsAIResult = {
   484→          familySupportScore: data.familySupportScore || result.categoryScores.find(c => c.category === 'dukungan_keluarga')?.riskLevel || 'rendah',
   485→          socialRiskScore: data.socialRiskScore || result.overallRiskLevel,
   486→          caregiverBurnoutScore: data.caregiverBurnoutScore || result.categoryScores.find(c => c.category === 'caregiver')?.riskLevel || 'rendah',
   487→          accessToCareScore: data.accessToCareScore || result.categoryScores.find(c => c.category === 'akses_layanan')?.riskLevel || 'rendah',
   488→          financialRiskScore: data.financialRiskScore || result.categoryScores.find(c => c.category === 'ekonomi')?.riskLevel || 'rendah',
   489→          socialIsolationScore: data.socialIsolationScore || result.categoryScores.find(c => c.category === 'interaksi_sosial')?.riskLevel || 'rendah',
   490→          recommendations: (data.recommendations || []).map((a: { priority: number; action: string; reason: string; category: string }, i: number) => ({
   491→            priority: a.priority || i + 1,
   492→            action: a.action,
   493→            reason: a.reason || '',
   494→            category: a.category || 'monitoring_intensif',
   495→          })),
   496→          analysisSummary: data.analysisSummary || data.socialConditionSummary || 'Analisis AI kebutuhan sosial selesai.',
   497→          earlyWarnings: (data.earlyWarnings || []).map((w: { type: string; severity: string; title: string; description: string }) => ({
   498→            type: w.type || '',
   499→            severity: w.severity || 'warning',
   500→            title: w.title || '',
   501→            description: w.description || '',
   502→          })),
   503→          generatedAt: data.generatedAt || new Date().toISOString(),
   504→        };
   505→        setAiResult(mapped);
   506→      } else {
   507→        throw new Error('API failed');
   508→      }
   509→    } catch {
   510→      // Fallback to local AI analysis
   511→      const localResult = generateLocalAIAnalysis(result, answers);
   512→      setAiResult(localResult);
   513→      toast({
   514→        title: 'Menggunakan Analisis Lokal',
   515→        description: 'API AI tidak tersedia, menggunakan analisis lokal sebagai fallback.',
   516→      });
   517→    } finally {
   518→      setAiLoading(false);
   519→    }
   520→  }, [screeningResult, liveResult, answers, currentUser, toast]);
   521→
   522→  // ── Render: Question ─────────────────────────────────────────────────────
   523→
   524→  const renderQuestion = (question: SocialNeedsQuestion) => {
   525→    const currentAnswer = answers[question.id];
   526→
   527→    return (
   528→      <div key={question.id} className="space-y-1.5 sm:space-y-3 pb-3 sm:pb-4">
   529→        <div className="flex items-start gap-1.5 sm:gap-2">
   530→          <span className="text-xs sm:text-sm font-semibold text-[#2D8C7A] shrink-0">
   531→            Q{question.questionNumber}.
   532→          </span>
   533→          <Label className="text-xs sm:text-sm font-medium leading-snug">
   534→            {question.questionText}
   535→            {question.required && <span className="text-red-500 ml-1">*</span>}
   536→          </Label>
   537→        </div>
   538→
   539→        {/* Single Choice (Radio) */}
   540→        {question.type === 'single_choice' && question.options && (
   541→          <div className="space-y-1.5 sm:space-y-2 pl-1 sm:pl-2">
   542→            {question.options.map((option) => {
   543→              const isSelected = currentAnswer === option.value;
   544→              const hasTooltip = question.hasTooltip && option.tooltip;
   545→
   546→              return (
   547→                <div
   548→                  key={option.value}
   549→                  className={cn(
   550→                    'flex items-center gap-2 sm:gap-3 p-2 sm:p-2.5 rounded-lg border cursor-pointer transition-all duration-200',
   551→                    isSelected
   552→                      ? 'border-[#2D8C7A] bg-[#2D8C7A]/5 shadow-sm'
   553→                      : 'border-gray-200 hover:border-gray-300 hover:bg-gray-50'
   554→                  )}
   555→                  onClick={() => handleSingleChoice(question.id, option.value)}
   556→                >
   557→                  <div
   558→                    className={cn(
   559→                      'w-4 h-4 rounded-full border-2 flex items-center justify-center shrink-0 transition-all',
   560→                      isSelected
   561→                        ? 'border-[#2D8C7A] bg-[#2D8C7A]'
   562→                        : 'border-gray-300'

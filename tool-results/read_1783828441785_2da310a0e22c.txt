     4→     4→     4→// ───────────────────────────────────────────────────────────────────────────
     5→     5→     5→//
     6→     6→     6→// STORAGE STRATEGY
     7→     7→     7→// ─────────────────────────
     8→     8→     8→// DDL access is unavailable, so we reuse the existing `patient_documents`
     9→     9→     9→// table for all four supporting examination types. The table already has a
    10→    10→    10→// `jenis` CHECK constraint that allows 'lab', 'radiologi', 'gambar', 'pdf',
    11→    11→    11→// 'lainnya' — perfect for our use case.
    12→    12→    12→//
    13→    13→    13→//   patient_documents(
    14→    14→    14→//     id           uuid PK,
    15→    15→    15→//     patient_id   uuid NOT NULL FK,
    16→    16→    16→//     jenis        text NOT NULL CHECK in (lab, radiologi, gambar, pdf, lainnya),
    17→    17→    17→//     nama_file    text NOT NULL,
    18→    18→    18→//     storage_path text NOT NULL,
    19→    19→    19→//     url          text,
    20→    20→    20→//     keterangan   text,        -- we store a JSON string here for structured data
    21→    21→    21→//     tanggal      date NOT NULL DEFAULT current_date,
    22→    22→    22→//     uploaded_by  text,
    23→    23→    23→//     created_at   timestamptz
    24→    24→    24→//   )
    25→    25→    25→//
    26→    26→    26→// Mapping:
    27→    27→    27→//   • Laboratorium  → jenis='lab',       keterangan=JSON{type, gdp, gds, hba1c, ...}
    28→    28→    28→//   • USG           → jenis='gambar',    keterangan=JSON{type='usg', jenis_usg, hasil, catatan, doctor_id}
    29→    29→    29→//                     + foto uploaded to Storage (foto_url stored in `url`)
    30→    30→    30→//   • EKG           → jenis='gambar',    keterangan=JSON{type='ekg', interpretasi, catatan, doctor_id}
    31→    31→    31→//                     + foto uploaded to Storage (foto_url stored in `url`)
    32→    32→    32→//   • Radiologi     → jenis='radiologi', keterangan=JSON{type='radiology', jenis_radiologi, hasil, catatan, doctor_id}
    33→    33→    33→//                     + foto uploaded to Storage (foto_url stored in `url`)
    34→    34→    34→//
    35→    35→    35→// Photos are uploaded to the `patient-files` Storage bucket (same as

   286→   286→   286→async function uploadPhoto(
   287→   287→   287→  patientId: string,
   288→   288→   288→  jenis: string,
   289→   289→   289→  file: File | Blob
   290→   290→   290→): Promise<{ url?: string; storagePath: string; fileName: string } | null> {
   291→   291→   291→  const fileName = (file as File).name ?? `upload-${Date.now()}`;
   292→   292→   292→  const storagePath = buildStoragePath(patientId, jenis, fileName);
   293→   293→   293→  const uploadRes = await safeQuery(
   294→   294→   294→    supabase.storage.from(BUCKET).upload(storagePath, file, {
   295→   295→   295→      cacheControl: '3600',
   296→   296→   296→      upsert: false,
   297→   297→   297→    }),
   298→   298→   298→    null as any,
   299→   299→   299→    'supportingExamService.uploadPhoto'
   300→   300→   300→  );
   301→   301→   301→  if (uploadRes === null) return null;
   302→   302→   302→  let url: string | undefined;
   303→   303→   303→  try {
   304→   304→   304→    const { data: pub } = supabase.storage.from(BUCKET).getPublicUrl(storagePath);
   305→   305→   305→    if (pub?.publicUrl) url = pub.publicUrl;
   306→   306→   306→  } catch {
   307→   307→   307→    /* leave url undefined */
   308→   308→   308→  }
   309→   309→   309→  return { url, storagePath, fileName };
   310→   310→   310→}

   825→   825→   825→    return all;

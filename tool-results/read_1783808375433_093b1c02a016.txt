     1→// ───────────────────────────────────────────────────────────────────────────
     2→// SupabaseSyncProvider — keeps the Zustand store in sync with Supabase.
     3→//
     4→// Responsibilities:
     5→//   1. On mount, load all Monitoring-Paliatif data from Supabase via the 13
     6→//      service files and populate the Zustand store. Local demo data wins on
     7→//      conflict — we only replace the store if Supabase returns non-empty
     8→//      arrays.
     9→//   2. Subscribe to Supabase Realtime for 12 tables so any DB change is
    10→//      mirrored into the store immediately (INSERT/UPDATE/DELETE).
    11→//   3. ALWAYS render children immediately. Supabase loading is fire-and-forget
    12→//      and never blocks the UI. If Supabase is unreachable (missing tables,
    13→//      RLS denied, network down), we log a warning and continue with local
    14→//      Zustand data.
    15→//   4. NEVER throw — every subscription and load is wrapped in try/catch.
    16→// ───────────────────────────────────────────────────────────────────────────
    17→'use client';

    43→/** Check whether a row id already exists in a store array. */
    44→function hasId<T extends { id: string }>(list: T[], id: string): boolean {
    45→  return list.some((x) => x.id === id);
    46→}
    47→
    48→/** Log a non-fatal sync warning. */
    49→function warn(label: string, err: unknown) {
    50→  console.warn(`[SupabaseSync] ${label}:`, err);
    51→}

    53→// ─── Initial-load handlers ─────────────────────────────────────────────────
    54→//
    55→// Each handler is self-contained: it loads data from Supabase, validates it,
    56→// and dispatches the corresponding Zustand action(s). Wrapped in try/catch by
    57→// the caller. We only REPLACE the store if Supabase returns non-empty data —
    58→// otherwise we keep the local demo data (per the task's "local-wins" rule).
    59→
    60→async function loadPatients(store: ReturnType<typeof useStore.getState>) {
    61→  const patients = await svc.patientService.getAll();
    62→  if (Array.isArray(patients) && patients.length > 0) {
    63→    store.setPalliativePatients(patients as PalliativePatientInfo[]);
    64→  }
    65→  return patients;
    66→}

    68→async function loadPatientScopedData(
    69→  store: ReturnType<typeof useStore.getState>,
    70→  patients: PalliativePatientInfo[]
    71→) {
    72→  for (const patient of patients) {
    73→    const pid = patient.id;
    74→
    75→    // Vitals
    76→    try {
    77→      const rows = await svc.vitalService.getAll(pid);
    78→      if (Array.isArray(rows) && rows.length > 0) {
    79→        // Replace existing vitals for this patient to avoid stale data,
    80→        // but keep vitals from other patients intact.
    81→        const others = store.vitalSignRecords.filter((v) => v.palliativePatientId !== pid);
    82→        store.setVitalSignRecords([
    83→          ...others,
    84→          ...(rows as VitalSignRecordInfo[]),
    85→        ]);
    86→      }
    87→    } catch (e) { warn('vitalService.getAll', e); }
    88→
    89→    // Screenings
    90→    try {
    91→      const rows = await svc.screeningService.getAll(pid);
    92→      if (Array.isArray(rows) && rows.length > 0) {
    93→        const others = store.palliativeScreeningRecords.filter((s) => s.palliativePatientId !== pid);
    94→        store.setPalliativeScreeningRecords([
    95→          ...others,
    96→          ...(rows as PalliativeScreeningRecordInfo[]),
    97→        ]);
    98→      }
    99→    } catch (e) { warn('screeningService.getAll', e); }
   100→
   101→    // Medications
   102→    try {
   103→      const rows = await svc.medicationService.getAll(pid);
   104→      if (Array.isArray(rows) && rows.length > 0) {
   105→        const others = store.palliativeMedications.filter((m) => m.palliativePatientId !== pid);
   106→        store.setPalliativeMedications([
   107→          ...others,
   108→          ...(rows as PalliativeMedicationInfo[]),
   109→        ]);
   110→      }
   111→    } catch (e) { warn('medicationService.getAll', e); }
   112→
   113→    // Nutrition
   114→    try {
   115→      const rows = await svc.nutritionService.getAll(pid);
   116→      if (Array.isArray(rows) && rows.length > 0) {
   117→        const others = store.nutritionRecords.filter((n) => n.palliativePatientId !== pid);
   118→        store.setNutritionRecords([
   119→          ...others,
   120→          ...(rows as NutritionRecordInfo[]),
   121→        ]);
   122→      }
   123→    } catch (e) { warn('nutritionService.getAll', e); }
   124→
   125→    // Daily complaints
   126→    try {
   127→      const rows = await svc.complaintService.getAll(pid);
   128→      if (Array.isArray(rows) && rows.length > 0) {
   129→        const others = store.dailyComplaints.filter((c) => c.palliativePatientId !== pid);
   130→        store.setDailyComplaints([
   131→          ...others,
   132→          ...(rows as DailyComplaintRecord[]),
   133→        ]);
   134→      }
   135→    } catch (e) { warn('complaintService.getAll', e); }
   136→
   137→    // Social assessments
   138→    try {
   139→      const rows = await svc.socialService.getAll(pid);
   140→      if (Array.isArray(rows) && rows.length > 0) {
   141→        const others = store.socialAssessments.filter((s) => s.palliativePatientId !== pid);
   142→        const fresh = store.socialAssessments.filter((s) => s.palliativePatientId === pid);
   143→        // Replace this patient's records
   144→        const next = [...others];
   145→        for (const r of rows as SocialAssessmentRecord[]) {
   146→          if (!hasId(fresh, r.id)) next.push(r);
   147→          else next.push(r); // overwrite in-place
   148→        }
   149→        // Use direct set since addSocialAssessment prepends (we want replace)
   150→        store.socialAssessments = next;
   151→        // Trigger Zustand update via the official setter (addSocialAssessment
   152→        // prepends; we want a clean replace per-patient, so set directly):
   153→        useStore.setState({ socialAssessments: next });
   154→      }
   155→    } catch (e) { warn('socialService.getAll', e); }
   156→
   157→    // ACP
   158→    try {
   159→      const rows = await svc.acpService.getAll(pid);
   160→      if (Array.isArray(rows) && rows.length > 0) {
   161→        const others = store.advanceCarePlans.filter((a) => a.palliativePatientId !== pid);
   162→        store.setAdvanceCarePlans([
   163→          ...others,
   164→          ...(rows as AdvanceCarePlanInfo[]),
   165→        ]);
   166→      }
   167→    } catch (e) { warn('acpService.getAll', e); }
   168→
   169→    // Clinical alerts — loaded via direct Supabase query (the sibling agent's
   170→    // notificationService is for user notifications, not clinical alerts).
   171→    try {
   172→      const rows = await safeQuery(
   173→        supabase
   174→          .from('clinical_alerts')
   175→          .select('*')
   176→          .eq('patient_id', pid)
   177→          .order('created_at', { ascending: false }),
   178→        [] as any[],
   179→        'loadPatientScopedData.clinical_alerts'
   180→      );
   181→      if (Array.isArray(rows) && rows.length > 0) {
   182→        for (const r of rows as any[]) {
   183→          const alert: PalliativeClinicalAlert = {
   184→            id: r.id,
   185→            patientId: r.patient_id ?? pid,
   186→            alertType: r.alert_type ?? 'form_tidak_diisi',
   187→            severity: r.severity ?? 'kuning',
   188→            title: r.title ?? '',
   189→            description: r.description ?? '',
   190→            values: r.values ?? undefined,
   191→            isRead: !!r.is_read,
   192→            createdAt: r.created_at ?? new Date().toISOString(),
   193→          };
   194→          if (!hasId(store.palliativeClinicalAlerts, alert.id)) {
   195→            store.addPalliativeClinicalAlert(alert);
   196→          }
   197→        }
   198→      }
   199→    } catch (e) { warn('clinical_alerts load', e); }
   200→
   201→    // Audit log — same pattern, direct Supabase query.
   202→    try {
   203→      const rows = await safeQuery(
   204→        supabase
   205→          .from('audit_log')
   206→          .select('*')
   207→          .eq('patient_id', pid)
   208→          .order('created_at', { ascending: false }),
   209→        [] as any[],
   210→        'loadPatientScopedData.audit_log'
   211→      );
   212→      if (Array.isArray(rows) && rows.length > 0) {
   213→        for (const r of rows as any[]) {
   214→          const entry: PalliativeAuditEntry = {
   215→            id: r.id,
   216→            patientId: r.patient_id ?? pid,
   217→            action: r.action ?? 'clinical_action',
   218→            performedBy: r.performed_by ?? 'system',
   219→            performedByRole: r.performed_by_role ?? 'system',
   220→            details: r.details ?? undefined,
   221→            ipAddress: r.ip_address ?? undefined,
   222→            device: r.device ?? undefined,
   223→            createdAt: r.created_at ?? new Date().toISOString(),
   224→          };
   225→          if (!hasId(store.palliativeAuditLog, entry.id)) {
   226→            store.addPalliativeAuditEntry(entry);
   227→          }
   228→        }
   229→      }
   230→    } catch (e) { warn('audit_log load', e); }
   231→
   232→    // Chat messages — the sibling agent's chatService.getMessages takes a
   233→    // `roomId` (not a patientId). To list messages per patient we'd need to
   234→    // resolve the room first; the chat panel handles this lazily itself, so
   235→    // we skip preloading here to avoid a costly N+1 lookup.
   236→  }
   237→}

   239→// ─── Realtime handlers ─────────────────────────────────────────────────────
   240→//
   241→// Each handler receives a PostgresChangesPayload and dispatches the matching
   242→// Zustand action. We dedupe by id on INSERT, replace on UPDATE, filter on
   243→// DELETE. Wrapped in try/catch by the caller so a single bad event can never
   244→// break the subscription.
   245→
   246→type RealtimePayload = {
   247→  eventType: 'INSERT' | 'UPDATE' | 'DELETE' | '*';
   248→  new: Record<string, any> | null;
   249→  old: Record<string, any> | null;
   250→  errors: string[] | null;
   251→};
   252→
   253→function handlePatientEvent(store: ReturnType<typeof useStore.getState>, p: RealtimePayload) {
   254→  const row = p.new;
   255→  const id = (row?.id ?? p.old?.id) as string | undefined;
   256→  if (!id) return;
   257→  if (p.eventType === 'DELETE') {
   258→    if (hasId(store.palliativePatients, id)) store.removePalliativePatient(id);
   259→    return;
   260→  }
   261→  // INSERT or UPDATE — try to load the full record via the service so the
   262→  // row goes through the proper snake_case → camelCase mapping.
   263→  svc.patientService
   264→    .getById(id)
   265→    .then((fresh: PalliativePatientInfo | null) => {
   266→      if (!fresh) return;
   267→      const exists = hasId(useStore.getState().palliativePatients, id);
   268→      if (exists) {
   269→        store.updatePalliativePatient(id, fresh);
   270→      } else {
   271→        store.addPalliativePatient(fresh);
   272→      }
   273→    })
   274→    .catch((e: unknown) => warn('patient realtime getById', e));
   275→}

   277→function handleVitalEvent(store: ReturnType<typeof useStore.getState>, p: RealtimePayload) {
   278→  const row = p.new;
   279→  const id = (row?.id ?? p.old?.id) as string | undefined;
   280→  if (!id) return;
   281→  if (p.eventType === 'DELETE') {
   282→    useStore.setState((s) => ({
   283→      vitalSignRecords: s.vitalSignRecords.filter((v) => v.id !== id),
   284→    }));
   285→    return;
   286→  }
   287→  // Reload via service for proper mapping
   288→  svc.vitalService
   289→    .getAll(row?.patient_id as string)
   290→    .then((rows: VitalSignRecordInfo[]) => {
   291→      const fresh = rows.find((r) => r.id === id);
   292→      if (!fresh) return;
   293→      const state = useStore.getState();
   294→      const exists = hasId(state.vitalSignRecords, id);
   295→      if (exists) {
   296→        useStore.setState({
   297→          vitalSignRecords: state.vitalSignRecords.map((v) => (v.id === id ? fresh : v)),
   298→        });
   299→      } else {
   300→        store.addVitalSignRecord(fresh);
   301→      }
   302→    })
   303→    .catch((e: unknown) => warn('vital realtime', e));
   304→}

   481→function handleMessageEvent(store: ReturnType<typeof useStore.getState>, p: RealtimePayload) {
   482→  const row = p.new;
   483→  const id = (row?.id ?? p.old?.id) as string | undefined;
   484→  if (!id) return;
   485→  if (p.eventType === 'DELETE') {
   486→    useStore.setState((s) => ({
   487→      palliativeChatMessages: s.palliativeChatMessages.filter((m) => m.id !== id),
   488→    }));
   489→    return;
   490→  }
   491→  // Map the row directly — chatService doesn't expose a getById method, and
   492→  // the row payload already contains everything we need.
   493→  try {
   494→    const formData: any = row?.form_data ?? undefined;
   495→    const fresh: PalliativeChatMessage = {
   496→      id,
   497→      roomId: row?.room_id ?? '',
   498→      senderId: row?.sender_id ?? '',
   499→      senderName: row?.sender_name ?? '',
   500→      senderRole: row?.sender_role ?? 'system',
   501→      type: row?.type ?? 'text',
   502→      content: row?.content ?? '',
   503→      status: row?.status ?? 'sent',
   504→      formType: row?.form_type ?? undefined,
   505→      formData,
   506→      formResponse: row?.form_response ?? undefined,
   507→      screeningType: row?.screening_type ?? undefined,
   508→      aiSummary: row?.ai_summary ?? undefined,
   509→      clinicalAlert: formData?.clinicalAlert ?? undefined,
   510→      imageUrl: row?.image_url ?? undefined,
   511→      createdAt: row?.created_at ?? new Date().toISOString(),
   512→      readAt: row?.read_at ?? undefined,
   513→    };
   514→    const state = useStore.getState();
   515→    const exists = hasId(state.palliativeChatMessages, id);
   516→    if (exists) {
   517→      useStore.setState({
   518→        palliativeChatMessages: state.palliativeChatMessages.map((m) =>
   519→          m.id === id ? fresh : m
   520→        ),
   521→      });
   522→    } else {
   523→      store.addPalliativeChatMessage(fresh);
   524→    }
   525→  } catch (e) { warn('message mapping', e); }
   526→}

   528→function handleClinicalAlertEvent(store: ReturnType<typeof useStore.getState>, p: RealtimePayload) {
   529→  const row = p.new;
   530→  const id = (row?.id ?? p.old?.id) as string | undefined;

   535→    }));

   537→  }
   538→  // Build a PalliativeClinicalAlert from the row directly (alerts aren't
   539→  // fetched via a dedicated service method in the loader — we map here).
   540→  try {
   541→    const fresh: PalliativeClinicalAlert = {
   542→      id,
   543→      patientId: row?.patient_id ?? '',
   544→      alertType: row?.alert_type ?? 'form_tidak_diisi',
   545→      severity: row?.severity ?? 'kuning',
   546→      title: row?.title ?? '',
   547→      description: row?.description ?? '',
   548→      values: row?.values ?? undefined,
   549→      isRead: !!row?.is_read,
   550→      createdAt: row?.created_at ?? new Date().toISOString(),
   551→    };
   552→    const state = useStore.getState();
   553→    const exists = hasId(state.palliativeClinicalAlerts, id);
   554→    if (exists) {
   555→      useStore.setState({
   556→        palliativeClinicalAlerts: state.palliativeClinicalAlerts.map((a) =>
   557→          a.id === id ? { ...a, ...fresh } : a
   558→        ),
   559→      });
   560→    } else {
   561→      store.addPalliativeClinicalAlert(fresh);
   562→    }
   563→  } catch (e) { warn('alert mapping', e); }
   564→}

   566→function handleAuditLogEvent(store: ReturnType<typeof useStore.getState>, p: RealtimePayload) {
   567→  const row = p.new;
   568→  const id = (row?.id ?? p.old?.id) as string | undefined;
   569→  if (!id) return;
   570→  if (p.eventType === 'DELETE') {
   571→    useStore.setState((s) => ({
   572→      palliativeAuditLog: s.palliativeAuditLog.filter((a) => a.id !== id),
   573→    }));
   574→    return;
   575→  }
   576→  try {
   577→    const fresh: PalliativeAuditEntry = {
   578→      id,
   579→      patientId: row?.patient_id ?? '',
   580→      action: row?.action ?? 'clinical_action',
   581→      performedBy: row?.performed_by ?? 'system',
   582→      performedByRole: row?.performed_by_role ?? 'system',
   583→      details: row?.details ?? undefined,
   584→      ipAddress: row?.ip_address ?? undefined,
   585→      device: row?.device ?? undefined,
   586→      createdAt: row?.created_at ?? new Date().toISOString(),
   587→    };
   588→    const state = useStore.getState();
   589→    const exists = hasId(state.palliativeAuditLog, id);
   590→    if (exists) {
   591→      useStore.setState({
   592→        palliativeAuditLog: state.palliativeAuditLog.map((a) =>
   593→          a.id === id ? { ...a, ...fresh } : a
   594→        ),
   595→      });
   596→    } else {
   597→      store.addPalliativeAuditEntry(fresh);
   598→    }
   599→  } catch (e) { warn('audit mapping', e); }
   600→}

   602→function handleNotificationEvent(_store: ReturnType<typeof useStore.getState>, _p: RealtimePayload) {
   603→  // Notifications table maps to the global Notification[] (clinicalAlerts in
   604→  // the store) — but the global notifications are loaded via /api/notifications,
   605→  // not Supabase. We deliberately skip to avoid double-loading. If you want
   606→  // Supabase notifications to populate the store, dispatch here.

   609→// ─── Provider ──────────────────────────────────────────────────────────────
   610→
   611→export function SupabaseSyncProvider({ children }: SupabaseSyncProviderProps) {
   612→  const store = useStore();
   613→  const loadedRef = useRef(false);
   614→  const channelRef = useRef<ReturnType<typeof supabase.channel> | null>(null);
   615→
   616→  useEffect(() => {
   617→    if (loadedRef.current) return;
   618→    loadedRef.current = true;
   619→
   620→    // Snapshot the store once — actions on this snapshot remain stable for
   621→    // the lifetime of the provider, even if the component re-renders.
   622→    const s = useStore.getState;
   623→
   624→    // ── 1. Initial load (fire-and-forget, never blocks UI) ────────────────
   625→    (async () => {
   626→      try {
   627→        const patients = await loadPatients(s());
   628→        if (Array.isArray(patients) && patients.length > 0) {
   629→          await loadPatientScopedData(s(), patients as PalliativePatientInfo[]);
   630→        }
   631→        console.log('[SupabaseSync] initial load complete');
   632→      } catch (e) {
   633→        warn('initial load skipped', e);
   634→      }
   635→    })();
   636→
   637→    // ── 2. Realtime subscriptions (best-effort, silently skipped on error) ─
   638→    try {
   639→      const channel = supabase.channel('carelivia-realtime');
   640→      channelRef.current = channel;
   641→
   642→      const tableHandlers: Array<[string, (store: ReturnType<typeof s>, p: RealtimePayload) => void]> = [
   643→        ['patients', handlePatientEvent],
   644→        ['vital_signs', handleVitalEvent],
   645→        ['screenings', handleScreeningEvent],
   646→        ['medications', handleMedicationEvent],
   647→        ['nutrition', handleNutritionEvent],
   648→        ['daily_complaints', handleComplaintEvent],
   649→        ['social_assessments', handleSocialEvent],
   650→        ['acp', handleAcpEvent],
   651→        ['chat_rooms', handleChatRoomEvent],
   652→        ['messages', handleMessageEvent],
   653→        ['clinical_alerts', handleClinicalAlertEvent],
   654→        ['audit_log', handleAuditLogEvent],
   655→        ['notifications', handleNotificationEvent],
   656→      ];
   657→
   658→      let builder = channel;
   659→      for (const [table, handler] of tableHandlers) {
   660→        try {
   661→          builder = builder.on(
   662→            'postgres_changes',
   663→            { event: '*', schema: 'public', table },
   664→            (payload: any) => {
   665→              try {
   666→                handler(s(), {
   667→                  eventType: payload.eventType,
   668→                  new: payload.new,
   669→                  old: payload.old,
   670→                  errors: payload.errors,
   671→                });
   672→              } catch (e) {
   673→                warn(`realtime handler for ${table}`, e);
   674→              }
   675→            }
   676→          );
   677→        } catch (e) {
   678→          warn(`subscribe to ${table} skipped`, e);
   679→        }
   680→      }
   681→
   682→      try {
   683→        builder.subscribe((status: string) => {
   684→          if (status === 'CHANNEL_ERROR' || status === 'TIMED_OUT') {
   685→            warn(`realtime channel status: ${status}`, null);
   686→          } else {
   687→            // 'SUBSCRIBED' | 'CLOSED' — both are non-fatal
   688→          }
   689→        });
   690→      } catch (e) {
   691→        warn('channel.subscribe skipped', e);
   692→      }
   693→    } catch (e) {
   694→      warn('realtime setup skipped', e);
   695→    }
   696→
   697→    // ── 3. Cleanup on unmount ─────────────────────────────────────────────
   698→    return () => {
   699→      try {
   700→        if (channelRef.current) {
   701→          supabase.removeChannel(channelRef.current);
   702→          channelRef.current = null;
   703→        }
   704→      } catch (e) {
   705→        warn('removeChannel', e);
   706→      }
   707→    };
   708→  }, []);
   709→
   710→  // ALWAYS render children immediately. Supabase loads in the background.
   711→  return <>{children}</>;
   712→}

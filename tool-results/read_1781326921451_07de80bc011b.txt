    85→function RiskGauge({
    86→  percentage,
    87→  size = 120,
    88→  strokeWidth = 10,
    89→  label,
    90→  riskLevel,
    91→  showLabel = true,
    92→}: {
    93→  percentage: number;
    94→  size?: number;
    95→  strokeWidth?: number;
    96→  label?: string;
    97→  riskLevel: SocialNeedsRiskLevel;
    98→  showLabel?: boolean;
    99→}) {
   100→  const radius = (size - strokeWidth) / 2;
   101→  const circumference = 2 * Math.PI * radius;
   102→  const offset = circumference - (percentage / 100) * circumference;
   103→  const display = getRiskLevelDisplay(riskLevel);
   104→
   105→  const colorMap: Record<SocialNeedsRiskLevel, string> = {
   106→    rendah: '#16A34A',
   107→    sedang: '#CA8A04',
   108→    tinggi: '#EA580C',
   109→    sangat_tinggi: '#DC2626',
   110→  };
   111→
   112→  return (
   113→    <div className="flex flex-col items-center gap-1">
   114→      <div className="relative" style={{ width: size, height: size }}>
   115→        <svg width={size} height={size} className="-rotate-90">
   116→          <circle
   117→            cx={size / 2}
   118→            cy={size / 2}
   119→            r={radius}
   120→            fill="none"
   121→            stroke="#E5E7EB"
   122→            strokeWidth={strokeWidth}
   123→          />
   124→          <circle
   125→            cx={size / 2}
   126→            cy={size / 2}
   127→            r={radius}
   128→            fill="none"
   129→            stroke={colorMap[riskLevel]}
   130→            strokeWidth={strokeWidth}
   131→            strokeDasharray={circumference}
   132→            strokeDashoffset={offset}
   133→            strokeLinecap="round"
   134→            className="transition-all duration-700 ease-out"
   135→          />
   136→        </svg>
   137→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   138→          <span className="text-2xl font-bold" style={{ color: colorMap[riskLevel] }}>
   139→            {percentage}%
   140→          </span>
   141→          <span className="text-xs text-muted-foreground">skor</span>
   142→        </div>
   143→      </div>
   144→      {showLabel && label && (
   145→        <span className="text-xs font-medium text-center mt-1 max-w-[100px]">{label}</span>
   146→      )}
   147→      {showLabel && (
   148→        <Badge
   149→          className="text-[10px] px-1.5 py-0"
   150→          style={{
   151→            backgroundColor: display.bgColor,
   152→            color: display.color,
   153→            borderColor: display.borderColor,
   154→            border: `1px solid ${display.borderColor}`,
   155→          }}
   156→        >
   157→          {display.label}
   158→        </Badge>
   159→      )}
   160→    </div>
   161→  );
   162→}

   410→export function SocialNeedsScreeningPanel({ embedded = false }: SocialNeedsScreeningPanelProps) {
   411→  const { currentUser } = useStore();
   412→  const { toast } = useToast();
   413→
   414→  // ── State ────────────────────────────────────────────────────────────────
   415→  const [activeTab, setActiveTab] = useState<string>('screening');
   416→  const [currentCategoryIndex, setCurrentCategoryIndex] = useState(0);
   417→  const [answers, setAnswers] = useState<Record<string, string | string[]>>({});
   418→  const [screeningResult, setScreeningResult] = useState<SocialNeedsScreeningResult | null>(null);
   419→  const [aiResult, setAiResult] = useState<SocialNeedsAIResult | null>(null);
   420→  const [aiLoading, setAiLoading] = useState(false);
   421→  const [isCompleted, setIsCompleted] = useState(false);
   422→  const [showSummary, setShowSummary] = useState(false);
   423→  const [showHistory, setShowHistory] = useState(false);
   424→
   425→  // ── Derived data ─────────────────────────────────────────────────────────
   426→  const questionsByCategory = useMemo(() => getQuestionsByCategory(), []);
   427→  const currentCategory = CATEGORY_ORDER[currentCategoryIndex];
   428→  const currentQuestions = questionsByCategory[currentCategory] || [];
   429→  const totalQuestions = SOCIAL_NEEDS_QUESTIONS.length;
   430→  const answeredCount = Object.keys(answers).length;
   431→
   432→  // Count answered required questions
   433→  const answeredRequiredCount = useMemo(() => {
   434→    return SOCIAL_NEEDS_QUESTIONS.filter(
   435→      (q) => q.required && answers[q.id] !== undefined && answers[q.id] !== ''
   436→    ).length;
   437→  }, [answers]);
   438→
   439→  const totalRequiredCount = SOCIAL_NEEDS_QUESTIONS.filter((q) => q.required).length;
   440→
   441→  // Category progress
   442→  const categoryProgress = useMemo(() => {
   443→    const progress: Record<string, { answered: number; total: number }> = {};
   444→    for (const cat of CATEGORY_ORDER) {
   445→      const qs = questionsByCategory[cat] || [];
   446→      const answered = qs.filter(
   447→        (q) => answers[q.id] !== undefined && answers[q.id] !== ''
   448→      ).length;
   449→      progress[cat] = { answered, total: qs.length };
   450→    }
   451→    return progress;
   452→  }, [answers, questionsByCategory]);
   453→
   454→  // Live screening result
   455→  const liveResult = useMemo(() => {
   456→    if (answeredCount === 0) return null;
   457→    return calculateScreeningResult(answers);
   458→  }, [answers, answeredCount]);
   459→
   460→  // ── Handlers ─────────────────────────────────────────────────────────────
   461→  const handleSingleChoice = useCallback((questionId: string, value: string) => {
   462→    setAnswers((prev) => ({ ...prev, [questionId]: value }));
   463→  }, []);
   464→
   465→  const handleMultipleChoice = useCallback(
   466→    (questionId: string, value: string, checked: boolean) => {
   467→      setAnswers((prev) => {
   468→        const current = (prev[questionId] as string[]) || [];
   469→        if (checked) {
   470→          return { ...prev, [questionId]: [...current, value] };
   471→        } else {
   472→          return { ...prev, [questionId]: current.filter((v) => v !== value) };
   473→        }
   474→      });
   475→    },
   476→    []
   477→  );
   478→
   479→  const handleTextArea = useCallback((questionId: string, value: string) => {
   480→    setAnswers((prev) => ({ ...prev, [questionId]: value }));
   481→  }, []);
   482→
   483→  const handlePrevious = useCallback(() => {
   484→    setCurrentCategoryIndex((prev) => Math.max(0, prev - 1));
   485→  }, []);
   486→
   487→  const handleNext = useCallback(() => {
   488→    setCurrentCategoryIndex((prev) =>
   489→      Math.min(CATEGORY_ORDER.length - 1, prev + 1)
   490→    );
   491→  }, []);
   492→
   493→  const handleSaveDraft = useCallback(() => {
   494→    toast({
   495→      title: 'Draft Tersimpan',
   496→      description: `${answeredCount} dari ${totalQuestions} pertanyaan telah disimpan.`,
   497→    });
   498→  }, [answeredCount, totalQuestions, toast]);
   499→
   500→  const handleCompleteScreening = useCallback(() => {
   501→    // Check if all required questions are answered
   502→    const unansweredRequired = SOCIAL_NEEDS_QUESTIONS.filter(
   503→      (q) => q.required && (!answers[q.id] || answers[q.id] === '')
   504→    );
   505→
   506→    if (unansweredRequired.length > 0) {
   507→      toast({
   508→        title: 'Pertanyaan Wajib Belum Lengkap',
   509→        description: `Masih ada ${unansweredRequired.length} pertanyaan wajib yang belum dijawab.`,
   510→        variant: 'destructive',
   511→      });
   512→      return;
   513→    }
   514→
   515→    const result = calculateScreeningResult(answers);
   516→    setScreeningResult(result);
   517→    setIsCompleted(true);
   518→    setShowSummary(true);
   519→    setActiveTab('results');
   520→    toast({
   521→      title: 'Skrining Selesai',
   522→      description: `Skor keseluruhan: ${result.overallPercentage}% (${getRiskLevelDisplay(result.overallRiskLevel).label})`,
   523→    });
   524→  }, [answers, toast]);
   525→
   526→  const handleResetForm = useCallback(() => {
   527→    setAnswers({});
   528→    setScreeningResult(null);
   529→    setAiResult(null);
   530→    setIsCompleted(false);
   531→    setShowSummary(false);
   532→    setCurrentCategoryIndex(0);
   533→    setActiveTab('screening');
   534→    toast({ title: 'Form Direset', description: 'Semua jawaban telah dihapus.' });
   535→  }, [toast]);
   536→
   537→  const handleRunAI = useCallback(async () => {
   538→    const result = screeningResult || liveResult;
   539→    if (!result) return;
   540→
   541→    setAiLoading(true);
   542→    setAiResult(null);
   543→
   544→    try {
   545→      const response = await fetch('/api/social-needs-screening-ai', {
   546→        method: 'POST',
   547→        headers: { 'Content-Type': 'application/json' },
   548→        body: JSON.stringify({
   549→          screeningResult: result,
   550→          answers,
   551→          patientData: {
   552→            name: currentUser?.name,
   553→            role: currentUser?.role,
   554→          },
   555→        }),
   556→      });
   557→
   558→      if (response.ok) {
   559→        const data = await response.json();
   560→        // Map API response to SocialNeedsAIResult format
   561→        const mapped: SocialNeedsAIResult = {
   562→          familySupportScore: data.familySupportScore || result.categoryScores.find(c => c.category === 'dukungan_keluarga')?.riskLevel || 'rendah',
   563→          socialRiskScore: data.socialRiskScore || result.overallRiskLevel,

     1→// ───────────────────────────────────────────────────────────────────────────
     2→// supportingExamService — CRUD for Pemeriksaan Penunjang
     3→// (Laboratorium, USG, EKG, Radiologi)
     4→// ───────────────────────────────────────────────────────────────────────────
     5→//
     6→// STORAGE STRATEGY
     7→// ─────────────────────────
     8→// DDL access is unavailable, so we reuse the existing `patient_documents`
     9→// table for all four supporting examination types. The table already has a
    10→// `jenis` CHECK constraint that allows 'lab', 'radiologi', 'gambar', 'pdf',
    11→// 'lainnya' — perfect for our use case.
    12→//
    13→//   patient_documents(
    14→//     id           uuid PK,
    15→//     patient_id   uuid NOT NULL FK,
    16→//     jenis        text NOT NULL CHECK in (lab, radiologi, gambar, pdf, lainnya),
    17→//     nama_file    text NOT NULL,
    18→//     storage_path text NOT NULL,
    19→//     url          text,
    20→//     keterangan   text,        -- we store a JSON string here for structured data
    21→//     tanggal      date NOT NULL DEFAULT current_date,
    22→//     uploaded_by  text,
    23→//     created_at   timestamptz
    24→//   )
    25→//
    26→// Mapping:
    27→//   • Laboratorium  → jenis='lab',       keterangan=JSON{type, gdp, gds, hba1c, ...}
    28→//   • USG           → jenis='gambar',    keterangan=JSON{type='usg', jenis_usg, hasil, catatan, doctor_id}
    29→//                     + foto uploaded to Storage (foto_url stored in `url`)
    30→//   • EKG           → jenis='gambar',    keterangan=JSON{type='ekg', interpretasi, catatan, doctor_id}
    31→//                     + foto uploaded to Storage (foto_url stored in `url`)
    32→//   • Radiologi     → jenis='radiologi', keterangan=JSON{type='radiology', jenis_radiologi, hasil, catatan, doctor_id}
    33→//                     + foto uploaded to Storage (foto_url stored in `url`)
    34→//
    35→// Photos are uploaded to the `patient-files` Storage bucket (same as
    36→// documentService) under path: {patientId}/{jenis}/{timestamp}-{filename}
    37→//
    38→// The DDL for future dedicated tables (laboratory_results, ultrasound_results,
    39→// ecg_results, radiology_results, supporting_examinations) is documented in
    40→// supabase/schema.sql for when DDL access becomes available.

   190→function todayStr(): string {
   191→  return new Date().toISOString().slice(0, 10);

   194→function buildStoragePath(patientId: string, jenis: string, fileName: string): string {
   195→  const sanitized = (fileName || 'file').replace(/[^a-zA-Z0-9._-]/g, '_');
   196→  return `${patientId}/${jenis}/${Date.now()}-${sanitized}`;
   197→}

   199→function safeParseKeterangan(raw: any): Record<string, any> {
   200→  if (raw == null) return {};
   201→  if (typeof raw === 'object') return raw as Record<string, any>;
   202→  if (typeof raw === 'string') {
   203→    try {
   204→      return JSON.parse(raw) as Record<string, any>;
   205→    } catch {
   206→      return {};
   207→    }
   208→  }
   209→  return {};
   210→}

   284→// ── Storage upload helper ───────────────────────────────────────────────────
   285→
   286→async function uploadPhoto(
   287→  patientId: string,
   288→  jenis: string,
   289→  file: File | Blob
   290→): Promise<{ url?: string; storagePath: string; fileName: string } | null> {
   291→  const fileName = (file as File).name ?? `upload-${Date.now()}`;
   292→  const storagePath = buildStoragePath(patientId, jenis, fileName);
   293→  const uploadRes = await safeQuery(
   294→    supabase.storage.from(BUCKET).upload(storagePath, file, {
   295→      cacheControl: '3600',
   296→      upsert: false,
   297→    }),
   298→    null as any,
   299→    'supportingExamService.uploadPhoto'
   300→  );
   301→  if (uploadRes === null) return null;
   302→  let url: string | undefined;
   303→  try {
   304→    const { data: pub } = supabase.storage.from(BUCKET).getPublicUrl(storagePath);
   305→    if (pub?.publicUrl) url = pub.publicUrl;
   306→  } catch {
   307→    /* leave url undefined */
   308→  }
   309→  return { url, storagePath, fileName };
   310→}
   311→
   312→async function removeStorage(storagePath: string): Promise<void> {
   313→  if (!storagePath) return;
   314→  await safeQuery(
   315→    supabase.storage.from(BUCKET).remove([storagePath]),
   316→    null as any,
   317→    'supportingExamService.removeStorage'
   318→  );
   319→}

   345→      return null;

   348→      type: 'laboratorium',
   349→      gdp: input.gdp,

   352→      ureum: input.ureum,

   359→      catatan: input.catatan,
   360→      doctorId: validUuidOrUndefined(input.doctorId),
   361→      createdBy: input.createdBy,
   362→    });
   363→    const payload = {
   364→      patient_id: input.patientId,
   365→      jenis: 'lab' as const,
   366→      nama_file: `lab-${input.tanggal ?? todayStr()}`,

   370→      uploaded_by: input.createdBy,

   372→    const { data: row, error } = await safeInsert<any>(

   375→    );

   548→      supabase.from('patient_documents').delete().eq('id', id),
   549→      null as any,
   550→      'supportingExamService.deleteUsg'

   806→  async listAll(patientId: string): Promise<SupportingExamUnion[]> {
   807→    if (!isValidUuid(patientId)) return [];
   808→    const [lab, usg, ecg, rad] = await Promise.all([
   809→      this.listLab(patientId),
   810→      this.listUsg(patientId),
   811→      this.listEcg(patientId),
   812→      this.listRadiology(patientId),
   813→    ]);
   814→    const all: SupportingExamUnion[] = [
   815→      ...lab.map((data) => ({ type: 'laboratorium' as const, data })),
   816→      ...usg.map((data) => ({ type: 'usg' as const, data })),
   817→      ...ecg.map((data) => ({ type: 'ekg' as const, data })),
   818→      ...rad.map((data) => ({ type: 'radiologi' as const, data })),
   819→    ];
   820→    all.sort((a, b) => {
   821→      const ta = new Date((a.data as any).tanggal ?? (a.data as any).createdAt).getTime();
   822→      const tb = new Date((b.data as any).tanggal ?? (b.data as any).createdAt).getTime();
   823→      return tb - ta;
   824→    });
   825→    return all;
   826→  },

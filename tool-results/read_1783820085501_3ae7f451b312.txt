    87→function RadioOption<T extends string>({
    88→  options,
    89→  labels,
    90→  value,
    91→  onChange,
    92→  colorMap,
    93→}: {
    94→  options: T[];
    95→  labels: Record<T, string>;
    96→  value: T | '';
    97→  onChange: (val: T) => void;
    98→  colorMap?: Partial<Record<T, string>>;
    99→}) {
   100→  return (
   101→    <div className="flex flex-wrap gap-2">
   102→      {options.map((opt) => {
   103→        const isSelected = value === opt;
   104→        const colorCls = colorMap?.[opt];
   105→        return (
   106→          <button
   107→            key={opt}
   108→            type="button"
   109→            onClick={() => onChange(opt)}
   110→            className={`px-3 py-2 rounded-lg border text-sm transition-all whitespace-nowrap
   111→              ${isSelected
   112→                ? colorCls || 'bg-primary text-primary-foreground border-primary shadow-sm'
   113→                : 'bg-white hover:bg-slate-50 border-slate-200 text-slate-700'}
   114→              `}
   115→          >
   116→            {labels[opt]}
   117→          </button>
   118→        );
   119→      })}
   120→    </div>
   121→  );
   122→}

   126→function ConditionalTextarea({

   165→  const addPalliativeClinicalAlert = useStore((s) => s.addPalliativeClinicalAlert);
   166→  const addPalliativeMonitoringNotification = useStore((s) => s.addPalliativeMonitoringNotification);
   167→  const palliativePatients = useStore((s) => s.palliativePatients);
   168→
   169→  const [submitting, setSubmitting] = useState(false);
   170→  const [submitted, setSubmitted] = useState(false);
   171→
   172→  // Form state
   173→  const [kondisiHariIni, setKondisiHariIni] = useState<DailyCondition | ''>('');

   180→  const [alasanMakanMinum, setAlasanMakanMinum] = useState('');
   181→  const [tidur, setTidur] = useState<DailyYesNo | ''>('');
   182→  const [alasanTidur, setAlasanTidur] = useState('');
   183→  const [masalahObat, setMasalahObat] = useState<DailyMedicineProblem | ''>('');
   184→  const [deskripsiMasalahObat, setDeskripsiMasalahObat] = useState('');
   185→
   186→  const patient = palliativePatients.find((p) => p.id === palliativePatientId);
   187→  const patientName = patient?.patientName || 'Pasien';
   188→
   189→  const resetForm = () => {
   190→    setSubmitted(false);
   191→    setKondisiHariIni('');
   192→    setAlasanKondisi('');
   193→    setKeluhanBaru('');
   194→    setDeskripsiKeluhanBaru('');
   195→    setKondisiNyeri('');
   196→    setKondisiSesak('');
   197→    setMakanMinum('');
   198→    setAlasanMakanMinum('');
   199→    setTidur('');
   200→    setAlasanTidur('');
   201→    setMasalahObat('');
   202→    setDeskripsiMasalahObat('');
   203→  };
   204→
   205→  const canSubmit = useMemo(() => {
   206→    if (!kondisiHariIni || !keluhanBaru || !kondisiNyeri || !kondisiSesak || !makanMinum || !tidur || !masalahObat) return false;
   207→    if (kondisiHariIni === 'tidak_baik' && !alasanKondisi.trim()) return false;
   208→    if (keluhanBaru === 'ada' && !deskripsiKeluhanBaru.trim()) return false;
   209→    if (makanMinum === 'tidak' && !alasanMakanMinum.trim()) return false;
   210→    if (tidur === 'tidak' && !alasanTidur.trim()) return false;
   211→    if (masalahObat === 'ya' && !deskripsiMasalahObat.trim()) return false;
   212→    return true;
   213→  }, [kondisiHariIni, keluhanBaru, kondisiNyeri, kondisiSesak, makanMinum, tidur, masalahObat,
   214→      alasanKondisi, deskripsiKeluhanBaru, alasanMakanMinum, alasanTidur, deskripsiMasalahObat]);
   215→
   216→  const handleSubmit = async () => {
   217→    if (!canSubmit) return;
   218→    // ── UUID validation ──
   219→    if (!isValidUuid(palliativePatientId)) {
   220→      console.error('[daily-complaint] ABORTED — patient_id is not a valid UUID:', palliativePatientId);
   221→      toast({
   222→        title: 'Patient UUID tidak ditemukan',
   223→        description: 'Pilih pasien yang valid sebelum mengisi keluhan harian.',
   224→        variant: 'destructive',
   225→      });
   226→      return;
   227→    }
   228→    setSubmitting(true);
   229→
   230→    try {
   231→      const formData: DailyComplaintFormInput = {
   232→        palliativePatientId,
   233→        kondisiHariIni: kondisiHariIni as DailyCondition,
   234→        alasanKondisi: kondisiHariIni === 'tidak_baik' ? alasanKondisi : undefined,
   235→        keluhanBaru: keluhanBaru as DailyComplaintYesNo,
   236→        deskripsiKeluhanBaru: keluhanBaru === 'ada' ? deskripsiKeluhanBaru : undefined,
   237→        kondisiNyeri: kondisiNyeri as DailyPainCondition,
   238→        kondisiSesak: kondisiSesak as DailyDyspneaCondition,
   239→        makanMinum: makanMinum as DailyYesNo,
   240→        alasanMakanMinum: makanMinum === 'tidak' ? alasanMakanMinum : undefined,
   241→        tidur: tidur as DailyYesNo,
   242→        alasanTidur: tidur === 'tidak' ? alasanTidur : undefined,
   243→        masalahObat: masalahObat as DailyMedicineProblem,
   244→        deskripsiMasalahObat: masalahObat === 'ya' ? deskripsiMasalahObat : undefined,
   245→        sumberPengisian: source,
   246→      };
   247→
   248→      const res = await fetch('/api/daily-complaints', {
   249→        method: 'POST',
   250→        headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify(formData),
   252→      });
   253→
   254→      if (!res.ok) throw new Error('Gagal menyimpan keluhan harian');
   255→
   256→      const data = await res.json();
   257→      const savedComplaint: DailyComplaintRecord = {
   258→        ...data.complaint,
   259→        submittedAt: data.complaint.submittedAt,
   260→        createdAt: data.complaint.createdAt,
   261→      };
   262→
   263→      // Add to store
   264→      addDailyComplaint(savedComplaint);
   265→
   266→      // Generate clinical alerts in the store
   267→      if (data.alerts && data.alerts.length > 0) {
   268→        for (const alert of data.alerts) {
   269→          addPalliativeClinicalAlert({
   270→            id: `alert-dc-${Date.now()}-${Math.random().toString(36).substring(2, 7)}`,
   271→            patientId: palliativePatientId,
   272→            alertType: alert.severity === 'merah' ? 'perburukan' : 'gejala_berat',
   273→            severity: alert.severity === 'merah' ? 'merah' : 'kuning',
   274→            title: alert.title,
   275→            description: alert.message,
   276→            isRead: false,
   277→            createdAt: new Date().toISOString(),
   278→          });
   279→
   280→          addPalliativeMonitoringNotification({
   281→            id: `notif-dc-${Date.now()}-${Math.random().toString(36).substring(2, 7)}`,
   282→            patientId: palliativePatientId,
   283→            patientName,
   284→            type: alert.severity === 'merah' ? 'pain_increase' : 'screening_completed',
   285→            title: alert.title,
   286→            description: alert.message,
   287→            severity: alert.severity === 'merah' ? 'critical' : 'warning',
   288→            isRead: false,
   289→            referenceId: savedComplaint.id,
   290→            createdAt: new Date().toISOString(),
   291→          });
   292→        }
   293→      }
   294→
   295→      setSubmitted(true);
   296→      toast({
   297→        title: 'Keluhan Harian Terkirim',
   298→        description: 'Terima kasih telah mengisi keluhan harian. Tim paliatif akan memantau kondisi Anda.',
   299→      });
   300→
   301→      // Notify parent component (e.g., chat panel)
   302→      if (onSubmitSuccess) {
   303→        onSubmitSuccess(savedComplaint);
   304→      }
   305→    } catch (error) {
   306→      toast({
   307→        title: 'Gagal Mengirim',
   308→        description: 'Terjadi kesalahan saat menyimpan keluhan. Silakan coba lagi.',
   309→        variant: 'destructive',
   310→      });
   311→    } finally {
   312→      setSubmitting(false);
   313→    }
   314→  };
   315→
   316→  if (submitted) {
   317→    return (
   318→      <Card className="border-emerald-200 bg-emerald-50/50">
   319→        <CardContent className="py-12 flex flex-col items-center gap-4 text-center">
   320→          <div className="h-16 w-16 rounded-full bg-emerald-100 flex items-center justify-center">
   321→            <CheckCircle2 className="h-8 w-8 text-emerald-600" />
   322→          </div>
   323→          <div>
   324→            <h3 className="text-lg font-semibold text-emerald-900">Keluhan Harian Berhasil Dikirim</h3>
   325→            <p className="text-sm text-emerald-700 mt-1">
   326→              Terima kasih telah mengisi keluhan harian. Tim paliatif akan memantau kondisi Anda.
   327→            </p>
   328→          </div>
   329→          <Button
   330→            variant="outline"
   331→            onClick={resetForm}
   332→            className="mt-2"
   333→          >
   334→            Isi Keluhan Baru
   335→          </Button>
   336→        </CardContent>
   337→      </Card>
   338→    );
   339→  }
   340→
   341→  return (
   342→    <Card>
   343→      <CardHeader className="pb-4">
   344→        <div className="flex items-center gap-2">
   345→          <ClipboardList className="h-5 w-5 text-primary" />
   346→          <CardTitle className="text-lg">Form Keluhan Harian</CardTitle>
   347→        </div>
   348→        <CardDescription>
   349→          Isilah form berikut untuk melaporkan kondisi Anda hari ini. Data Anda akan dipantau oleh tim paliatif.
   350→        </CardDescription>
   351→      </CardHeader>
   352→      <CardContent className="space-y-6">
   353→        {/* Q1: Kondisi Hari Ini */}
   354→        <div className="space-y-2">
   355→          <Label className="text-sm font-semibold flex items-center gap-1.5">
   356→            <Smile className="h-4 w-4 text-teal-600" />
   357→            1. Bagaimana kondisi Anda hari ini?
   358→          </Label>
   359→          <RadioOption
   360→            options={['baik', 'tidak_baik'] as DailyCondition[]}
   361→            labels={conditionLabels}
   362→            value={kondisiHariIni}
   363→            onChange={setKondisiHariIni}
   364→            colorMap={{
   365→              baik: 'bg-emerald-600 text-white border-emerald-600 shadow-sm',
   366→              tidak_baik: 'bg-red-600 text-white border-red-600 shadow-sm',
   367→            }}
   368→          />
   369→          <ConditionalTextarea
   370→            show={kondisiHariIni === 'tidak_baik'}
   371→            label="Alasan kondisi tidak baik hari ini"
   372→            value={alasanKondisi}
   373→            onChange={setAlasanKondisi}
   374→            placeholder="Jelaskan alasan kondisi Anda tidak baik..."
   375→          />
   376→        </div>
   377→
   378→        <Separator />
   379→
   380→        {/* Q2: Keluhan Baru */}
   381→        <div className="space-y-2">
   382→          <Label className="text-sm font-semibold flex items-center gap-1.5">
   383→            <AlertCircle className="h-4 w-4 text-teal-600" />
   384→            2. Apakah ada keluhan baru hari ini?
   385→          </Label>
   386→          <RadioOption
   387→            options={['tidak_ada', 'ada'] as DailyComplaintYesNo[]}
   388→            labels={complaintYesNoLabels}
   389→            value={keluhanBaru}
   390→            onChange={setKeluhanBaru}
   391→            colorMap={{
   392→              tidak_ada: 'bg-emerald-600 text-white border-emerald-600 shadow-sm',

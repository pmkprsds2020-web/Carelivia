   146→function checkTTVAlerts(ttv: TTVFormAnswers): { alert: boolean; severity: 'hijau' | 'kuning' | 'merah'; messages: string[] } {
   147→  const messages: string[] = [];
   148→  let maxSeverity: 'hijau' | 'kuning' | 'merah' = 'hijau';
   149→
   150→  if (ttv.systolicBP !== undefined && ttv.systolicBP < 90) {
   151→    messages.push(`TD rendah: ${ttv.systolicBP} mmHg`);
   152→    maxSeverity = 'merah';
   153→  }
   154→  if (ttv.systolicBP !== undefined && ttv.systolicBP > 180) {
   155→    messages.push(`TD tinggi: ${ttv.systolicBP} mmHg`);
   156→    maxSeverity = 'merah';
   157→  }
   158→  if (ttv.heartRate !== undefined && ttv.heartRate > 120) {
   159→    messages.push(`Nadi tinggi: ${ttv.heartRate} bpm`);
   160→    maxSeverity = 'merah';
   161→  }
   162→  if (ttv.respiratoryRate !== undefined && ttv.respiratoryRate > 30) {
   163→    messages.push(`RR tinggi: ${ttv.respiratoryRate}/menit`);
   164→    maxSeverity = 'merah';
   165→  } else if (ttv.respiratoryRate !== undefined && ttv.respiratoryRate > 24) {
   166→    messages.push(`RR meningkat: ${ttv.respiratoryRate}/menit`);
   167→    if (maxSeverity !== 'merah') maxSeverity = 'kuning';
   168→  }
   169→  if (ttv.oxygenSat !== undefined && ttv.oxygenSat < 90) {
   170→    messages.push(`SpO2 rendah: ${ttv.oxygenSat}%`);
   171→    maxSeverity = 'merah';
   172→  }
   173→  if (ttv.temperature !== undefined && ttv.temperature > 38) {
   174→    messages.push(`Demam: ${ttv.temperature}°C`);
   175→    if (maxSeverity !== 'merah') maxSeverity = 'kuning';
   176→  }
   177→  if (ttv.painScore !== undefined && ttv.painScore >= 7) {
   178→    messages.push(`Nyeri berat: ${ttv.painScore}/10`);
   179→    if (maxSeverity !== 'merah') maxSeverity = 'kuning';
   180→  }
   181→
   182→  return { alert: messages.length > 0, severity: maxSeverity, messages };
   183→}

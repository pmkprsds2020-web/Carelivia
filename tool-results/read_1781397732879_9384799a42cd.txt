    84→    84→function formatDate(dateStr: string): string {
    85→    85→  try {
    86→    86→    return new Date(dateStr).toLocaleDateString('id-ID', {
    87→    87→      day: 'numeric',
    88→    88→      month: 'long',
    89→    89→      year: 'numeric',
    90→    90→    });
    91→    91→  } catch {
    92→    92→    return dateStr;
    93→    93→  }
    94→    94→}
    95→    95→
    96→    96→function formatDateTime(dateStr: string): string {
    97→    97→  try {
    98→    98→    return new Date(dateStr).toLocaleDateString('id-ID', {
    99→    99→      day: 'numeric',
   100→   100→      month: 'short',
   101→   101→      year: 'numeric',
   102→   102→      hour: '2-digit',
   103→   103→      minute: '2-digit',
   104→   104→    });
   105→   105→  } catch {
   106→   106→    return dateStr;
   107→   107→  }
   108→   108→}

   110→   110→function genId(prefix: string): string {
   111→   111→  return `${prefix}-${Date.now()}-${Math.random().toString(36).substring(2, 7)}`;
   112→   112→}
   113→   113→
   114→   114→function genDocNumber(prefix: string): string {
   115→   115→  const now = new Date();
   116→   116→  const y = now.getFullYear();
   117→   117→  const m = String(now.getMonth() + 1).padStart(2, '0');
   118→   118→  const seq = String(Math.floor(Math.random() * 9000) + 1000);
   119→   119→  return `${prefix}/${y}/${m}/${seq}`;
   120→   120→}

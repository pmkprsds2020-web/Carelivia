   252→        // No email confirmation required → auto-login
   253→        if (res.user) {
   254→          applyAuthSuccess(res.user, `Selamat datang, ${res.user.fullName}.`);
   255→        } else {
   256→          setAuthInfo('Registrasi berhasil. Silakan masuk.');

   264→      setAuthError('Terjadi kesalahan. Silakan coba lagi.');

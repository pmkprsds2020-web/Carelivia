   192→
   193→function formatCurrency(amount: number): string {
   194→  return `Rp ${new Intl.NumberFormat('id-ID').format(amount)}`;
   195→}
   196→
   197→function formatDate(dateStr: string): string {
   198→  return new Date(dateStr).toLocaleDateString('id-ID', {
   199→    day: 'numeric',
   200→    month: 'short',
   201→    year: 'numeric',
   202→    hour: '2-digit',
   203→    minute: '2-digit',
   204→  });
   205→}
   206→
   207→function generateInvoiceNumber(): string {
   208→  return `INV-${Date.now()}`;
   209→}
   210→
   211→// ---------------------------------------------------------------------------
   212→// Build merged payments from demo + store
   213→// ---------------------------------------------------------------------------
   214→
   215→function buildMergedPayments(storePayments: Payment[]): MergedPayment[] {
   216→  const payMap = new Map<string, MergedPayment>();
   217→
   218→  // Add demo payments
   219→  for (const dp of demoPayments) {
   220→    payMap.set(dp.id, dp);
   221→  }
   222→
   223→  // Add store payments
   224→  for (const sp of storePayments) {
   225→    const typeLabel = sp.type === 'prescription' ? 'E-Resep' : sp.type === 'consultation' ? 'Konsultasi' : sp.type === 'pharmacy' ? 'Farmasi' : sp.type === 'homecare' ? 'Home Care' : 'Konsultasi';
   226→    payMap.set(sp.id, {
   227→      id: sp.id,
   228→      invoiceNumber: sp.invoiceNumber || `INV-${sp.id}`,
   229→      type: typeLabel as PaymentType,
   230→      amount: sp.amount,
   231→      method: sp.method,
   232→      status: sp.status,
   233→      date: sp.paidAt || sp.createdAt,
   234→      description: sp.type === 'prescription' ? `Pembayaran E-Resep - INV-${sp.id.slice(-6)}` : `Pembayaran ${typeLabel}`,
   235→      referenceId: sp.referenceId,
   236→      paidAt: sp.paidAt,
   237→    });
   238→  }
   239→
   240→  return Array.from(payMap.values()).sort(
   241→    (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime(),
   242→  );
   243→}

   137→function extractPatientKey(name: string): string {
   138→  if (!name) return '';
   139→  const first = name.trim().split(' ')[0].toLowerCase();
   140→  for (const [key, val] of Object.entries(PATIENT_NAME_MAP)) {
   141→    if (val.toLowerCase().startsWith(first)) return key;
   142→  }
   143→  return '';
   144→}

   146→function generateRmNumber(mr: MedicalRecord, index: number): string {
   147→  const dateStr = mr.recordDate || mr.createdAt;
   148→  const d = new Date(dateStr);
   149→  const y = d.getFullYear();
   150→  const m = String(d.getMonth() + 1).padStart(2, '0');
   151→  const day = String(d.getDate()).padStart(2, '0');
   152→  const seq = String(index + 1).padStart(4, '0');
   153→  return `RM-${y}${m}${day}-${seq}`;
   154→}

   182→function formatDate(dateStr: string): string {
   183→  return new Date(dateStr).toLocaleDateString('id-ID', {
   184→    day: 'numeric',
   185→    month: 'long',
   186→    year: 'numeric',
   187→  });
   188→}
   189→
   190→function formatTime(dateStr: string): string {
   191→  return new Date(dateStr).toLocaleTimeString('id-ID', {
   192→    hour: '2-digit',
   193→    minute: '2-digit',
   194→  });
   195→}
   196→
   197→function formatCurrency(amount: number): string {
   198→  return new Intl.NumberFormat('id-ID', {
   199→    style: 'currency',
   200→    currency: 'IDR',
   201→    minimumFractionDigits: 0,
   202→    maximumFractionDigits: 0,
   203→  }).format(amount);
   204→}

   677→interface MergedMedicalRecord extends MedicalRecord {
   678→  patientName: string;
   679→  consultationType?: string;
   680→  linkedPrescription?: Prescription;
   681→}
   682→
   683→function buildMergedRecords(
   684→  storeRecords: MedicalRecord[],
   685→  storeConsultations: Consultation[],
   686→  storePrescriptions: Prescription[],
   687→): MergedMedicalRecord[] {
   688→  // Step 1: Merge store records with demo records using Map dedup
   689→  const recordMap = new Map<string, MedicalRecord>();
   690→  for (const mr of DOCTOR_DEMO_MEDICAL_RECORDS) {
   691→    recordMap.set(mr.id, mr);
   692→  }
   693→  for (const mr of storeRecords) {
   694→    recordMap.set(mr.id, mr);
   695→  }
   696→
   697→  // Step 2: Merge consultations using Map dedup
   698→  const consultationMap = new Map<string, Consultation>();
   699→  for (const c of DOCTOR_DEMO_CONSULTATIONS) {
   700→    consultationMap.set(c.id, c);
   701→  }
   702→  for (const c of storeConsultations) {
   703→    consultationMap.set(c.id, c);
   704→  }
   705→
   706→  // Step 3: Merge prescriptions using Map dedup
   707→  const prescriptionMap = new Map<string, Prescription>();
   708→  for (const p of DOCTOR_DEMO_PRESCRIPTIONS) {
   709→    prescriptionMap.set(p.id, p);
   710→  }
   711→  for (const p of storePrescriptions) {
   712→    prescriptionMap.set(p.id, p);
   713→  }
   714→
   715→  // Step 4: Build consultationId → consultation lookup
   716→  const consById = new Map<string, Consultation>(consultationMap);
   717→
   718→  // Step 5: Build consultationId → prescription lookup
   719→  const rxByConsultation = new Map<string, Prescription>();
   720→  for (const [, rx] of prescriptionMap) {
   721→    if (rx.consultationId) {
   722→      rxByConsultation.set(rx.consultationId, rx);
   723→    }
   724→  }
   725→
   726→  // Step 6: Build patientId → patient name lookup from consultations
   727→  const patientNameById = new Map<string, string>();
   728→  for (const [, cons] of consById) {
   729→    if (cons.patient?.name) {
   730→      patientNameById.set(cons.patientId, cons.patient.name);
   731→    }
   732→  }
   733→  // Also fill from PATIENT_NAME_MAP
   734→  for (const [pid, pname] of Object.entries(PATIENT_NAME_MAP)) {
   735→    if (!patientNameById.has(pid)) {
   736→      patientNameById.set(pid, pname);
   737→    }
   738→  }
   739→
   740→  // Step 7: Build merged records with triple-matching strategy
   741→  const merged: MergedMedicalRecord[] = [];
   742→  let rmIndex = 0;
   743→  const sortedRecords = Array.from(recordMap.values()).sort(
   744→    (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime(),
   745→  );
   746→
   747→  for (const mr of sortedRecords) {
   748→    rmIndex++;
   749→
   750→    // Resolve patient name via triple-matching:
   751→    // 1. consultationId → consultation → patient.name
   752→    // 2. patientId → patientNameById (from consultations or PATIENT_NAME_MAP)
   753→    // 3. patient name key from PATIENT_NAME_MAP
   754→    let patientName = '';
   755→
   756→    // Strategy 1: consultationId → consultation → patient.name
   757→    if (mr.consultationId) {
   758→      const cons = consById.get(mr.consultationId);
   759→      if (cons?.patient?.name) {
   760→        patientName = cons.patient.name;
   761→      }
   762→    }
   763→
   764→    // Strategy 2: patientId → patientNameById
   765→    if (!patientName && mr.patientId) {
   766→      const name = patientNameById.get(mr.patientId);
   767→      if (name) {
   768→        patientName = name;
   769→      }
   770→    }
   771→
   772→    // Strategy 3: Try patient name key matching
   773→    if (!patientName && mr.patientId) {
   774→      const mappedName = PATIENT_NAME_MAP[mr.patientId];
   775→      if (mappedName) {
   776→        patientName = mappedName;
   777→      }
   778→    }
   779→
   780→    // Resolve consultation type
   781→    let consultationType: string | undefined;
   782→    if (mr.consultationId) {
   783→      const cons = consById.get(mr.consultationId);
   784→      if (cons) {
   785→        consultationType = cons.type;
   786→      }
   787→    }
   788→
   789→    // Resolve linked prescription
   790→    let linkedPrescription: Prescription | undefined;
   791→    if (mr.consultationId) {
   792→      linkedPrescription = rxByConsultation.get(mr.consultationId);
   793→    }
   794→
   795→    // Generate RM number if missing
   796→    const rmNumber = mr.rmNumber || generateRmNumber(mr, rmIndex - 1);
   797→
   798→    merged.push({
   799→      ...mr,
   800→      rmNumber,
   801→      patientName: patientName || mr.patientId,
   802→      consultationType,
   803→      linkedPrescription,
   804→    });
   805→  }
   806→
   807→  return merged;
   808→}

   527→// SCORING FUNCTIONS
   528→// ═══════════════════════════════════════════════════════════════════════════
   529→
   530→/**
   531→ * Get the max possible score for a single-choice question
   532→ */
   533→function getMaxScoreForQuestion(q: SocialNeedsQuestion): number {
   534→  if (q.type === 'text_area') return 0;
   535→  if (!q.options || q.options.length === 0) return 0;
   536→  return Math.max(...q.options.map((o) => o.score));
   537→}
   538→
   539→/**
   540→ * Calculate the score for a single-choice answer
   541→ */
   542→function getSingleChoiceScore(questionId: string, value: string): number {
   543→  const question = SOCIAL_NEEDS_QUESTIONS.find((q) => q.id === questionId);
   544→  if (!question || !question.options) return 0;
   545→  const option = question.options.find((o) => o.value === value);
   546→  return option?.score ?? 0;
   547→}
   548→
   549→/**
   550→ * Calculate the score for a multiple-choice answer
   551→ * Uses the highest individual score from selected options
   552→ * (selecting more risky options = higher overall risk)
   553→ */
   554→function getMultipleChoiceScore(questionId: string, values: string[]): number {
   555→  const question = SOCIAL_NEEDS_QUESTIONS.find((q) => q.id === questionId);
   556→  if (!question || !question.options) return 0;
   557→
   558→  let totalScore = 0;
   559→  for (const val of values) {
   560→    const option = question.options.find((o) => o.value === val);
   561→    totalScore += option?.score ?? 0;
   562→  }
   563→
   564→  // Cap at the max possible for this question type
   565→  const maxScore = getMaxScoreForQuestion(question);
   566→  // For multiple choice, the max is the sum of all positive scores
   567→  const sumMax = question.options.reduce((sum, o) => sum + o.score, 0);
   568→  return Math.min(totalScore, sumMax);
   569→}
   570→
   571→/**
   572→ * Get max possible score for a multiple choice question (sum of all positive scores)
   573→ */
   574→function getMaxScoreForMultipleChoice(q: SocialNeedsQuestion): number {
   575→  if (!q.options) return 0;
   576→  // For "tidak_ada" type options with high score, use that as max
   577→  const hasNegativeIndicator = q.options.some(o => o.score >= 3);
   578→  if (hasNegativeIndicator) {
   579→    // If there's a "no facility" option worth 3, max is 3
   580→    return Math.max(...q.options.map(o => o.score));
   581→  }
   582→  // Otherwise, max is sum of top 3 selections (realistic scenario)
   583→  const sortedScores = q.options.map(o => o.score).sort((a, b) => b - a);
   584→  return sortedScores.slice(0, Math.min(3, sortedScores.length)).reduce((sum, s) => sum + s, 0);
   585→}

   587→/**
   588→ * Determine risk level from percentage
   589→ */
   590→export function getRiskLevel(percentage: number): SocialNeedsRiskLevel {
   591→  if (percentage <= 25) return 'rendah';
   592→  if (percentage <= 50) return 'sedang';
   593→  if (percentage <= 75) return 'tinggi';
   594→  return 'sangat_tinggi';
   595→}

   613→/**
   614→ * Calculate full screening result from all answers
   615→ */
   616→export function calculateScreeningResult(
   617→  answers: Record<string, string | string[]>
   618→): SocialNeedsScreeningResult {
   619→  const categories = Object.keys(CATEGORY_META) as SocialNeedsCategory[];
   620→  const categoryScores: SocialNeedsCategoryScore[] = [];
   621→  let totalScore = 0;
   622→  let totalMaxScore = 0;
   623→
   624→  for (const category of categories) {
   625→    const categoryQuestions = SOCIAL_NEEDS_QUESTIONS.filter(
   626→      (q) => q.category === category
   627→    );
   628→
   629→    let catScore = 0;
   630→    let catMaxScore = 0;
   631→
   632→    for (const q of categoryQuestions) {
   633→      if (q.type === 'text_area') continue; // text areas don't contribute to score
   634→
   635→      const answer = answers[q.id];
   636→      if (!answer) continue;
   637→
   638→      // Calculate max score for this question
   639→      if (q.type === 'multiple_choice') {
   640→        catMaxScore += getMaxScoreForMultipleChoice(q);
   641→      } else {
   642→        catMaxScore += getMaxScoreForQuestion(q);
   643→      }
   644→
   645→      // Calculate actual score
   646→      if (q.type === 'single_choice' && typeof answer === 'string') {
   647→        catScore += getSingleChoiceScore(q.id, answer);
   648→      } else if (q.type === 'multiple_choice' && Array.isArray(answer)) {
   649→        catScore += getMultipleChoiceScore(q.id, answer);
   650→      }
   651→    }
   652→
   653→    // Skip categories with no scorable questions (like pertanyaan_terbuka)
   654→    if (catMaxScore === 0) continue;
   655→
   656→    const percentage = catMaxScore > 0 ? Math.round((catScore / catMaxScore) * 100) : 0;
   657→    categoryScores.push({
   658→      category,
   659→      categoryLabel: CATEGORY_META[category].label,
   660→      totalScore: catScore,
   661→      maxScore: catMaxScore,
   662→      percentage,
   663→      riskLevel: getRiskLevel(percentage),
   664→    });
   665→
   666→    totalScore += catScore;
   667→    totalMaxScore += catMaxScore;
   668→  }
   669→
   670→  const overallPercentage = totalMaxScore > 0 ? Math.round((totalScore / totalMaxScore) * 100) : 0;
   671→
   672→  return {
   673→    totalScore,
   674→    maxScore: totalMaxScore,
   675→    overallPercentage,
   676→    overallRiskLevel: getRiskLevel(overallPercentage),
   677→    categoryScores,
   678→    completedAt: new Date().toISOString(),
   679→  };
   680→}

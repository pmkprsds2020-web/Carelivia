   530→   530→/**
   531→   531→ * Get the max possible score for a single-choice question
   532→   532→ */
   533→   533→function getMaxScoreForQuestion(q: SocialNeedsQuestion): number {
   534→   534→  if (q.type === 'text_area') return 0;
   535→   535→  if (!q.options || q.options.length === 0) return 0;
   536→   536→  return Math.max(...q.options.map((o) => o.score));
   537→   537→}
   538→   538→
   539→   539→/**
   540→   540→ * Calculate the score for a single-choice answer
   541→   541→ */
   542→   542→function getSingleChoiceScore(questionId: string, value: string): number {
   543→   543→  const question = SOCIAL_NEEDS_QUESTIONS.find((q) => q.id === questionId);
   544→   544→  if (!question || !question.options) return 0;
   545→   545→  const option = question.options.find((o) => o.value === value);
   546→   546→  return option?.score ?? 0;
   547→   547→}
   548→   548→
   549→   549→/**
   550→   550→ * Calculate the score for a multiple-choice answer
   551→   551→ * Uses the highest individual score from selected options
   552→   552→ * (selecting more risky options = higher overall risk)
   553→   553→ */
   554→   554→function getMultipleChoiceScore(questionId: string, values: string[]): number {
   555→   555→  const question = SOCIAL_NEEDS_QUESTIONS.find((q) => q.id === questionId);
   556→   556→  if (!question || !question.options) return 0;
   557→   557→
   558→   558→  let totalScore = 0;
   559→   559→  for (const val of values) {
   560→   560→    const option = question.options.find((o) => o.value === val);
   561→   561→    totalScore += option?.score ?? 0;
   562→   562→  }
   563→   563→
   564→   564→  // Cap at the max possible for this question type
   565→   565→  const maxScore = getMaxScoreForQuestion(question);
   566→   566→  // For multiple choice, the max is the sum of all positive scores
   567→   567→  const sumMax = question.options.reduce((sum, o) => sum + o.score, 0);
   568→   568→  return Math.min(totalScore, sumMax);
   569→   569→}
   570→   570→
   571→   571→/**
   572→   572→ * Get max possible score for a multiple choice question (sum of all positive scores)
   573→   573→ */
   574→   574→function getMaxScoreForMultipleChoice(q: SocialNeedsQuestion): number {
   575→   575→  if (!q.options) return 0;
   576→   576→  // For "tidak_ada" type options with high score, use that as max
   577→   577→  const hasNegativeIndicator = q.options.some(o => o.score >= 3);
   578→   578→  if (hasNegativeIndicator) {
   579→   579→    // If there's a "no facility" option worth 3, max is 3
   580→   580→    return Math.max(...q.options.map(o => o.score));
   581→   581→  }
   582→   582→  // Otherwise, max is sum of top 3 selections (realistic scenario)
   583→   583→  const sortedScores = q.options.map(o => o.score).sort((a, b) => b - a);
   584→   584→  return sortedScores.slice(0, Math.min(3, sortedScores.length)).reduce((sum, s) => sum + s, 0);
   585→   585→}

   587→   587→/**
   588→   588→ * Determine risk level from percentage
   589→   589→ */
   590→   590→export function getRiskLevel(percentage: number): SocialNeedsRiskLevel {
   591→   591→  if (percentage <= 25) return 'rendah';
   592→   592→  if (percentage <= 50) return 'sedang';
   593→   593→  if (percentage <= 75) return 'tinggi';
   594→   594→  return 'sangat_tinggi';
   595→   595→}

   614→   614→ * Calculate full screening result from all answers
   615→   615→ */
   616→   616→export function calculateScreeningResult(
   617→   617→  answers: Record<string, string | string[]>
   618→   618→): SocialNeedsScreeningResult {
   619→   619→  const categories = Object.keys(CATEGORY_META) as SocialNeedsCategory[];
   620→   620→  const categoryScores: SocialNeedsCategoryScore[] = [];
   621→   621→  let totalScore = 0;
   622→   622→  let totalMaxScore = 0;
   623→   623→
   624→   624→  for (const category of categories) {
   625→   625→    const categoryQuestions = SOCIAL_NEEDS_QUESTIONS.filter(
   626→   626→      (q) => q.category === category
   627→   627→    );
   628→   628→
   629→   629→    let catScore = 0;
   630→   630→    let catMaxScore = 0;
   631→   631→
   632→   632→    for (const q of categoryQuestions) {
   633→   633→      if (q.type === 'text_area') continue; // text areas don't contribute to score
   634→   634→
   635→   635→      const answer = answers[q.id];
   636→   636→      if (!answer) continue;
   637→   637→
   638→   638→      // Calculate max score for this question
   639→   639→      if (q.type === 'multiple_choice') {
   640→   640→        catMaxScore += getMaxScoreForMultipleChoice(q);
   641→   641→      } else {
   642→   642→        catMaxScore += getMaxScoreForQuestion(q);
   643→   643→      }
   644→   644→
   645→   645→      // Calculate actual score
   646→   646→      if (q.type === 'single_choice' && typeof answer === 'string') {
   647→   647→        catScore += getSingleChoiceScore(q.id, answer);
   648→   648→      } else if (q.type === 'multiple_choice' && Array.isArray(answer)) {
   649→   649→        catScore += getMultipleChoiceScore(q.id, answer);
   650→   650→      }
   651→   651→    }
   652→   652→
   653→   653→    // Skip categories with no scorable questions (like pertanyaan_terbuka)
   654→   654→    if (catMaxScore === 0) continue;
   655→   655→
   656→   656→    const percentage = catMaxScore > 0 ? Math.round((catScore / catMaxScore) * 100) : 0;
   657→   657→    categoryScores.push({
   658→   658→      category,
   659→   659→      categoryLabel: CATEGORY_META[category].label,
   660→   660→      totalScore: catScore,
   661→   661→      maxScore: catMaxScore,
   662→   662→      percentage,
   663→   663→      riskLevel: getRiskLevel(percentage),
   664→   664→    });
   665→   665→
   666→   666→    totalScore += catScore;
   667→   667→    totalMaxScore += catMaxScore;
   668→   668→  }
   669→   669→
   670→   670→  const overallPercentage = totalMaxScore > 0 ? Math.round((totalScore / totalMaxScore) * 100) : 0;
   671→   671→
   672→   672→  return {
   673→   673→    totalScore,
   674→   674→    maxScore: totalMaxScore,
   675→   675→    overallPercentage,
   676→   676→    overallRiskLevel: getRiskLevel(overallPercentage),
   677→   677→    categoryScores,
   678→   678→    completedAt: new Date().toISOString(),
   679→   679→  };
   680→   680→}

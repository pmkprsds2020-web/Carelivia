   385→  // Simple Markdown renderer
   386→  const renderMarkdown = (text: string) => {
   387→    const lines = text.split('\n');
   388→    const elements: React.ReactNode[] = [];
   389→    let inList = false;
   390→    let listItems: React.ReactNode[] = [];
   391→    let listKey = 0;
   392→
   393→    const flushList = () => {
   394→      if (listItems.length > 0) {
   395→        elements.push(<ul key={`list-${listKey++}`} className="list-disc pl-5 space-y-1 my-2">{listItems}</ul>);
   396→        listItems = [];
   397→        inList = false;
   398→      }
   399→    };
   400→
   401→    lines.forEach((line, idx) => {
   402→      // Headings
   403→      if (line.startsWith('## ')) {
   404→        flushList();
   405→        elements.push(<h2 key={idx} className="text-base font-bold text-foreground mt-4 mb-2">{line.slice(3)}</h2>);
   406→        return;
   407→      }
   408→      if (line.startsWith('### ')) {
   409→        flushList();
   410→        elements.push(<h3 key={idx} className="text-sm font-bold text-foreground mt-3 mb-1">{line.slice(4)}</h3>);
   411→        return;
   412→      }
   413→
   414→      // List items
   415→      const listMatch = line.match(/^- (.+)/);
   416→      if (listMatch) {
   417→        inList = true;
   418→        const content = listMatch[1];
   419→        listItems.push(<li key={idx} className="text-sm text-foreground">{renderInlineMarkdown(content)}</li>);
   420→        return;
   421→      }
   422→
   423→      flushList();
   424→
   425→      // Horizontal rule
   426→      if (line.match(/^---+$/)) {
   427→        elements.push(<Separator key={idx} className="my-3" />);
   428→        return;
   429→      }
   430→
   431→      // Empty line
   432→      if (line.trim() === '') {
   433→        elements.push(<div key={idx} className="h-2" />);
   434→        return;
   435→      }
   436→
   437→      // Regular paragraph
   438→      elements.push(<p key={idx} className="text-sm text-foreground leading-relaxed">{renderInlineMarkdown(line)}</p>);
   439→    });
   440→
   441→    flushList();
   442→    return elements;
   443→  };

   445→  const renderInlineMarkdown = (text: string): React.ReactNode => {
   446→    // Split by bold patterns first, then italic within each segment
   447→    const segments: React.ReactNode[] = [];
   448→    let key = 0;
   449→
   450→    // Process bold: **text**
   451→    const boldParts = text.split(/(\*\*[^*]+\*\*)/g);
   452→    for (const part of boldParts) {
   453→      const boldMatch = part.match(/^\*\*(.+)\*\*$/);
   454→      if (boldMatch) {
   455→        segments.push(React.createElement('strong', { key: `b-${key++}`, className: 'font-semibold' }, boldMatch[1]));
   456→      } else {
   457→        // Process italic within non-bold parts: *text*
   458→        const italicParts = part.split(/(\*[^*]+\*)/g);
   459→        for (const ipart of italicParts) {
   460→          const italicMatch = ipart.match(/^\*(.+)\*$/);
   461→          if (italicMatch) {
   462→            segments.push(React.createElement('em', { key: `i-${key++}`, className: 'italic' }, italicMatch[1]));
   463→          } else {
   464→            segments.push(ipart);
   465→          }
   466→        }
   467→      }
   468→    }
   469→
   470→    if (segments.length === 0) return text;
   471→    if (segments.length === 1) return segments[0];
   472→    return React.createElement(React.Fragment, { key: `frag-${key++}` }, ...segments);
   473→  };

   326→function generateId(): string {
   327→  return `sk-${Date.now()}-${Math.random().toString(36).substring(2, 9)}`;
   328→}
   329→
   330→function formatDate(dateStr: string): string {
   331→  try {
   332→    return new Date(dateStr).toLocaleDateString('id-ID', {
   333→      day: 'numeric', month: 'short', year: 'numeric',
   334→      hour: '2-digit', minute: '2-digit',
   335→    });
   336→  } catch { return dateStr; }
   337→}
   338→
   339→function vasColor(value: number): string {
   340→  if (value <= 3) return 'text-emerald-600';
   341→  if (value <= 6) return 'text-amber-600';
   342→  return 'text-red-600';
   343→}
   344→
   345→function vasBg(value: number): string {
   346→  if (value <= 3) return 'bg-emerald-500';
   347→  if (value <= 6) return 'bg-amber-500';
   348→  return 'bg-red-500';
   349→}
   350→
   351→function getEwsBadge(level: 'merah' | 'kuning' | 'hijau'): { label: string; color: string; bg: string } {
   352→  switch (level) {
   353→    case 'merah': return { label: 'Kritis', color: 'text-red-700', bg: 'bg-red-100 border-red-300' };
   354→    case 'kuning': return { label: 'Perhatian', color: 'text-amber-700', bg: 'bg-amber-100 border-amber-300' };
   355→    case 'hijau': return { label: 'Normal', color: 'text-emerald-700', bg: 'bg-emerald-100 border-emerald-300' };
   356→  }
   357→}

   675→  const calcEORTC = useCallback((): { physicalFunction: number; symptomBurden: number; globalQol: number } => {
   676→    // Physical Function: Q1-Q7 (invert: higher raw = worse, so we invert for PF score)
   677→    // EORTC scoring: RS = (raw - 1)/3 * 100 for 4-point items
   678→    // PF: average of Q1-Q7 RS, then PF = 100 - RS (higher = better)

   680→    for (let i = 0; i < 7; i++) {

   696→    const globalQol = Math.round((q15Raw - 1) / 6 * 100);

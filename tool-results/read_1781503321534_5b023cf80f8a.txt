   141→function formatDate(dateStr: string | null | undefined): string {
   142→  if (!dateStr) return '-';
   143→  try {
   144→    return new Date(dateStr).toLocaleDateString('id-ID', {
   145→      day: 'numeric',
   146→      month: 'long',
   147→      year: 'numeric',
   148→    });
   149→  } catch {
   150→    return dateStr;
   151→  }
   152→}
   153→
   154→function formatDateTime(dateStr: string | null | undefined): string {
   155→  if (!dateStr) return '-';
   156→  try {
   157→    return new Date(dateStr).toLocaleDateString('id-ID', {
   158→      day: 'numeric',
   159→      month: 'short',
   160→      year: 'numeric',
   161→      hour: '2-digit',
   162→      minute: '2-digit',
   163→    });
   164→  } catch {
   165→    return dateStr;
   166→  }
   167→}

   169→function genId(prefix: string): string {
   170→  return `${prefix}-${Date.now()}-${Math.random().toString(36).substring(2, 7)}`;
   171→}

   173→function genDocNumber(prefix: string): string {
   174→  const now = new Date();
   175→  const y = now.getFullYear();
   176→  const m = String(now.getMonth() + 1).padStart(2, '0');
   177→  const seq = String(Math.floor(Math.random() * 9000) + 1000);
   178→  return `${prefix}/${y}/${m}/${seq}`;
   179→}

   209→function getTrendBadge(trend: string) {
   210→  const t = trend?.toLowerCase() || '';
   211→  if (t.includes('membaik')) return { label: 'Membaik', className: 'bg-green-100 text-green-800 border-green-300' };
   212→  if (t.includes('stabil')) return { label: 'Stabil', className: 'bg-blue-100 text-blue-800 border-blue-300' };
   213→  if (t.includes('fluktuatif')) return { label: 'Fluktuatif', className: 'bg-amber-100 text-amber-800 border-amber-300' };
   214→  if (t.includes('memburuk')) return { label: 'Memburuk', className: 'bg-orange-100 text-orange-800 border-orange-300' };
   215→  if (t.includes('terminal')) return { label: 'Terminal', className: 'bg-red-100 text-red-800 border-red-300' };
   216→  if (t.includes('end of life') || t.includes('end_of_life')) return { label: 'End of Life', className: 'bg-red-200 text-red-900 border-red-400' };
   217→  return { label: trend || 'N/A', className: 'bg-slate-100 text-slate-700 border-slate-300' };
   218→}

   220→function getSeverityBadge(level: string | undefined) {
   221→  if (!level) return null;
   222→  const l = level.toLowerCase();
   223→  if (l.includes('ringan')) return <Badge className="bg-green-100 text-green-800 border border-green-300 text-[10px]">Ringan</Badge>;
   224→  if (l.includes('sedang')) return <Badge className="bg-amber-100 text-amber-800 border border-amber-300 text-[10px]">Sedang</Badge>;
   225→  if (l.includes('berat') || l.includes('kritis')) return <Badge className="bg-red-100 text-red-800 border border-red-300 text-[10px]">Berat</Badge>;
   226→  return <Badge variant="outline" className="text-[10px]">{level}</Badge>;
   227→}

   241→// ── Generic record renderer for unknown[] data ──
   242→function renderGenericRecords(records: unknown[], title: string): React.ReactNode {
   243→  if (!records || !Array.isArray(records) || records.length === 0) return null;
   244→  return (
   245→    <div className="space-y-3 mb-3">
   246→      <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">{title} ({records.length})</h4>
   247→      {records.map((rec, idx) => {
   248→        if (typeof rec !== 'object' || rec === null) return null;
   249→        const entries = Object.entries(rec as Record<string, unknown>)
   250→          .filter(([k, v]) => v != null && v !== '' && !['id', 'patientId', 'createdAt', 'updatedAt'].includes(k));
   251→        if (entries.length === 0) return null;
   252→        return (
   253→          <div key={idx} className="rounded-lg border p-2.5 bg-muted/20 text-sm">
   254→            {entries.map(([key, val]) => (
   255→              <div key={key} className="flex items-start gap-2 py-0.5">
   256→                <span className="text-xs text-muted-foreground w-36 shrink-0">{key.replace(/([A-Z])/g, ' $1').replace(/^./, s => s.toUpperCase())}</span>
   257→                <span className="text-xs">{String(val)}</span>
   258→              </div>
   259→            ))}
   260→          </div>
   261→        );
   262→      })}
   263→    </div>
   264→  );
   265→}

   318→  const keluhan = doc.keluhanHarian as PalliativeResumeKeluhan | undefined;
   319→  const esas = doc.esasScores as { skorAwal: Record<string, unknown> | null; skorTertinggi: Record<string, unknown> | null; skorTerakhir: Record<string, unknown> | null } | undefined;
   320→  const obat = doc.obat as Record<string, unknown> | undefined;
   321→  const nutrisi = doc.nutrisi as { catatan?: unknown[]; ringkasan?: string } | undefined;

   324→  const ai = doc.aiAnalysis as PalliativeResumeAIAnalysis | undefined;
   325→
   326→  const escHtml = (s: string | null | undefined) => (s ?? '-').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
   327→
   328→  const sectionHeader = (title: string) => `<h2 style="background:#f0fdf8;color:#1a5c4a;padding:6px 10px;margin:18px 0 8px;font-size:12pt;border-left:4px solid #2D8C7A;">${title}</h2>`;

   330→  const kvTable = (rows: string[]) => `<table style="width:100%;border-collapse:collapse;">${rows.join('')}</table>`;
   331→
   332→  let html = '';
   333→
   334→  // DATA PASIEN
   335→  html += sectionHeader('DATA PASIEN');

   337→    kvRow('Nama', dp?.nama || doc.patientName),
   338→    kvRow('No. RM', dp?.noRM || doc.rmNumber),

   340→    kvRow('Tanggal Lahir', dp?.tanggalLahir),

   344→    kvRow('Diagnosa Penyerta', dp?.diagnosaPenyerta),
   345→    kvRow('Stadium', dp?.stadiumPenyakit),

   347→    kvRow('Tingkat Risiko', dp?.tingkatRisiko),

   352→  if (ttv?.ttvAwal || ttv?.ttvKritis || ttv?.ttvTerakhir) {
   353→    html += sectionHeader('TTV SERIAL');

   359→          kvRow('Nadi', d.nadi != null ? `${d.nadi} x/mnt` : '-'),
   360→          kvRow('RR', d.rr != null ? `${d.rr} x/mnt` : '-'),
   361→          kvRow('Suhu', d.suhu != null ? `${d.suhu}°C` : '-'),
   362→          kvRow('SpO2', d.spo2 != null ? `${d.spo2}%` : '-'),

String(awal) : '-'}</td><td style="padding:3px 6px;border:1px solid #ddd;text-align:center;">${tinggi != null ?    424→    };
   425→    html += renderMedBlock('Analgesik', obat.analgesik as unknown[] ?? []);

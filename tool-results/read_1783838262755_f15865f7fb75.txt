     1→// ───────────────────────────────────────────────────────────────────────────
     2→// supportingExamService — CRUD for Pemeriksaan Penunjang
     3→// (Laboratorium, USG, EKG, Radiologi)
     4→// ───────────────────────────────────────────────────────────────────────────
     5→//
     6→// STORAGE STRATEGY
     7→// ─────────────────────────
     8→// DDL access is unavailable, so we reuse the existing `patient_documents`
     9→// table for all four supporting examination types. The table already has a
    10→// `jenis` CHECK constraint that allows 'lab', 'radiologi', 'gambar', 'pdf',
    11→// 'lainnya' — perfect for our use case.
    12→//
    13→//   patient_documents(
    14→//     id           uuid PK,
    15→//     patient_id   uuid NOT NULL FK,
    16→//     jenis        text NOT NULL CHECK in (lab, radiologi, gambar, pdf, lainnya),
    17→//     nama_file    text NOT NULL,
    18→//     storage_path text NOT NULL,
    19→//     url          text,
    20→//     keterangan   text,        -- we store a JSON string here for structured data
    21→//     tanggal      date NOT NULL DEFAULT current_date,
    22→//     uploaded_by  text,
    23→//     created_at   timestamptz
    24→//   )
    25→//
    26→// Mapping:
    27→//   • Laboratorium  → jenis='lab',       keterangan=JSON{type, gdp, gds, hba1c, ...}
    28→//   • USG           → jenis='gambar',    keterangan=JSON{type='usg', jenis_usg, hasil, catatan, doctor_id}
    29→//                     + foto uploaded to Storage (foto_url stored in `url`)
    30→//   • EKG           → jenis='gambar',    keterangan=JSON{type='ekg', interpretasi, catatan, doctor_id}
    31→//                     + foto uploaded to Storage (foto_url stored in `url`)
    32→//   • Radiologi     → jenis='radiologi', keterangan=JSON{type='radiology', jenis_radiologi, hasil, catatan, doctor_id}
    33→//                     + foto uploaded to Storage (foto_url stored in `url`)
    34→//
    35→// Photos are uploaded to the `patient-files` Storage bucket (same as
    36→// documentService) under path: {patientId}/{jenis}/{timestamp}-{filename}
    37→//
    38→// The DDL for future dedicated tables (laboratory_results, ultrasound_results,
    39→// ecg_results, radiology_results, supporting_examinations) is documented in
    40→// supabase/schema.sql for when DDL access becomes available.

   199→function safeParseKeterangan(raw: any): Record<string, any> {
   200→  if (raw == null) return {};
   201→  if (typeof raw === 'object') return raw as Record<string, any>;
   202→  if (typeof raw === 'string') {
   203→    try {
   204→      return JSON.parse(raw) as Record<string, any>;
   205→    } catch {
   206→      return {};
   207→    }
   208→  }
   209→  return {};
   210→}

   284→// ── Storage helpers (server-side via API routes to bypass RLS) ──────────────
   285→//
   286→// The browser anon client is subject to Storage RLS and gets:
   287→//   "new row violates row-level security policy"
   288→// when uploading to the `patient-files` bucket. All Storage + DB operations
   289→// for photo exams therefore go through server-side API routes that use the
   290→// service-role key (getSupabaseAdmin) to bypass RLS.
   291→//
   292→// Routes:
   293→//   POST /api/supporting-exams/upload       — upload file + INSERT row
   294→//   POST /api/supporting-exams/update       — optional new file + UPDATE row
   295→//   POST /api/supporting-exams/delete-file  — delete file + DELETE row
   296→//
   297→
   298→export type UploadProgressCb = (phase: 'uploading' | 'inserting' | 'done' | 'error', pct: number, msg?: string) => void;
   299→
   300→/**
   301→ * Call the /api/supporting-exams/upload route.
   302→ * Uploads a file to Storage AND inserts a metadata row in one atomic call.
   303→ * Returns the inserted DB row (raw snake_case) or throws on failure.
   304→ */
   305→async function callUploadApi(
   306→  file: File | Blob,
   307→  metadata: Record<string, any>,
   308→  onProgress?: UploadProgressCb
   309→): Promise<any> {
   310→  const fileName = (file as File).name ?? `upload-${Date.now()}`;
   311→  onProgress?.('uploading', 10, `Mengunggah ${fileName}...`);
   312→
   313→  const fd = new FormData();
   314→  // Coerce to File if it's a Blob (some callers pass Blob without name)
   315→  if (!(file instanceof File) && file instanceof Blob) {
   316→    fd.append('file', file, fileName);
   317→  } else {
   318→    fd.append('file', file as File);
   319→  }
   320→  fd.append('metadata', JSON.stringify(metadata));
   321→
   322→  onProgress?.('uploading', 30);
   323→  let res: Response;
   324→  try {
   325→    res = await fetch('/api/supporting-exams/upload', {
   326→      method: 'POST',
   327→      body: fd,
   328→    });
   329→  } catch (e: any) {
   330→    onProgress?.('error', 0, e?.message ?? 'Network error');
   331→    throw new Error('Upload gagal (network): ' + (e?.message ?? String(e)));
   332→  }
   333→  onProgress?.('uploading', 80);
   334→
   335→  let body: any;
   336→  try {
   337→    body = await res.json();
   338→  } catch {
   339→    body = { error: 'Invalid JSON response from server' };
   340→  }
   341→
   342→  if (!res.ok || !body?.ok) {
   343→    const msg = body?.error || `HTTP ${res.status}`;
   344→    onProgress?.('error', 0, msg);
   345→    // Surface a clear error so the UI can toast it
   346→    const err = new Error(msg);
   347→    (err as any).code = body?.code;
   348→    (err as any).status = res.status;
   349→    throw err;
   350→  }
   351→
   352→  onProgress?.('done', 100);
   353→  return body.row;
   354→}

   356→/**
   357→ * Call the /api/supporting-exams/update route.
   358→ * Optionally uploads a new file (and deletes the old one), then UPDATEs the row.
   359→ */
   360→async function callUpdateApi(
   361→  id: string,
   362→  metadata: Record<string, any>,
   363→  file?: File | Blob | null,
   364→  oldStoragePath?: string,
   365→  onProgress?: UploadProgressCb
   366→): Promise<any> {
   367→  const fd = new FormData();
   368→  fd.append('id', id);
   369→  fd.append('metadata', JSON.stringify(metadata));
   370→  if (oldStoragePath) fd.append('oldStoragePath', oldStoragePath);
   371→  if (file) {
   372→    const fileName = (file as File).name ?? `upload-${Date.now()}`;
   373→    onProgress?.('uploading', 20, `Mengunggah ${fileName}...`);
   374→    if (!(file instanceof File) && file instanceof Blob) {
   375→      fd.append('file', file, fileName);
   376→    } else {
   377→      fd.append('file', file as File);
   378→    }
   379→  } else {
   380→    onProgress?.('inserting', 50, 'Menyimpan perubahan...');
   381→  }
   382→
   383→  let res: Response;
   384→  try {
   385→    res = await fetch('/api/supporting-exams/update', {
   386→      method: 'POST',
   387→      body: fd,
   388→    });
   389→  } catch (e: any) {
   390→    onProgress?.('error', 0, e?.message ?? 'Network error');
   391→    throw new Error('Update gagal (network): ' + (e?.message ?? String(e)));
   392→  }
   393→
   394→  let body: any;
   395→  try {
   396→    body = await res.json();
   397→  } catch {
   398→    body = { error: 'Invalid JSON response from server' };
   399→  }
   400→
   401→  if (!res.ok || !body?.ok) {
   402→    const msg = body?.error || `HTTP ${res.status}`;
   403→    onProgress?.('error', 0, msg);
   404→    const err = new Error(msg);
   405→    (err as any).code = body?.code;
   406→    (err as any).status = res.status;
   407→    throw err;
   408→  }
   409→
   410→  onProgress?.('done', 100);
   411→  return body.row;
   412→}

   415→ * Call the /api/supporting-exams/delete-file route.
   416→ * Deletes the file from Storage AND the metadata row.
   417→ */
   418→async function callDeleteApi(id: string, storagePath?: string): Promise<boolean> {
   419→  let res: Response;
   420→  try {
   421→    res = await fetch('/api/supporting-exams/delete-file', {
   422→      method: 'POST',
   423→      headers: { 'Content-Type': 'application/json' },
   424→      body: JSON.stringify({ id, storagePath }),
   425→    });
   426→  } catch (e: any) {
   427→    console.error('[supportingExamService.callDeleteApi] network error:', e);
   428→    return false;
   429→  }
   430→
   431→  let body: any;
   432→  try {
   433→    body = await res.json();
   434→  } catch {
   435→    body = {};
   436→  }
   437→
   438→  if (!res.ok || !body?.ok) {
   439→    console.error('[supportingExamService.callDeleteApi] failed:', body?.error ?? res.status);
   440→    return false;
   441→  }
   442→  return true;
   443→}

   466→  async createLab(input: LabInput): Promise<LabResult | null> {

   469→      return null;

   491→      storage_path: `lab/${input.patientId}/${Date.now()}`,
   492→      keterangan,
   493→      tanggal: input.tanggal ?? todayStr(),
   494→      uploaded_by: input.createdBy,
   495→    };
   496→    const { data: row, error } = await safeInsert<any>(
   497→      supabase.from('patient_documents').insert(payload).select().single(),
   498→      'supportingExamService.createLab'
   499→    );
   500→    if (error) throw new Error(error);
   501→    const created = row ? fromDbLab(row) : null;

   505→    // LDL ≥ 190, Kreatinin meningkat, Mikroalbumin positif — maka otomatis

   507→    // and persists via clinicalAlertService. Fire-and-forget — never blocks
   508→    // the lab save.
   509→    if (created) {
   510→      try {
   511→        // Lazy-import to avoid circular dependency at module load time.
   512→        const { evaluateAndPersist } = await import('./clinicalAlertEngine');
   513→        evaluateAndPersist({
   514→          patientId: input.patientId,
   515→          doctorId: input.doctorId,
   516→          vitals: [],
   517→          screenings: [],
   518→          medications: [],
   519→          nutrition: [],
   520→          dailyComplaints: [],
   521→          socialAssessments: [],
   522→          labResults: [created],
   523→        }).catch((e) =>
   524→          console.error('[supportingExamService.createLab] alert engine error:', e)
   525→        );
   526→      } catch (e) {
   527→        console.error('[supportingExamService.createLab] failed to trigger alert engine:', e);
   528→      }
   529→    }
   530→
   531→    return created;
   532→  },

  1016→  async listAll(patientId: string): Promise<SupportingExamUnion[]> {
  1017→    if (!isValidUuid(patientId)) return [];
  1018→    const [lab, usg, ecg, rad] = await Promise.all([
  1019→      this.listLab(patientId),
  1020→      this.listUsg(patientId),
  1021→      this.listEcg(patientId),
  1022→      this.listRadiology(patientId),
  1023→    ]);
  1024→    const all: SupportingExamUnion[] = [
  1025→      ...lab.map((data) => ({ type: 'laboratorium' as const, data })),
  1026→      ...usg.map((data) => ({ type: 'usg' as const, data })),
  1027→      ...ecg.map((data) => ({ type: 'ekg' as const, data })),
  1028→      ...rad.map((data) => ({ type: 'radiologi' as const, data })),
  1029→    ];
  1030→    all.sort((a, b) => {
  1031→      const ta = new Date((a.data as any).tanggal ?? (a.data as any).createdAt).getTime();
  1032→      const tb = new Date((b.data as any).tanggal ?? (b.data as any).createdAt).getTime();
  1033→      return tb - ta;
  1034→    });
  1035→    return all;
  1036→  },

    83→
    84→function RiskGauge({
    85→  percentage,
    86→  size = 120,
    87→  strokeWidth = 10,
    88→  label,
    89→  riskLevel,
    90→  showLabel = true,
    91→}: {
    92→  percentage: number;
    93→  size?: number;
    94→  strokeWidth?: number;
    95→  label?: string;
    96→  riskLevel: SocialNeedsRiskLevel;
    97→  showLabel?: boolean;
    98→}) {
    99→  const radius = (size - strokeWidth) / 2;
   100→  const circumference = 2 * Math.PI * radius;
   101→  const offset = circumference - (percentage / 100) * circumference;
   102→  const display = getRiskLevelDisplay(riskLevel);
   103→
   104→  const colorMap: Record<SocialNeedsRiskLevel, string> = {
   105→    rendah: '#16A34A',
   106→    sedang: '#CA8A04',
   107→    tinggi: '#EA580C',
   108→    sangat_tinggi: '#DC2626',
   109→  };
   110→
   111→  return (
   112→    <div className="flex flex-col items-center gap-1">
   113→      <div className="relative" style={{ width: size, height: size }}>
   114→        <svg width={size} height={size} className="-rotate-90">
   115→          <circle
   116→            cx={size / 2}
   117→            cy={size / 2}
   118→            r={radius}
   119→            fill="none"
   120→            stroke="#E5E7EB"
   121→            strokeWidth={strokeWidth}
   122→          />
   123→          <circle
   124→            cx={size / 2}
   125→            cy={size / 2}
   126→            r={radius}
   127→            fill="none"
   128→            stroke={colorMap[riskLevel]}
   129→            strokeWidth={strokeWidth}
   130→            strokeDasharray={circumference}
   131→            strokeDashoffset={offset}
   132→            strokeLinecap="round"
   133→            className="transition-all duration-700 ease-out"
   134→          />
   135→        </svg>
   136→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   137→          <span className="text-2xl font-bold" style={{ color: colorMap[riskLevel] }}>
   138→            {percentage}%
   139→          </span>
   140→          <span className="text-xs text-muted-foreground">skor</span>
   141→        </div>
   142→      </div>
   143→      {showLabel && label && (
   144→        <span className="text-xs font-medium text-center mt-1 max-w-[100px]">{label}</span>
   145→      )}
   146→      {showLabel && (
   147→        <Badge
   148→          className="text-[10px] px-1.5 py-0"
   149→          style={{
   150→            backgroundColor: display.bgColor,
   151→            color: display.color,
   152→            borderColor: display.borderColor,
   153→            border: `1px solid ${display.borderColor}`,
   154→          }}
   155→        >
   156→          {display.label}
   157→        </Badge>
   158→      )}
   159→    </div>
   160→  );
   161→}

   334→// MAIN COMPONENT
   335→// ═══════════════════════════════════════════════════════════════════════════
   336→
   337→export function SocialNeedsScreeningPanel() {
   338→  const { currentUser } = useStore();
   339→  const { toast } = useToast();
   340→
   341→  // ── State ────────────────────────────────────────────────────────────────
   342→  const [activeTab, setActiveTab] = useState<string>('screening');
   343→  const [currentCategoryIndex, setCurrentCategoryIndex] = useState(0);
   344→  const [answers, setAnswers] = useState<Record<string, string | string[]>>({});
   345→  const [screeningResult, setScreeningResult] = useState<SocialNeedsScreeningResult | null>(null);
   346→  const [aiResult, setAiResult] = useState<SocialNeedsAIResult | null>(null);
   347→  const [aiLoading, setAiLoading] = useState(false);
   348→  const [isCompleted, setIsCompleted] = useState(false);
   349→
   350→  // ── Derived data ─────────────────────────────────────────────────────────
   351→  const questionsByCategory = useMemo(() => getQuestionsByCategory(), []);
   352→  const currentCategory = CATEGORY_ORDER[currentCategoryIndex];
   353→  const currentQuestions = questionsByCategory[currentCategory] || [];
   354→  const totalQuestions = SOCIAL_NEEDS_QUESTIONS.length;
   355→  const answeredCount = Object.keys(answers).length;
   356→
   357→  // Count answered required questions
   358→  const answeredRequiredCount = useMemo(() => {
   359→    return SOCIAL_NEEDS_QUESTIONS.filter(
   360→      (q) => q.required && answers[q.id] !== undefined && answers[q.id] !== ''
   361→    ).length;
   362→  }, [answers]);
   363→
   364→  const totalRequiredCount = SOCIAL_NEEDS_QUESTIONS.filter((q) => q.required).length;
   365→
   366→  // Category progress
   367→  const categoryProgress = useMemo(() => {
   368→    const progress: Record<string, { answered: number; total: number }> = {};
   369→    for (const cat of CATEGORY_ORDER) {
   370→      const qs = questionsByCategory[cat] || [];
   371→      const answered = qs.filter(
   372→        (q) => answers[q.id] !== undefined && answers[q.id] !== ''
   373→      ).length;
   374→      progress[cat] = { answered, total: qs.length };
   375→    }
   376→    return progress;
   377→  }, [answers, questionsByCategory]);
   378→
   379→  // Live screening result
   380→  const liveResult = useMemo(() => {
   381→    if (answeredCount === 0) return null;
   382→    return calculateScreeningResult(answers);
   383→  }, [answers, answeredCount]);
   384→
   385→  // ── Handlers ─────────────────────────────────────────────────────────────
   386→  const handleSingleChoice = useCallback((questionId: string, value: string) => {
   387→    setAnswers((prev) => ({ ...prev, [questionId]: value }));
   388→  }, []);
   389→
   390→  const handleMultipleChoice = useCallback(
   391→    (questionId: string, value: string, checked: boolean) => {
   392→      setAnswers((prev) => {
   393→        const current = (prev[questionId] as string[]) || [];
   394→        if (checked) {
   395→          return { ...prev, [questionId]: [...current, value] };
   396→        } else {
   397→          return { ...prev, [questionId]: current.filter((v) => v !== value) };
   398→        }
   399→      });
   400→    },
   401→    []
   402→  );
   403→
   404→  const handleTextArea = useCallback((questionId: string, value: string) => {
   405→    setAnswers((prev) => ({ ...prev, [questionId]: value }));
   406→  }, []);
   407→
   408→  const handlePrevious = useCallback(() => {
   409→    setCurrentCategoryIndex((prev) => Math.max(0, prev - 1));
   410→  }, []);
   411→
   412→  const handleNext = useCallback(() => {
   413→    setCurrentCategoryIndex((prev) =>
   414→      Math.min(CATEGORY_ORDER.length - 1, prev + 1)
   415→    );
   416→  }, []);
   417→
   418→  const handleSaveDraft = useCallback(() => {
   419→    toast({
   420→      title: 'Draft Tersimpan',
   421→      description: `${answeredCount} dari ${totalQuestions} pertanyaan telah disimpan.`,
   422→    });
   423→  }, [answeredCount, totalQuestions, toast]);
   424→
   425→  const handleCompleteScreening = useCallback(() => {
   426→    // Check if all required questions are answered
   427→    const unansweredRequired = SOCIAL_NEEDS_QUESTIONS.filter(
   428→      (q) => q.required && (!answers[q.id] || answers[q.id] === '')
   429→    );
   430→
   431→    if (unansweredRequired.length > 0) {
   432→      toast({
   433→        title: 'Pertanyaan Wajib Belum Lengkap',
   434→        description: `Masih ada ${unansweredRequired.length} pertanyaan wajib yang belum dijawab.`,
   435→        variant: 'destructive',
   436→      });
   437→      return;
   438→    }
   439→
   440→    const result = calculateScreeningResult(answers);
   441→    setScreeningResult(result);
   442→    setIsCompleted(true);
   443→    setActiveTab('results');
   444→    toast({
   445→      title: 'Skrining Selesai',
   446→      description: `Skor keseluruhan: ${result.overallPercentage}% (${getRiskLevelDisplay(result.overallRiskLevel).label})`,
   447→    });
   448→  }, [answers, toast]);
   449→
   450→  const handleResetForm = useCallback(() => {
   451→    setAnswers({});
   452→    setScreeningResult(null);
   453→    setAiResult(null);
   454→    setIsCompleted(false);
   455→    setCurrentCategoryIndex(0);
   456→    setActiveTab('screening');
   457→    toast({ title: 'Form Direset', description: 'Semua jawaban telah dihapus.' });
   458→  }, [toast]);
   459→
   460→  const handleRunAI = useCallback(async () => {
   461→    const result = screeningResult || liveResult;
   462→    if (!result) return;
   463→
   464→    setAiLoading(true);
   465→    setAiResult(null);
   466→
   467→    try {
   468→      const response = await fetch('/api/social-needs-screening-ai', {
   469→        method: 'POST',
   470→        headers: { 'Content-Type': 'application/json' },
   471→        body: JSON.stringify({
   472→          screeningResult: result,
   473→          answers,
   474→          patientData: {
   475→            name: currentUser?.name,
   476→            role: currentUser?.role,
   477→          },
   478→        }),
   479→      });
   480→
   481→      if (response.ok) {
   482→        const data = await response.json();
   483→        // Map API response to SocialNeedsAIResult format
   484→        const mapped: SocialNeedsAIResult = {
   485→          familySupportScore: data.familySupportScore || data.familySupportAnalysis?.familySupportScore > 70 ? 'tinggi' : 'rendah',
   486→          socialRiskScore: result.overallRiskLevel,
   487→          caregiverBurnoutScore: data.caregiverAnalysis?.status === 'berat' ? 'sangat_tinggi' : data.caregiverAnalysis?.status === 'sedang' ? 'tinggi' : data.caregiverAnalysis?.status === 'ringan' ? 'sedang' : 'rendah',
   488→          accessToCareScore: data.transportAnalysis?.accessRiskLevel === 'tinggi' ? 'tinggi' : data.transportAnalysis?.accessRiskLevel === 'sedang' ? 'sedang' : 'rendah',
   489→          financialRiskScore: data.financialAnalysis?.economicConstraintLevel === 'tinggi' ? 'tinggi' : data.financialAnalysis?.economicConstraintLevel === 'sedang' ? 'sedang' : 'rendah',
   490→          socialIsolationScore: data.socialRisks?.some((r: { riskType: string }) => r.riskType === 'isolasi_sosial') ? 'tinggi' : 'rendah',
   491→          recommendations: (data.actionPlan || []).map((a: { action: string; priority: string; reason?: string; category?: string }, i: number) => ({
   492→            priority: i + 1,
   493→            action: a.action,
   494→            reason: a.reason || `Prioritas: ${a.priority}`,
   495→            category: mapActionCategory(a.category),
   496→          })),
   497→          analysisSummary: data.socialConditionSummary || 'Analisis AI selesai.',
   498→          earlyWarnings: (data.earlyWarnings || []).map((w: { type: string; severity: string; title: string; description: string }) => ({
   499→            type: w.type,
   500→            severity: w.severity || 'warning',
   501→            title: w.title,
   502→            description: w.description,
   503→          })),
   504→          generatedAt: data.generatedAt || new Date().toISOString(),
   505→        };
   506→        setAiResult(mapped);
   507→      } else {
   508→        throw new Error('API failed');
   509→      }
   510→    } catch {
   511→      // Fallback to local AI analysis
   512→      const localResult = generateLocalAIAnalysis(result, answers);
   513→      setAiResult(localResult);
   514→      toast({
   515→        title: 'Menggunakan Analisis Lokal',
   516→        description: 'API AI tidak tersedia, menggunakan analisis lokal sebagai fallback.',
   517→      });
   518→    } finally {
   519→      setAiLoading(false);
   520→    }
   521→  }, [screeningResult, liveResult, answers, currentUser, toast]);
   522→
   523→  // ── Render: Question ─────────────────────────────────────────────────────
   524→
   525→  const renderQuestion = (question: SocialNeedsQuestion) => {
   526→    const currentAnswer = answers[question.id];
   527→
   528→    return (
   529→      <div key={question.id} className="space-y-3 pb-4">
   530→        <div className="flex items-start gap-2">
   531→          <span className="text-sm font-semibold text-[#2D8C7A] shrink-0">
   532→            Q{question.questionNumber}.
   533→          </span>
   534→          <Label className="text-sm font-medium leading-snug">
   535→            {question.questionText}
   536→            {question.required && <span className="text-red-500 ml-1">*</span>}
   537→          </Label>
   538→        </div>
   539→
   540→        {/* Single Choice (Radio) */}
   541→        {question.type === 'single_choice' && question.options && (
   542→          <div className="space-y-2 pl-2">
   543→            {question.options.map((option) => {
   544→              const isSelected = currentAnswer === option.value;
   545→              const hasTooltip = question.hasTooltip && option.tooltip;
   546→
   547→              return (
   548→                <div
   549→                  key={option.value}
   550→                  className={cn(
   551→                    'flex items-center gap-3 p-2.5 rounded-lg border cursor-pointer transition-all duration-200',
   552→                    isSelected
   553→                      ? 'border-[#2D8C7A] bg-[#2D8C7A]/5 shadow-sm'
   554→                      : 'border-gray-200 hover:border-gray-300 hover:bg-gray-50'
   555→                  )}
   556→                  onClick={() => handleSingleChoice(question.id, option.value)}
   557→                >
   558→                  <div
   559→                    className={cn(
   560→                      'w-4 h-4 rounded-full border-2 flex items-center justify-center shrink-0 transition-all',
   561→                      isSelected
   562→                        ? 'border-[#2D8C7A] bg-[#2D8C7A]'
   563→                        : 'border-gray-300'
   564→                    )}

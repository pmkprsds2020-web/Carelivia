   209→   209→function formatTime(dateStr: string): string {
   210→   210→  try {
   211→   211→    const d = new Date(dateStr);
   212→   212→    return d.toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit', hour12: false });
   213→   213→  } catch {
   214→   214→    return '';
   215→   215→  }
   216→   216→}
   217→   217→
   218→   218→function formatDate(dateStr: string): string {
   219→   219→  try {
   220→   220→    const d = new Date(dateStr);
   221→   221→    const today = new Date();
   222→   222→    const yesterday = new Date(today);
   223→   223→    yesterday.setDate(yesterday.getDate() - 1);
   224→   224→    if (d.toDateString() === today.toDateString()) return 'Hari Ini';
   225→   225→    if (d.toDateString() === yesterday.toDateString()) return 'Kemarin';
   226→   226→    return d.toLocaleDateString('id-ID', { day: 'numeric', month: 'long', year: 'numeric' });
   227→   227→  } catch {
   228→   228→    return '';
   229→   229→  }
   230→   230→}
   231→   231→
   232→   232→function isDifferentDay(a: string, b: string): boolean {
   233→   233→  return new Date(a).toDateString() !== new Date(b).toDateString();
   234→   234→}

   244→   244→function generateRmNumber(): string {
   245→   245→  const now = new Date();
   246→   246→  const y = now.getFullYear();
   247→   247→  const m = String(now.getMonth() + 1).padStart(2, '0');
   248→   248→  const d = String(now.getDate()).padStart(2, '0');
   249→   249→  const seq = String(Math.floor(Math.random() * 9999) + 1).padStart(4, '0');
   250→   250→  return `RM-${y}${m}${d}-${seq}`;
   251→   251→}

   142→// ── Helper: Format date in Indonesian ────────────────────────────────────────
   143→function formatIndonesianDate(dateStr: string | null | undefined): string {
   144→  if (!dateStr) return '-';
   145→  const bulan = [
   146→    'Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni',
   147→    'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember',
   148→  ];
   149→  const d = new Date(dateStr);
   150→  if (isNaN(d.getTime())) return dateStr;
   151→  return `${d.getDate()} ${bulan[d.getMonth()]} ${d.getFullYear()}`;
   152→}

   154→// ── Helper: Safe string ─────────────────────────────────────────────────────
   155→function s(val: unknown): string {
   156→  if (val === null || val === undefined || val === '') return '-';
   157→  return String(val);
   158→}

   160→// ── Helper: Draw double horizontal line ──────────────────────────────────────
   161→function drawDoubleLine(pdf: jsPDF, x1: number, x2: number, y: number): void {
   162→  pdf.setLineWidth(0.5);
   163→  pdf.line(x1, y, x2, y);
   164→  pdf.setLineWidth(0.2);
   165→  pdf.line(x1, y + 1.2, x2, y + 1.2);
   166→  pdf.setLineWidth(0.2);
   167→}

   169→// ── Helper: Add page number footer ──────────────────────────────────────────
   170→function addPageNumber(pdf: jsPDF, pageNum: number): void {
   171→  const pageWidth = pdf.internal.pageSize.getWidth();
   172→  const pageHeight = pdf.internal.pageSize.getHeight();
   173→  pdf.setFont('times', 'normal');
   174→  pdf.setFontSize(9);
   175→  pdf.setTextColor(100, 100, 100);
   176→  pdf.text(`Halaman ${pageNum}`, pageWidth / 2, pageHeight - 8, { align: 'center' });
   177→  pdf.setTextColor(0, 0, 0);
   178→}

   180→// ── Helper: Check page break needed ─────────────────────────────────────────
   181→function checkPageBreak(
   182→  ctx: PdfContext,
   183→  needed: number,
   184→): PdfContext {
   185→  if (ctx.y + needed > ctx.pageHeight - ctx.margin - 15) {
   186→    ctx.pdf.addPage();
   187→    ctx.pageNum++;
   188→    addPageNumber(ctx.pdf, ctx.pageNum);
   189→    ctx.y = ctx.margin;
   190→  }
   191→  return ctx;
   192→}

   194→// ── Helper: Draw section header bar ─────────────────────────────────────────
   195→function drawSectionHeader(ctx: PdfContext, title: string): PdfContext {
   196→  ctx.y += 3;
   197→  const pb = checkPageBreak(ctx, 12);
   198→  ctx.y = pb.y;
   199→  ctx.pageNum = pb.pageNum;
   200→
   201→  // Background bar
   202→  ctx.pdf.setFillColor(230, 237, 244);
   203→  ctx.pdf.rect(ctx.margin, ctx.y - 4, ctx.contentWidth, 7, 'F');
   204→
   205→  // Left accent bar
   206→  ctx.pdf.setFillColor(45, 140, 122);
   207→  ctx.pdf.rect(ctx.margin, ctx.y - 4, 2, 7, 'F');
   208→
   209→  // Title text
   210→  ctx.pdf.setFont('times', 'bold');
   211→  ctx.pdf.setFontSize(11);
   212→  ctx.pdf.setTextColor(20, 60, 80);
   213→  ctx.pdf.text(title, ctx.margin + 5, ctx.y);
   214→  ctx.pdf.setTextColor(0, 0, 0);
   215→
   216→  ctx.y += 6;
   217→  return ctx;
   218→}

   236→// ── Helper: Draw two-column key-value table (Data Pasien style) ────────────
   237→function drawKeyValueTable(
   238→  ctx: PdfContext,
   239→  rows: Array<[string, string]>,
   240→  colWidths?: [number, number],
   241→): PdfContext {
   242→  const labelW = colWidths ? colWidths[0] : 50;
   243→  const valueW = colWidths ? colWidths[1] : ctx.contentWidth - labelW;
   244→  const rowH = 6;
   245→  const halfW = ctx.contentWidth / 2;
   246→
   247→  for (let i = 0; i < rows.length; i += 2) {
   248→    const pb = checkPageBreak(ctx, rowH + 2);
   249→    ctx.y = pb.y;
   250→    ctx.pageNum = pb.pageNum;
   251→
   252→    // Left column
   253→    const [label1, value1] = rows[i];
   254→    ctx.pdf.setFillColor(248, 250, 252);
   255→    ctx.pdf.rect(ctx.margin, ctx.y - 4, halfW, rowH, 'F');
   256→    ctx.pdf.setDrawColor(200, 210, 220);
   257→    ctx.pdf.rect(ctx.margin, ctx.y - 4, halfW, rowH);
   258→
   259→    ctx.pdf.setFont('times', 'bold');
   260→    ctx.pdf.setFontSize(9);
   261→    ctx.pdf.text(label1, ctx.margin + 2, ctx.y);
   262→    ctx.pdf.setFont('times', 'normal');
   263→    const v1Lines = ctx.pdf.splitTextToSize(value1, halfW - labelW - 4);
   264→    ctx.pdf.text(v1Lines, ctx.margin + labelW, ctx.y);
   265→
   266→    // Right column (if exists)
   267→    if (i + 1 < rows.length) {
   268→      const [label2, value2] = rows[i + 1];
   269→      const x2 = ctx.margin + halfW;
   270→      ctx.pdf.setFillColor(248, 250, 252);
   271→      ctx.pdf.rect(x2, ctx.y - 4, halfW, rowH, 'F');
   272→      ctx.pdf.setDrawColor(200, 210, 220);
   273→      ctx.pdf.rect(x2, ctx.y - 4, halfW, rowH);
   274→
   275→      ctx.pdf.setFont('times', 'bold');
   276→      ctx.pdf.text(label2, x2 + 2, ctx.y);
   277→      ctx.pdf.setFont('times', 'normal');
   278→      const v2Lines = ctx.pdf.splitTextToSize(value2, halfW - labelW - 4);
   279→      ctx.pdf.text(v2Lines, x2 + labelW, ctx.y);
   280→    }
   281→
   282→    ctx.y += rowH;
   283→  }
   284→  ctx.y += 2;
   285→  return ctx;
   286→}

   288→// ── Helper: Draw grid table with header row ─────────────────────────────────
   289→function drawGridTable(
   290→  ctx: PdfContext,
   291→  headers: string[],
   292→  rows: string[][],
   293→  colWidths?: number[],
   294→): PdfContext {
   295→  if (!colWidths) {
   296→    const w = ctx.contentWidth / headers.length;
   297→    colWidths = headers.map(() => w);
   298→  }
   299→
   300→  const rowH = 6;
   301→
   302→  // Header row
   303→  const pbH = checkPageBreak(ctx, rowH + 2);
   304→  ctx.y = pbH.y;
   305→  ctx.pageNum = pbH.pageNum;
   306→
   307→  ctx.pdf.setFillColor(45, 140, 122);
   308→  ctx.pdf.rect(ctx.margin, ctx.y - 4, ctx.contentWidth, rowH, 'F');
   309→
   310→  ctx.pdf.setFont('times', 'bold');
   311→  ctx.pdf.setFontSize(8.5);
   312→  ctx.pdf.setTextColor(255, 255, 255);
   313→  let xOff = ctx.margin;
   314→  for (let c = 0; c < headers.length; c++) {
   315→    ctx.pdf.text(headers[c], xOff + 2, ctx.y - 0.5, { maxWidth: colWidths[c] - 4 });
   316→    xOff += colWidths[c];
   317→  }
   318→  ctx.pdf.setTextColor(0, 0, 0);
   319→  ctx.y += rowH;
   320→
   321→  // Data rows
   322→  for (let r = 0; r < rows.length; r++) {
   323→    const pbR = checkPageBreak(ctx, rowH + 2);
   324→    ctx.y = pbR.y;
   325→    ctx.pageNum = pbR.pageNum;
   326→
   327→    // Alternate row color
   328→    if (r % 2 === 0) {
   329→      ctx.pdf.setFillColor(248, 252, 250);
   330→    } else {
   331→      ctx.pdf.setFillColor(255, 255, 255);
   332→    }
   333→    ctx.pdf.rect(ctx.margin, ctx.y - 4, ctx.contentWidth, rowH, 'F');
   334→
   335→    // Grid lines
   336→    ctx.pdf.setDrawColor(200, 210, 220);
   337→    ctx.pdf.rect(ctx.margin, ctx.y - 4, ctx.contentWidth, rowH);
   338→
   339→    ctx.pdf.setFont('times', 'normal');
   340→    ctx.pdf.setFontSize(8.5);
   341→    xOff = ctx.margin;
   342→    for (let c = 0; c < rows[r].length && c < headers.length; c++) {
   343→      const cellText = s(rows[r][c]);
   344→      const lines = ctx.pdf.splitTextToSize(cellText, colWidths[c] - 4);
   345→      ctx.pdf.text(lines[0] || '', xOff + 2, ctx.y - 0.5);
   346→      xOff += colWidths[c];
   347→    }
   348→    ctx.y += rowH;
   349→  }
   350→  ctx.y += 3;
   351→  return ctx;
   352→}

   354→// ── Helper: Draw wrapped paragraph text ─────────────────────────────────────
   355→function drawParagraph(ctx: PdfContext, text: string, indent: number = 2): PdfContext {
   356→  if (!text || text.trim() === '' || text === '-') return ctx;
   357→
   358→  ctx.pdf.setFont('times', 'normal');
   359→  ctx.pdf.setFontSize(9.5);
   360→  const lines = ctx.pdf.splitTextToSize(text, ctx.contentWidth - indent * 2);
   361→
   362→  for (const line of lines) {
   363→    const pb = checkPageBreak(ctx, 5);
   364→    ctx.y = pb.y;
   365→    ctx.pageNum = pb.pageNum;
   366→    ctx.pdf.text(line, ctx.margin + indent, ctx.y);
   367→    ctx.y += 4.5;
   368→  }
   369→  return ctx;
   370→}

   372→// ── Helper: Draw a labeled text line ────────────────────────────────────────
   373→function drawLabelValue(ctx: PdfContext, label: string, value: string): PdfContext {
   374→  const pb = checkPageBreak(ctx, 8);
   375→  ctx.y = pb.y;
   376→  ctx.pageNum = pb.pageNum;
   377→
   378→  ctx.pdf.setFont('times', 'bold');
   379→  ctx.pdf.setFontSize(9.5);
   380→  ctx.pdf.text(label, ctx.margin + 2, ctx.y);
   381→  ctx.pdf.setFont('times', 'normal');
   382→  const lw = ctx.pdf.getTextWidth(label) + 2;
   383→  const lines = ctx.pdf.splitTextToSize(value, ctx.contentWidth - lw - 4);
   384→  ctx.pdf.text(lines, ctx.margin + 2 + lw, ctx.y);
   385→  ctx.y += Math.max(lines.length * 4.5, 5);
   386→  return ctx;
   387→}

   389→// ── Helper: Draw badge ──────────────────────────────────────────────────────
   390→function drawBadge(ctx: PdfContext, text: string, color: 'red' | 'green' | 'yellow' | 'gray'): PdfContext {
   391→  const tw = ctx.pdf.getTextWidth(text) + 6;
   392→  const bh = 5;
   393→
   394→  const colors = {
   395→    red: { fill: [254, 226, 226], text: [153, 27, 27] },
   396→    green: { fill: [220, 252, 231], text: [22, 101, 52] },
   397→    yellow: { fill: [254, 249, 195], text: [113, 63, 18] },
   398→    gray: { fill: [243, 244, 246], text: [75, 85, 99] },
   399→  };
   400→  const c = colors[color];
   401→
   402→  ctx.pdf.setFillColor(c.fill[0], c.fill[1], c.fill[2]);
   403→  ctx.pdf.roundedRect(ctx.margin + 2, ctx.y - 3.5, tw, bh, 1.5, 1.5, 'F');
   404→  ctx.pdf.setFont('times', 'bold');
   405→  ctx.pdf.setFontSize(8);
   406→  ctx.pdf.setTextColor(c.text[0], c.text[1], c.text[2]);
   407→  ctx.pdf.text(text, ctx.margin + 5, ctx.y);
   408→  ctx.pdf.setTextColor(0, 0, 0);
   409→
   410→  // Advance x position for next badge (we'll just move y for simplicity)
   411→  ctx.y += bh + 1;
   412→  return ctx;
   413→}

   415→// ── Render structured resume PDF ─────────────────────────────────────────────
   416→function renderStructuredResume(
   417→  ctx: PdfContext,
   418→  resumeData: ResumeStructuredData,
   419→  patientData: PatientData,
   420→  documentData: DocumentData,
   421→  facilityData: FacilityData | undefined,
   422→): PdfContext {
   423→  const { pdf } = ctx;
   424→
   425→  // ══════════════════════════════════════════════════════════════════════
   426→  // 1. HEADER SECTION
   427→  // ══════════════════════════════════════════════════════════════════════
   428→  const facilityName = facilityData?.name || 'FASILITAS KESEHATAN PRIMER';
   429→  const facilityAddress = facilityData?.address || 'Jl. Kesehatan No. 1, Kota, Provinsi';
   430→  const facilityPhone = facilityData?.phone || '(021) 123-4567';
   431→
   432→  // Logo placeholder
   433→  pdf.setDrawColor(150, 150, 150);
   434→  pdf.setFillColor(240, 240, 240);
   435→  pdf.roundedRect(ctx.margin, ctx.y, 22, 22, 2, 2, 'FD');
   436→  pdf.setFont('times', 'italic');
   437→  pdf.setFontSize(7);
   438→  pdf.setTextColor(120, 120, 120);
   439→  pdf.text('LOGO', ctx.margin + 11, ctx.y + 11, { align: 'center' });
   440→  pdf.text('FASILITAS', ctx.margin + 11, ctx.y + 14.5, { align: 'center' });
   441→  pdf.setTextColor(0, 0, 0);
   442→
   443→  // Facility info
   444→  pdf.setFont('times', 'bold');
   445→  pdf.setFontSize(14);
   446→  pdf.text(facilityName, ctx.margin + 26, ctx.y + 6);
   447→
   448→  pdf.setFont('times', 'normal');
   449→  pdf.setFontSize(9);
   450→  pdf.text(facilityAddress, ctx.margin + 26, ctx.y + 11);
   451→  pdf.text(`Telp: ${facilityPhone}`, ctx.margin + 26, ctx.y + 15.5);
   452→
   453→  ctx.y += 26;
   454→  drawDoubleLine(pdf, ctx.margin, ctx.pageWidth - ctx.margin, ctx.y);
   455→  ctx.y += 5;
   456→
   457→  // Document info row
   458→  pdf.setFont('times', 'normal');
   459→  pdf.setFontSize(9);
   460→  pdf.setTextColor(80, 80, 80);
   461→  pdf.text(`No. Dokumen: ${documentData.documentNumber || '-'}`, ctx.margin, ctx.y);
   462→  pdf.text(`Tanggal: ${formatIndonesianDate(documentData.generatedAt)}`, ctx.margin + 65, ctx.y);
   463→  if (documentData.version) {
   464→    pdf.text(`Versi: ${documentData.version}`, ctx.pageWidth - ctx.margin, ctx.y, { align: 'right' });
   465→  }
   466→  pdf.setTextColor(0, 0, 0);
   467→  ctx.y += 6;
   468→
   469→  // Thin separator
   470→  pdf.setDrawColor(180, 180, 180);
   471→  pdf.setLineWidth(0.1);
   472→  pdf.line(ctx.margin, ctx.y, ctx.pageWidth - ctx.margin, ctx.y);
   473→  pdf.setLineWidth(0.2);
   474→  ctx.y += 5;
   475→
   476→  // ══════════════════════════════════════════════════════════════════════
   477→  // 2. TITLE
   478→  // ══════════════════════════════════════════════════════════════════════
   479→  const title = 'RESUME MEDIS TELEPALIATIF';
   480→  pdf.setFont('times', 'bold');
   481→  pdf.setFontSize(16);
   482→  pdf.text(title, ctx.pageWidth / 2, ctx.y, { align: 'center' });
   483→  ctx.y += 3;
   484→
   485→  const titleWidth = pdf.getTextWidth(title);
   486→  pdf.setLineWidth(0.4);
   487→  pdf.line(
   488→    ctx.pageWidth / 2 - titleWidth / 2,
   489→    ctx.y,
   490→    ctx.pageWidth / 2 + titleWidth / 2,
   491→    ctx.y,
   492→  );
   493→  pdf.setLineWidth(0.2);
   494→  ctx.y += 7;
   495→
   496→  // ══════════════════════════════════════════════════════════════════════
   497→  // 3. DATA PASIEN
   498→  // ══════════════════════════════════════════════════════════════════════
   499→  ctx = drawSectionHeader(ctx, 'DATA PASIEN');
   500→  const dp = resumeData.dataPasien;
   501→  const kontak = dp.kontakKeluarga;
   502→  const kontakStr = kontak
   503→    ? `${s(kontak.nama)}${kontak.hubungan ? ` (${kontak.hubungan})` : ''}${kontak.telepon ? ` • ${kontak.telepon}` : ''}`
   504→    : '-';
   505→  const dpjpStr = dp.dpjp
   506→    ? `${dp.dpjp}${dp.dpjpSpesialisasi ? ` (${dp.dpjpSpesialisasi})` : ''}${dp.dpjpSIP ? ` - SIP: ${dp.dpjpSIP}` : ''}`
   507→    : '-';
   508→
   509→  const patientRows: Array<[string, string]> = [
   510→    ['Nama', s(dp.nama || patientData.patientName)],
   511→    ['No. RM', s(dp.noRM || patientData.rmNumber)],
   512→    ['NIK', s(dp.nik || patientData.nik)],
   513→    ['No. BPJS', s(dp.noBPJS || patientData.bpjsNumber)],
   514→    ['Tanggal Lahir', formatIndonesianDate(dp.tanggalLahir)],
   515→    ['Jenis Kelamin', s(dp.jenisKelamin)],
   516→    ['Umur', s(dp.umur)],
   517→    ['No. Telepon', s(dp.noTelepon)],
   518→    ['Alamat', s(dp.alamat)],
   519→    ['Diagnosa Utama', s(dp.diagnosaUtama || patientData.primaryDiagnosis)],

  1492→      ? line.trim().replace(/^\*\*/, '').replace(/\*\*$/, '')

  1511→      ctx.y += 5;
  1512→    }
  1513→  }
  1514→
  1515→  return ctx;

  1519→export async function POST(req: NextRequest) {
  1520→  try {
  1521→    const body: PdfRequestBody = await req.json();
  1522→    const { documentType, documentData, patientData, facilityData, resumeData } = body;
  1523→
  1524→    // Validate required fields
  1525→    if (!documentType || !['resume', 'referral'].includes(documentType)) {

  1527→        { error: 'documentType must be "resume" or "referral"' },
  1528→        { status: 400 },
  1529→      );
  1530→    }
  1531→
  1532→    // fullContent is required only if resumeData is not provided for resume type
  1533→    if (!documentData?.fullContent && !(documentType === 'resume' && resumeData)) {
  1534→      return NextResponse.json(
  1535→        { error: 'documentData.fullContent or resumeData is required' },
  1536→        { status: 400 },
  1537→      );
  1538→    }
  1539→
  1540→    // ── Create PDF document ──────────────────────────────────────────────
  1541→    const pdf = new jsPDF('p', 'mm', 'a4');
  1542→    const pageWidth = pdf.internal.pageSize.getWidth();
  1543→    const pageHeight = pdf.internal.pageSize.getHeight();
  1544→    const margin = 20;
  1545→    const contentWidth = pageWidth - 2 * margin;
  1546→
  1547→    let ctx: PdfContext = {
  1548→      pdf,
  1549→      y: margin,

  1556→
  1557→    // Add page number to first page

  1559→

  1576→    ctx.y += 8;
  1577→

  1579→    ctx = checkPageBreak(ctx, 55);

  1581→    ctx.pageNum = ctx.pageNum;

  1591→    pdf.text('Dokter Penanggung Jawab:', margin, ctx.y);

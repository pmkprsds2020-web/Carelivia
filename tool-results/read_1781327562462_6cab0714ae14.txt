   209→function formatTime(dateStr: string): string {
   210→  try {
   211→    const d = new Date(dateStr);
   212→    return d.toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit', hour12: false });
   213→  } catch {
   214→    return '';
   215→  }
   216→}
   217→
   218→function formatDate(dateStr: string): string {
   219→  try {
   220→    const d = new Date(dateStr);
   221→    const today = new Date();
   222→    const yesterday = new Date(today);
   223→    yesterday.setDate(yesterday.getDate() - 1);
   224→    if (d.toDateString() === today.toDateString()) return 'Hari Ini';
   225→    if (d.toDateString() === yesterday.toDateString()) return 'Kemarin';
   226→    return d.toLocaleDateString('id-ID', { day: 'numeric', month: 'long', year: 'numeric' });
   227→  } catch {
   228→    return '';
   229→  }
   230→}
   231→
   232→function isDifferentDay(a: string, b: string): boolean {
   233→  return new Date(a).toDateString() !== new Date(b).toDateString();
   234→}
   235→
   236→function formatFee(fee: number): string {
   237→  return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(fee);
   238→}
   239→
   240→function generateId(): string {
   241→  return `${Date.now()}-${Math.random().toString(36).substring(2, 9)}`;
   242→}
   243→
   244→function generateRmNumber(): string {
   245→  const now = new Date();
   246→  const y = now.getFullYear();
   247→  const m = String(now.getMonth() + 1).padStart(2, '0');
   248→  const d = String(now.getDate()).padStart(2, '0');
   249→  const seq = String(Math.floor(Math.random() * 9999) + 1).padStart(4, '0');
   250→  return `RM-${y}${m}${d}-${seq}`;
   251→}

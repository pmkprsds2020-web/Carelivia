   154→// ── Helper: Compute vital sign criticality score ─────────────────────────────
   155→function computeCriticalityScore(v: VitalSignRecord): { score: number; reasons: string[] } {
   156→  let score = 0;
   157→  const reasons: string[] = [];
   158→
   159→  if (v.oxygenSat != null && v.oxygenSat < 90) {
   160→    score += 4;
   161→    reasons.push(`SpO2 ${v.oxygenSat}% (<90%)`);
   162→  }
   163→  if (v.heartRate != null) {
   164→    if (v.heartRate > 120) { score += 3; reasons.push(`Nadi ${v.heartRate} bpm (>120)`); }
   165→    else if (v.heartRate < 50) { score += 3; reasons.push(`Nadi ${v.heartRate} bpm (<50)`); }
   166→  }
   167→  if (v.respiratoryRate != null) {
   168→    if (v.respiratoryRate > 28) { score += 3; reasons.push(`RR ${v.respiratoryRate}/menit (>28)`); }
   169→    else if (v.respiratoryRate < 8) { score += 3; reasons.push(`RR ${v.respiratoryRate}/menit (<8)`); }
   170→  }
   171→  if (v.systolicBP != null) {
   172→    if (v.systolicBP > 180) { score += 3; reasons.push(`SBP ${v.systolicBP} mmHg (>180)`); }
   173→    else if (v.systolicBP < 90) { score += 3; reasons.push(`SBP ${v.systolicBP} mmHg (<90)`); }
   174→  }
   175→  if (v.temperature != null) {
   176→    if (v.temperature > 39) { score += 2; reasons.push(`Suhu ${v.temperature}°C (>39)`); }
   177→    else if (v.temperature < 35) { score += 2; reasons.push(`Suhu ${v.temperature}°C (<35)`); }
   178→  }
   179→
   180→  return { score, reasons };
   181→}

   183→// ── Helper: Categorize medication ────────────────────────────────────────────
   184→function categorizeMedication(med: MedicationWithAdherences): string {
   185→  const name = med.medicineName.toLowerCase();
   186→  const indication = (med.indication || '').toLowerCase();
   187→
   188→  // Analgesik
   189→  const analgesikKeywords = ['nyeri', 'pain', 'morphine', 'fentanyl', 'tramadol', 'ketorolac', 'paracetamol', 'ibuprofen', 'diclofenac', 'morfine', 'kodein', 'gabapentin', 'pregabalin'];
   190→  if (analgesikKeywords.some(k => name.includes(k) || indication.includes(k))) return 'analgesik';
   191→
   192→  // Antiemetik
   193→  const antiemetikKeywords = ['mual', 'nausea', 'vomiting', 'muntah', 'ondansetron', 'metoclopramide', 'domperidone', 'granisetron'];
   194→  if (antiemetikKeywords.some(k => name.includes(k) || indication.includes(k))) return 'antiemetik';
   195→
   196→  // Laksatif
   197→  const laksatifKeywords = ['konstipasi', 'constipation', 'sembelit', 'bisacodyl', 'lactulose', 'senna', 'peg', 'movicol'];
   198→  if (laksatifKeywords.some(k => name.includes(k) || indication.includes(k))) return 'laksatif';
   199→
   200→  // Antidepresan
   201→  const antidepresanKeywords = ['depresi', 'depression', 'sertraline', 'fluoxetine', 'mirtazapine', 'citalopram', 'escitalopram', 'venlafaxine', 'amitriptyline'];
   202→  if (antidepresanKeywords.some(k => name.includes(k) || indication.includes(k))) return 'antidepresan';
   203→
   204→  // Ansiolitik
   205→  const ansiolitikKeywords = ['cemas', 'anxiety', 'diazepam', 'lorazepam', 'alprazolam', 'clonazepam', 'midazolam'];
   206→  if (ansiolitikKeywords.some(k => name.includes(k) || indication.includes(k))) return 'ansiolitik';
   207→
   208→  return 'lainnya';
   209→}

   228→// ── Helper: Format TTV for narrative ─────────────────────────────────────────
   229→function formatTTVNarrative(v: VitalSignRecord, label: string): string {
   230→  const tgl = new Date(v.recordedAt).toLocaleString('id-ID');
   231→  const parts = [
   232→    `${label} (${tgl}):`,
   233→    `  TD: ${fmtVal(v.systolicBP, '')}/${fmtVal(v.diastolicBP, ' mmHg')}`,
   234→    `  Nadi: ${fmtVal(v.heartRate, ' bpm')}`,
   235→    `  RR: ${fmtVal(v.respiratoryRate, '/menit')}`,
   236→    `  Suhu: ${fmtVal(v.temperature, '°C')}`,
   237→    `  SpO2: ${fmtVal(v.oxygenSat, '%')}`,
   238→  ];
   239→  if (v.weight != null) parts.push(`  BB: ${v.weight} kg`);
   240→  if (v.bmi != null) parts.push(`  BMI: ${v.bmi}`);
   241→  if (v.notes) parts.push(`  Catatan: ${v.notes}`);
   242→  return parts.join('\n');
   243→}

   245→// ── Helper: Format complaint for narrative ───────────────────────────────────
   246→function formatComplaintNarrative(c: DailyComplaintRecord, label: string): string {
   247→  const tgl = new Date(c.submittedAt).toLocaleString('id-ID');
   248→  const parts = [`${label} (${tgl}):`];
   249→  parts.push(`  Kondisi: ${c.kondisiHariIni === 'baik' ? 'Baik' : 'Tidak Baik'}`);
   250→  if (c.alasanKondisi) parts.push(`  Alasan: ${c.alasanKondisi}`);
   251→  if (c.keluhanBaru === 'ada' && c.deskripsiKeluhanBaru) {
   252→    parts.push(`  Keluhan Baru: ${c.deskripsiKeluhanBaru}`);
   253→  }
   254→  parts.push(`  Nyeri: ${c.kondisiNyeri}`);
   255→  parts.push(`  Sesak: ${c.kondisiSesak}`);
   256→  parts.push(`  Makan/Minum: ${c.makanMinum === 'ya' ? 'Ya' : 'Tidak'}`);
   257→  if (c.makanMinum === 'tidak' && c.alasanMakanMinum) parts.push(`  Alasan: ${c.alasanMakanMinum}`);
   258→  parts.push(`  Tidur: ${c.tidur === 'ya' ? 'Ya' : 'Tidak'}`);
   259→  if (c.tidur === 'tidak' && c.alasanTidur) parts.push(`  Alasan: ${c.alasanTidur}`);
   260→  if (c.masalahObat === 'ya' && c.deskripsiMasalahObat) {
   261→    parts.push(`  Masalah Obat: ${c.deskripsiMasalahObat}`);
   262→  }
   263→  parts.push(`  Tingkat Keparahan: ${c.severityLevel}`);
   264→  return parts.join('\n');
   265→}

   642→    });

   645→
   646→  return lines.join('\n');
   647→}

   649→// ── Helper: Parse JSON from LLM response ─────────────────────────────────────
   650→function parseLLMJson(response: string): AIAnalysis | null {
   651→  // Try direct JSON parse
   652→  try {
   653→    const parsed = JSON.parse(response);
   654→    if (parsed && typeof parsed === 'object') return parsed as AIAnalysis;
   655→  } catch {
   656→    // Not direct JSON, continue
   657→  }
   658→
   659→  // Try to extract JSON from markdown code blocks
   660→  const jsonBlockMatch = response.match(/```(?:json)?\s*\n?([\s\S]*?)\n?\s*```/);
   661→  if (jsonBlockMatch) {
   662→    try {
   663→      const parsed = JSON.parse(jsonBlockMatch[1].trim());
   664→      if (parsed && typeof parsed === 'object') return parsed as AIAnalysis;
   665→    } catch {
   666→      // Failed to parse code block JSON
   667→    }
   668→  }
   669→
   670→  // Try to find JSON object in the response
   671→  const jsonMatch = response.match(/\{[\s\S]*\}/);
   672→  if (jsonMatch) {
   673→    try {
   674→      const parsed = JSON.parse(jsonMatch[0]);
   675→      if (parsed && typeof parsed === 'object') return parsed as AIAnalysis;
   676→    } catch {
   677→      // Failed to parse extracted JSON
   678→    }
   679→  }
   680→
   681→  return null;
   682→}

   914→      : 'Tidak Ada Data',

   920→      ? (acpDocuments[0].careGoal || 'Tidak ditentukan')

   924→      : patient.riskLevel === 'kuning'

   991→async function generateDocumentNumber(): Promise<string> {
   992→  const now = new Date();
   993→  const year = now.getFullYear();
   994→  const month = String(now.getMonth() + 1).padStart(2, '0');
   995→
   996→  // Count existing resumes this month to generate sequence
   997→  const startOfMonth = new Date(year, now.getMonth(), 1);
   998→  const startOfNextMonth = new Date(year, now.getMonth() + 1, 1);
   999→
  1000→  const count = await db.palliativeResume.count({
  1001→    where: {
  1002→      createdAt: {
  1003→        gte: startOfMonth,
  1004→        lt: startOfNextMonth,
  1005→      },
  1006→    },
  1007→  });
  1008→
  1009→  const sequence = String(count + 1).padStart(4, '0');
  1010→  return `RPM/${year}/${month}/${sequence}`;

  1188→      return NextResponse.json({ error: 'Patient ID required' }, { status: 400 });

  1190→
  1191→    const additionalData: AdditionalData = rawAdditionalData || {};
  1192→
  1193→    // ── 1. Fetch ALL patient data from DB ────────────────────────────────────

  1195→      where: { id: palliativePatientId },
  1196→      include: {
  1197→        vitalSigns: { orderBy: { recordedAt: 'asc' } }, // ALL, ordered ASC for serial analysis
  1198→        medications: {
  1199→          include: { adherences: { orderBy: { date: 'desc' } } }, // ALL medications (active + inactive)
  1200→        },
  1201→        adherences: { orderBy: { date: 'desc' } },

  1203→          include: { revisions: true }, // ALL ACP documents (including inactive)
  1204→        },
  1205→        screeningRecords: { orderBy: { performedAt: 'asc' } }, // ALL, ordered ASC for trend analysis
  1206→        dailyComplaints: { orderBy: { submittedAt: 'asc' } }, // ALL, ordered ASC
  1207→      },
  1208→    });
  1209→
  1210→    if (!patient) {
  1211→      return NextResponse.json({ error: 'Patient not found' }, { status: 404 });
  1212→    }
  1213→
  1214→    // ── Fetch latest resume for versioning (separate query to avoid include issues) ──
  1215→    const previousResume = await db.palliativeResume.findFirst({
  1216→      where: { palliativePatientId: patient.id },

  1218→    });
  1219→
  1220→    // ── 2. Fetch User for demographics ───────────────────────────────────────
  1221→    const user = await db.user.findUnique({ where: { id: patient.patientId } });

  1224→    let doctorName: string | null = null;

  1239→
  1240→    // ── 4. TTV Serial Analysis ───────────────────────────────────────────────
  1241→    const vitalSigns = patient.vitalSigns as unknown as VitalSignRecord[];

  1243→    let ttvKritis: VitalSignRecord | null = null;
  1244→    let ttvKritisReasons: string[] = [];
  1245→    let ttvTerakhir: VitalSignRecord | null = null;
  1246→

  1248→      // TTV Awal: first (earliest) record
  1249→      ttvAwal = vitalSigns[0];
  1250→
  1251→      // TTV Terakhir: most recent record
  1252→      ttvTerakhir = vitalSigns[vitalSigns.length - 1];
  1253→
  1254→      // TTV Kritis: find the worst vital sign record using composite scoring
  1255→      let worstScore = 0;
  1256→      let worstRecord: VitalSignRecord | null = null;
  1257→      let worstReasons: string[] = [];
  1258→
  1259→      for (const v of vitalSigns) {

  1262→          worstScore = score;
  1263→          worstRecord = v;
  1264→          worstReasons = reasons;
  1265→        }
  1266→      }
  1267→
  1268→      if (worstRecord && worstScore > 0) {
  1269→        ttvKritis = worstRecord;
  1270→        ttvKritisReasons = worstReasons;
  1271→      }
  1272→    }
  1273→
  1274→    const ttvSerial = { ttvAwal, ttvKritis, ttvKritisReasons, ttvTerakhir };

  1286→

  1288→      keluhanTerakhir = dailyComplaints[dailyComplaints.length - 1];
  1289→
  1290→      // Keluhan Terberat: highest severity, or most symptoms
  1291→      let bestSeverity = -1;
  1292→      let bestSymptomCount = 0;
  1293→
  1294→      for (const c of dailyComplaints) {
  1295→        const severityRank = SEVERITY_RANK[c.severityLevel] ?? 0;
  1296→
  1297→        // Count symptoms
  1298→        let symptomCount = 0;
  1299→        if (c.kondisiHariIni === 'tidak_baik') symptomCount++;
  1300→        if (c.keluhanBaru === 'ada') symptomCount++;
  1301→        if (c.kondisiNyeri === 'bertambah') symptomCount++;
  1302→        if (c.kondisiSesak === 'bertambah') symptomCount++;
  1303→        if (c.makanMinum === 'tidak') symptomCount++;
  1304→        if (c.tidur === 'tidak') symptomCount++;
  1305→        if (c.masalahObat === 'ya') symptomCount++;
  1306→
  1307→        // Compile symptom frequencies
  1308→        if (c.kondisiNyeri !== 'tidak_nyeri') {
  1309→          symptomFrequencies['Nyeri'] = (symptomFrequencies['Nyeri'] || 0) + 1;
  1310→        }
  1311→        if (c.kondisiSesak !== 'tidak_sesak') {
  1312→          symptomFrequencies['Sesak'] = (symptomFrequencies['Sesak'] || 0) + 1;
  1313→        }
  1314→        if (c.makanMinum === 'tidak') {
  1315→          symptomFrequencies['Gangguan Asupan'] = (symptomFrequencies['Gangguan Asupan'] || 0) + 1;
  1316→        }
  1317→        if (c.tidur === 'tidak') {
  1318→          symptomFrequencies['Gangguan Tidur'] = (symptomFrequencies['Gangguan Tidur'] || 0) + 1;
  1319→        }
  1320→        if (c.keluhanBaru === 'ada' && c.deskripsiKeluhanBaru) {
  1321→          // Try to extract individual complaints
  1322→          const symptoms = c.deskripsiKeluhanBaru.split(/[,;]/).map(s => s.trim()).filter(Boolean);
  1323→          for (const s of symptoms) {
  1324→            symptomFrequencies[s] = (symptomFrequencies[s] || 0) + 1;
  1325→          }
  1326→        }
  1327→        if (c.masalahObat === 'ya') {
  1328→          symptomFrequencies['Masalah Obat'] = (symptomFrequencies['Masalah Obat'] || 0) + 1;
  1329→        }
  1330→
  1331→        // Determine worst complaint
  1332→        if (severityRank > bestSeverity || (severityRank === bestSeverity && symptomCount > bestSymptomCount)) {
  1333→          bestSeverity = severityRank;
  1334→          bestSymptomCount = symptomCount;
  1335→          keluhanTerberat = c;
  1336→        }
  1337→      }
  1338→    }
  1339→
  1340→    const keluhanHarian = { keluhanAwal, keluhanTerberat, keluhanTerakhir, symptomFrequencies };
  1341→

  1347→      .filter(s => s.screeningType === 'esas')

  1349→
  1350→    let esasSkorAwal: Record<string, unknown> | null = null;
  1351→    let esasSkorTertinggi: Record<string, unknown> | null = null;
  1352→    let esasSkorTerakhir: Record<string, unknown> | null = null;
  1353→

  1355→      // Skor Awal
  1356→      const first = esasRecords[0];
  1357→      try {

  1359→      } catch {
  1360→        esasSkorAwal = { score: first.score, performedAt: first.performedAt, scoreLabel: first.scoreLabel };
  1361→      }
  1362→
  1363→      // Skor Terakhir
  1364→      const last = esasRecords[esasRecords.length - 1];
  1365→      try {
  1366→        esasSkorTerakhir = last.details ? { ...JSON.parse(last.details), score: last.score, performedAt: last.performedAt } : { score: last.score, performedAt: last.performedAt, scoreLabel: last.scoreLabel };
  1367→      } catch {
  1368→        esasSkorTerakhir = { score: last.score, performedAt: last.performedAt, scoreLabel: last.scoreLabel };
  1369→      }
  1370→
  1371→      // Skor Tertinggi (highest total score)
  1372→      let maxScore = -1;
  1373→      let maxRecord: ScreeningRecord | null = null;
  1374→      for (const r of esasRecords) {
  1375→        if (r.score != null && r.score > maxScore) {
  1376→          maxScore = r.score;
  1377→          maxRecord = r;
  1378→        }
  1379→      }
  1380→      if (maxRecord) {
  1381→        try {
  1382→          esasSkorTertinggi = maxRecord.details ? { ...JSON.parse(maxRecord.details), score: maxRecord.score, performedAt: maxRecord.performedAt } : { score: maxRecord.score, performedAt: maxRecord.performedAt, scoreLabel: maxRecord.scoreLabel };
  1383→        } catch {
  1384→          esasSkorTertinggi = { score: maxRecord.score, performedAt: maxRecord.performedAt, scoreLabel: maxRecord.scoreLabel };
  1385→        }
  1386→      }
  1387→    }
  1388→
  1389→    const esasAnalysis = { skorAwal: esasSkorAwal, skorTertinggi: esasSkorTertinggi, skorTerakhir: esasSkorTerakhir };

  1392→    const screeningsByType: Record<string, ScreeningRecord[]> = {};

  1395→      screeningsByType[s.screeningType].push(s);

  1399→    const medications = patient.medications as unknown as MedicationWithAdherences[];
  1400→    const medicationCategories: Record<string, MedicationWithAdherences[]> = {

  1402→      antiemetik: [],
  1403→      laksatif: [],

  1405→      ansiolitik: [],
  1406→      lainnya: [],
  1407→    };
  1408→
  1409→    for (const med of medications) {

  1413→
  1414→    // ── 8. Compile ACP data ──────────────────────────────────────────────────
  1415→    const acpDocuments = patient.acpDocuments as unknown as ACPDocument[];
  1416→
  1417→    // ── 9. Build clinical context for LLM ────────────────────────────────────

  1422→        bpjsNumber: patient.bpjsNumber,
  1423→        nik: patient.nik,
  1424→        primaryDiagnosis: patient.primaryDiagnosis,
  1425→        secondaryDiagnosis: patient.secondaryDiagnosis,

  1436→      },
  1437→      user: user ? { name: user.name, dateOfBirth: user.dateOfBirth, gender: user.gender, nik: user.nik, bpjsNumber: user.bpjsNumber, address: user.address } : null,
  1438→      doctorName,
  1439→      doctorSpecialization,
  1440→      doctorLicense,
  1441→      vitalSigns,
  1442→      medications,
  1443→      screeningRecords,
  1444→      dailyComplaints,

  1468→

  1477→          aiAnalysis = generateFallbackAnalysis({
  1478→            patient: {

  1485→            },
  1486→            user: user ? { name: user.name } : null,
  1487→            vitalSigns,

  1490→            keluhanHarian,

  1501→            secondaryDiagnosis: patient.secondaryDiagnosis,

  1507→          user: user ? { name: user.name } : null,

  1528→        },
  1529→        user: user ? { name: user.name } : null,

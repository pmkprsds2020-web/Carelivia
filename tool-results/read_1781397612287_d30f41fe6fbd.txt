    84→function formatDate(dateStr: string): string {
    85→  try {
    86→    return new Date(dateStr).toLocaleDateString('id-ID', {
    87→      day: 'numeric',
    88→      month: 'long',
    89→      year: 'numeric',
    90→    });
    91→  } catch {
    92→    return dateStr;
    93→  }
    94→}
    95→
    96→function formatDateTime(dateStr: string): string {
    97→  try {
    98→    return new Date(dateStr).toLocaleDateString('id-ID', {
    99→      day: 'numeric',
   100→      month: 'short',
   101→      year: 'numeric',
   102→      hour: '2-digit',
   103→      minute: '2-digit',
   104→    });
   105→  } catch {
   106→    return dateStr;
   107→  }
   108→}
   109→
   110→function genId(prefix: string): string {
   111→  return `${prefix}-${Date.now()}-${Math.random().toString(36).substring(2, 7)}`;
   112→}
   113→
   114→function genDocNumber(prefix: string): string {
   115→  const now = new Date();
   116→  const y = now.getFullYear();
   117→  const m = String(now.getMonth() + 1).padStart(2, '0');
   118→  const seq = String(Math.floor(Math.random() * 9000) + 1000);
   119→  return `${prefix}/${y}/${m}/${seq}`;
   120→}

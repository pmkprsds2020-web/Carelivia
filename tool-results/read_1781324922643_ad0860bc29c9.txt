   239→   239→const calculateSocialScore = (a: SocialAssessmentRecord): number => {
   240→   240→  const fields: string[] = [
   241→   241→    a.housingCondition, a.caregiverAvailability, a.familySupportLevel,
   242→   242→    a.transportDifficulty, a.economicConstraint, a.healthcareAccess,
   243→   243→    a.medicalEquipmentNeed, a.socialAssistanceNeed, a.socialIsolationRisk,

   248→   248→const calculatePriority = (form: ScreeningFormState): SocialScreeningPriority => {
   249→   249→  const highVals = ['berat', 'tidak_tersedia', 'sangat_sulit', 'tidak_layak', 'tinggi', 'tidak_ada'];
   250→   250→  const medVals = ['sedang', 'terbatas', 'sulit', 'kurang_layak', 'lemah'];
   251→   251→  const vals = Object.values(form).map(v => v.value);
   252→   252→  if (vals.some(v => highVals.includes(v))) return 'tinggi';
   253→   253→  if (vals.some(v => medVals.includes(v))) return 'sedang';
   254→   254→  return 'rendah';
   255→   255→};

   257→   257→const generateRecommendations = (form: ScreeningFormState): string[] => {
   258→   258→  const recs: string[] = [];

   264→   264→    recs.push('Konseling keluarga dan edukasi tentang dukungan paliatif, pertimbangkan dukungan komunitas');
   265→   265→  if (form.transport.value === 'berat' || form.transport.value === 'sedang')
   266→   266→    recs.push('Fasilitasi transportasi medis atau ambulans untuk kontrol rutin');

   268→   268→    recs.push('Evaluasi kelayakan bantuan BPJS, Jamkesda, PKH, dan program sosial lainnya');
   269→   269→  if (form.healthAccess.value === 'sulit' || form.healthAccess.value === 'sangat_sulit')
   270→   270→    recs.push('Tingkatkan akses telekonsultasi dan home visit, koordinasi dengan puskesmas terdekat');
   271→   271→  if (form.medicalEquip.value === 'berat' || form.medicalEquip.value === 'sedang')
   272→   272→    recs.push('Fasilitasi pengadaan alat kesehatan melalui program bantuan atau sewa');
   273→   273→  if (form.socialAssist.value === 'berat' || form.socialAssist.value === 'sedang')
   274→   274→    recs.push('Rujuk ke pekerja sosial untuk pendampingan akses bantuan sosial');
   275→   275→  if (form.socialIsolation.value === 'tinggi' || form.socialIsolation.value === 'sedang')
   276→   276→    recs.push('Intervensi psikososial, pertimbangkan kelompok dukungan dan kunjungan sukarelawan');
   277→   277→  if (recs.length === 0) recs.push('Pertahankan monitoring rutin dan dukungan sosial yang ada');
   278→   278→  return recs;
   279→   279→};

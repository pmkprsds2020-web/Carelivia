   142→function checkTTVAlerts(ttv: TTVFormAnswers): { alert: boolean; severity: 'hijau' | 'kuning' | 'merah'; messages: string[] } {
   143→  const messages: string[] = [];
   144→  let maxSeverity: 'hijau' | 'kuning' | 'merah' = 'hijau';
   145→
   146→  if (ttv.systolicBP !== undefined && ttv.systolicBP < 90) {
   147→    messages.push(`TD rendah: ${ttv.systolicBP} mmHg`);
   148→    maxSeverity = 'merah';
   149→  }
   150→  if (ttv.systolicBP !== undefined && ttv.systolicBP > 180) {
   151→    messages.push(`TD tinggi: ${ttv.systolicBP} mmHg`);
   152→    maxSeverity = 'merah';
   153→  }
   154→  if (ttv.heartRate !== undefined && ttv.heartRate > 120) {
   155→    messages.push(`Nadi tinggi: ${ttv.heartRate} bpm`);
   156→    maxSeverity = 'merah';
   157→  }
   158→  if (ttv.respiratoryRate !== undefined && ttv.respiratoryRate > 30) {
   159→    messages.push(`RR tinggi: ${ttv.respiratoryRate}/menit`);
   160→    maxSeverity = 'merah';
   161→  } else if (ttv.respiratoryRate !== undefined && ttv.respiratoryRate > 24) {
   162→    messages.push(`RR meningkat: ${ttv.respiratoryRate}/menit`);
   163→    if (maxSeverity !== 'merah') maxSeverity = 'kuning';
   164→  }
   165→  if (ttv.oxygenSat !== undefined && ttv.oxygenSat < 90) {
   166→    messages.push(`SpO2 rendah: ${ttv.oxygenSat}%`);
   167→    maxSeverity = 'merah';
   168→  }
   169→  if (ttv.temperature !== undefined && ttv.temperature > 38) {
   170→    messages.push(`Demam: ${ttv.temperature}°C`);
   171→    if (maxSeverity !== 'merah') maxSeverity = 'kuning';
   172→  }
   173→  if (ttv.painScore !== undefined && ttv.painScore >= 7) {
   174→    messages.push(`Nyeri berat: ${ttv.painScore}/10`);
   175→    if (maxSeverity !== 'merah') maxSeverity = 'kuning';
   176→  }
   177→
   178→  return { alert: messages.length > 0, severity: maxSeverity, messages };
   179→}

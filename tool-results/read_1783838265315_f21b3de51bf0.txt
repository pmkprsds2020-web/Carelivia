   284→   284→// ── Storage helpers (server-side via API routes to bypass RLS) ──────────────
   285→   285→//
   286→   286→// The browser anon client is subject to Storage RLS and gets:
   287→   287→//   "new row violates row-level security policy"
   288→   288→// when uploading to the `patient-files` bucket. All Storage + DB operations
   289→   289→// for photo exams therefore go through server-side API routes that use the
   290→   290→// service-role key (getSupabaseAdmin) to bypass RLS.
   291→   291→//
   292→   292→// Routes:
   293→   293→//   POST /api/supporting-exams/upload       — upload file + INSERT row
   294→   294→//   POST /api/supporting-exams/update       — optional new file + UPDATE row
   295→   295→//   POST /api/supporting-exams/delete-file  — delete file + DELETE row
   296→   296→//
   297→   297→
   298→   298→export type UploadProgressCb = (phase: 'uploading' | 'inserting' | 'done' | 'error', pct: number, msg?: string) => void;
   299→   299→
   300→   300→/**
   301→   301→ * Call the /api/supporting-exams/upload route.
   302→   302→ * Uploads a file to Storage AND inserts a metadata row in one atomic call.
   303→   303→ * Returns the inserted DB row (raw snake_case) or throws on failure.
   304→   304→ */
   305→   305→async function callUploadApi(
   306→   306→  file: File | Blob,
   307→   307→  metadata: Record<string, any>,
   308→   308→  onProgress?: UploadProgressCb
   309→   309→): Promise<any> {
   310→   310→  const fileName = (file as File).name ?? `upload-${Date.now()}`;
   311→   311→  onProgress?.('uploading', 10, `Mengunggah ${fileName}...`);
   312→   312→
   313→   313→  const fd = new FormData();
   314→   314→  // Coerce to File if it's a Blob (some callers pass Blob without name)
   315→   315→  if (!(file instanceof File) && file instanceof Blob) {
   316→   316→    fd.append('file', file, fileName);
   317→   317→  } else {
   318→   318→    fd.append('file', file as File);
   319→   319→  }
   320→   320→  fd.append('metadata', JSON.stringify(metadata));
   321→   321→
   322→   322→  onProgress?.('uploading', 30);
   323→   323→  let res: Response;
   324→   324→  try {
   325→   325→    res = await fetch('/api/supporting-exams/upload', {
   326→   326→      method: 'POST',
   327→   327→      body: fd,
   328→   328→    });
   329→   329→  } catch (e: any) {
   330→   330→    onProgress?.('error', 0, e?.message ?? 'Network error');
   331→   331→    throw new Error('Upload gagal (network): ' + (e?.message ?? String(e)));
   332→   332→  }
   333→   333→  onProgress?.('uploading', 80);
   334→   334→
   335→   335→  let body: any;
   336→   336→  try {
   337→   337→    body = await res.json();
   338→   338→  } catch {
   339→   339→    body = { error: 'Invalid JSON response from server' };
   340→   340→  }
   341→   341→
   342→   342→  if (!res.ok || !body?.ok) {
   343→   343→    const msg = body?.error || `HTTP ${res.status}`;
   344→   344→    onProgress?.('error', 0, msg);
   345→   345→    // Surface a clear error so the UI can toast it
   346→   346→    const err = new Error(msg);
   347→   347→    (err as any).code = body?.code;
   348→   348→    (err as any).status = res.status;
   349→   349→    throw err;
   350→   350→  }
   351→   351→
   352→   352→  onProgress?.('done', 100);
   353→   353→  return body.row;
   354→   354→}
   355→   355→
   356→   356→/**
   357→   357→ * Call the /api/supporting-exams/update route.
   358→   358→ * Optionally uploads a new file (and deletes the old one), then UPDATEs the row.
   359→   359→ */
   360→   360→async function callUpdateApi(
   361→   361→  id: string,
   362→   362→  metadata: Record<string, any>,
   363→   363→  file?: File | Blob | null,
   364→   364→  oldStoragePath?: string,

   370→   370→  if (oldStoragePath) fd.append('oldStoragePath', oldStoragePath);
   371→   371→  if (file) {
   372→   372→    const fileName = (file as File).name ?? `upload-${Date.now()}`;
   373→   373→    onProgress?.('uploading', 20, `Mengunggah ${fileName}...`);
   374→   374→    if (!(file instanceof File) && file instanceof Blob) {
   375→   375→      fd.append('file', file, fileName);
   376→   376→    } else {
   377→   377→      fd.append('file', file as File);
   378→   378→    }
   379→   379→  } else {
   380→   380→    onProgress?.('inserting', 50, 'Menyimpan perubahan...');
   381→   381→  }
   382→   382→
   383→   383→  let res: Response;
   384→   384→  try {
   385→   385→    res = await fetch('/api/supporting-exams/update', {
   386→   386→      method: 'POST',
   387→   387→      body: fd,
   388→   388→    });
   389→   389→  } catch (e: any) {
   390→   390→    onProgress?.('error', 0, e?.message ?? 'Network error');
   391→   391→    throw new Error('Update gagal (network): ' + (e?.message ?? String(e)));
   392→   392→  }
   393→   393→
   394→   394→  let body: any;
   395→   395→  try {
   396→   396→    body = await res.json();
   397→   397→  } catch {
   398→   398→    body = { error: 'Invalid JSON response from server' };
   399→   399→  }
   400→   400→
   401→   401→  if (!res.ok || !body?.ok) {
   402→   402→    const msg = body?.error || `HTTP ${res.status}`;
   403→   403→    onProgress?.('error', 0, msg);
   404→   404→    const err = new Error(msg);
   405→   405→    (err as any).code = body?.code;
   406→   406→    (err as any).status = res.status;
   407→   407→    throw err;
   408→   408→  }
   409→   409→
   410→   410→  onProgress?.('done', 100);
   411→   411→  return body.row;
   412→   412→}

   431→   431→  let body: any;
   432→   432→  try {

   438→   438→  if (!res.ok || !body?.ok) {
   439→   439→    console.error('[supportingExamService.callDeleteApi] failed:', body?.error ?? res.status);
   440→   440→    return false;
   441→   441→  }
   442→   442→  return true;

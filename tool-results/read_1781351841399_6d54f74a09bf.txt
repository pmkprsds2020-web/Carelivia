   240→const calculateSocialScore = (a: SocialAssessmentRecord): number => {
   241→  const fields: string[] = [
   242→    a.housingCondition, a.caregiverAvailability, a.familySupportLevel,
   243→    a.transportDifficulty, a.economicConstraint, a.healthcareAccess,
   244→    a.medicalEquipmentNeed, a.socialAssistanceNeed, a.socialIsolationRisk,
   245→  ];
   246→  return fields.reduce((sum, f) => sum + (scoreMap[f] ?? 0), 0);
   247→};

   249→const calculatePriority = (form: ScreeningFormState): SocialScreeningPriority => {
   250→  const highVals = ['berat', 'tidak_tersedia', 'sangat_sulit', 'tidak_layak', 'tinggi', 'tidak_ada'];
   251→  const medVals = ['sedang', 'terbatas', 'sulit', 'kurang_layak', 'lemah'];
   252→  const vals = Object.values(form).map(v => v.value);
   253→  if (vals.some(v => highVals.includes(v))) return 'tinggi';
   254→  if (vals.some(v => medVals.includes(v))) return 'sedang';
   255→  return 'rendah';
   256→};

   265→    recs.push('Konseling keluarga dan edukasi tentang dukungan paliatif, pertimbangkan dukungan komunitas');
   266→  if (form.transport.value === 'berat' || form.transport.value === 'sedang')
   267→    recs.push('Fasilitasi transportasi medis atau ambulans untuk kontrol rutin');

   269→    recs.push('Evaluasi kelayakan bantuan BPJS, Jamkesda, PKH, dan program sosial lainnya');

   271→    recs.push('Tingkatkan akses telekonsultasi dan home visit, koordinasi dengan puskesmas terdekat');

   342→export default function SocialSupportPanel({ palliativePatientId }: SocialSupportPanelProps) {
   343→  const { toast } = useToast();
   344→
   345→  const {
   346→    socialAssessments, addSocialAssessment, updateSocialAssessment,
   347→    caregivers, addCaregiver, updateCaregiver, removeCaregiver,
   348→    familyMeetings, addFamilyMeeting, updateFamilyMeeting,
   349→    eduMaterials, logEduMaterialAccess,
   350→    familyCoordinationNotes, addFamilyCoordinationNote, updateFamilyCoordinationNote,
   351→    emergencyContacts, addEmergencyContact, updateEmergencyContact, removeEmergencyContact,
   352→    financialSupportRecords, addFinancialSupportRecord, updateFinancialSupportRecord,
   353→    transportRecords, addTransportRecord, updateTransportRecord,
   354→    socialAlerts, addSocialAlert, markSocialAlertRead,
   355→    palliativePatients,
   356→    currentUser,
   357→  } = useStore();
   358→
   359→  const [activeTab, setActiveTab] = useState('dashboard');
   360→  const tabScrollRef = useRef<HTMLDivElement>(null);
   361→  const [canScrollLeft, setCanScrollLeft] = useState(false);
   362→  const [canScrollRight, setCanScrollRight] = useState(false);
   363→
   364→  // Check scroll position for fade indicators
   365→  const checkScrollPosition = () => {
   366→    if (tabScrollRef.current) {
   367→      const { scrollLeft, scrollWidth, clientWidth } = tabScrollRef.current;
   368→      setCanScrollLeft(scrollLeft > 2);
   369→      setCanScrollRight(scrollLeft + clientWidth < scrollWidth - 2);
   370→    }
   371→  };
   372→
   373→  // Auto-scroll active tab into view
   374→  useEffect(() => {
   375→    if (tabScrollRef.current) {
   376→      const activeTrigger = tabScrollRef.current.querySelector('[data-state="active"]') as HTMLElement;
   377→      if (activeTrigger) {
   378→        activeTrigger.scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'center' });
   379→      }
   380→      // Re-check scroll position after tab switch
   381→      setTimeout(checkScrollPosition, 300);
   382→    }
   383→  }, [activeTab]);
   384→
   385→  // Update scroll indicators on mount, resize, and when patient changes
   386→  useEffect(() => {
   387→    // Delay to ensure tab bar is rendered after patient selection
   388→    const timer = setTimeout(() => {
   389→      checkScrollPosition();
   390→    }, 100);
   391→    const el = tabScrollRef.current;
   392→    if (el) {
   393→      el.addEventListener('scroll', checkScrollPosition);
   394→      window.addEventListener('resize', checkScrollPosition);
   395→      return () => {
   396→        clearTimeout(timer);
   397→        el.removeEventListener('scroll', checkScrollPosition);
   398→        window.removeEventListener('resize', checkScrollPosition);
   399→      };
   400→    }

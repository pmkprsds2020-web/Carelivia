     1→// ───────────────────────────────────────────────────────────────────────────
     2→// SupabaseSyncProvider — keeps the Zustand store in sync with Supabase.
     3→//
     4→// Responsibilities:
     5→//   1. On mount, load all Monitoring-Paliatif data from Supabase via the 13
     6→//      service files and populate the Zustand store. Local demo data wins on
     7→//      conflict — we only replace the store if Supabase returns non-empty
     8→//      arrays.
     9→//   2. Subscribe to Supabase Realtime for 12 tables so any DB change is
    10→//      mirrored into the store immediately (INSERT/UPDATE/DELETE).
    11→//   3. ALWAYS render children immediately. Supabase loading is fire-and-forget
    12→//      and never blocks the UI. If Supabase is unreachable (missing tables,
    13→//      RLS denied, network down), we log a warning and continue with local
    14→//      Zustand data.
    15→//   4. NEVER throw — every subscription and load is wrapped in try/catch.
    16→// ───────────────────────────────────────────────────────────────────────────

    43→/** Check whether a row id already exists in a store array. */
    44→function hasId<T extends { id: string }>(list: T[], id: string): boolean {
    45→  return list.some((x) => x.id === id);
    46→}
    47→
    48→/** Log a non-fatal sync warning. */
    49→function warn(label: string, err: unknown) {
    50→  console.warn(`[SupabaseSync] ${label}:`, err);
    51→}

    53→// ─── Initial-load handlers ─────────────────────────────────────────────────
    54→//
    55→// Each handler is self-contained: it loads data from Supabase, validates it,
    56→// and dispatches the corresponding Zustand action(s). Wrapped in try/catch by
    57→// the caller. We ALWAYS replace the store with what Supabase returns — even
    58→// if it's an empty array — so the UI reflects the database state exactly.
    59→// (Initial store state is also empty, so there's never stale demo data.)
    60→
    61→async function loadPatients(store: ReturnType<typeof useStore.getState>) {
    62→  const patients = await svc.patientService.getAll();
    63→  // Always set — even if empty — so UI shows "Tidak ada data pasien" when DB is empty.
    64→  if (Array.isArray(patients)) {
    65→    store.setPalliativePatients(patients as PalliativePatientInfo[]);
    66→  }
    67→  return patients;
    68→}

   356→// ─── Realtime handlers ─────────────────────────────────────────────────────
   357→//
   358→// Each handler receives a PostgresChangesPayload and dispatches the matching
   359→// Zustand action. We dedupe by id on INSERT, replace on UPDATE, filter on
   360→// DELETE. Wrapped in try/catch by the caller so a single bad event can never
   361→// break the subscription.
   362→
   363→type RealtimePayload = {
   364→  eventType: 'INSERT' | 'UPDATE' | 'DELETE' | '*';
   365→  new: Record<string, any> | null;
   366→  old: Record<string, any> | null;
   367→  errors: string[] | null;
   368→};
   369→
   370→function handlePatientEvent(store: ReturnType<typeof useStore.getState>, p: RealtimePayload) {
   371→  const row = p.new;
   372→  const id = (row?.id ?? p.old?.id) as string | undefined;
   373→  if (!id) return;
   374→  if (p.eventType === 'DELETE') {
   375→    if (hasId(store.palliativePatients, id)) store.removePalliativePatient(id);
   376→    return;
   377→  }
   378→  // INSERT or UPDATE — try to load the full record via the service so the
   379→  // row goes through the proper snake_case → camelCase mapping.
   380→  svc.patientService
   381→    .getById(id)
   382→    .then((fresh: PalliativePatientInfo | null) => {
   383→      if (!fresh) return;
   384→      const exists = hasId(useStore.getState().palliativePatients, id);
   385→      if (exists) {
   386→        store.updatePalliativePatient(id, fresh);
   387→      } else {
   388→        store.addPalliativePatient(fresh);
   389→      }
   390→    })
   391→    .catch((e: unknown) => warn('patient realtime getById', e));
   392→}

   394→function handleVitalEvent(store: ReturnType<typeof useStore.getState>, p: RealtimePayload) {
   395→  const row = p.new;
   396→  const id = (row?.id ?? p.old?.id) as string | undefined;
   397→  if (!id) return;
   398→  if (p.eventType === 'DELETE') {
   399→    useStore.setState((s) => ({
   400→      vitalSignRecords: s.vitalSignRecords.filter((v) => v.id !== id),
   401→    }));
   402→    return;
   403→  }
   404→  // Reload via service for proper mapping
   405→  svc.vitalService
   406→    .getAll(row?.patient_id as string)
   407→    .then((rows: VitalSignRecordInfo[]) => {
   408→      const fresh = rows.find((r) => r.id === id);
   409→      if (!fresh) return;
   410→      const state = useStore.getState();
   411→      const exists = hasId(state.vitalSignRecords, id);
   412→      if (exists) {
   413→        useStore.setState({
   414→          vitalSignRecords: state.vitalSignRecords.map((v) => (v.id === id ? fresh : v)),
   415→        });
   416→      } else {
   417→        store.addVitalSignRecord(fresh);
   418→      }
   419→    })
   420→    .catch((e: unknown) => warn('vital realtime', e));
   421→}

   749→// ─── Generic realtime handlers for the new modules ─────────────────────────
   750→//
   751→// These tables (caregivers, family_meetings, family_coordination_notes,
   752→// emergency_contacts, financial_support, transport_records, palliative_resumes,
   753→// referral_letters, patient_documents) all follow the same pattern:
   754→//   - DELETE → filter the matching array by id
   755→//   - INSERT/UPDATE → reload via the service's getAll(patientId) and
   756→//     upsert the fresh row into the store array.
   757→//
   758→// We use a factory so each table gets its own typed handler without duplicating
   759→// the boilerplate.
   760→
   761→function makeGenericHandler<T extends { id: string; palliativePatientId?: string }>(opts: {
   762→  label: string;
   763→  getState: () => T[];
   764→  setState: (updater: (cur: T[]) => T[]) => void;
   765→  reload: (patientId: string) => Promise<T[]>;
   766→}): (store: ReturnType<typeof useStore.getState>, p: RealtimePayload) => void {
   767→  return (_store, p) => {
   768→    const row = p.new;
   769→    const id = (row?.id ?? p.old?.id) as string | undefined;
   770→    if (!id) return;
   771→    const patientId = (row?.patient_id ?? row?.palliativePatientId ?? p.old?.patient_id) as string | undefined;
   772→    if (p.eventType === 'DELETE') {
   773→      opts.setState((cur) => cur.filter((x) => x.id !== id));
   774→      return;
   775→    }
   776→    if (!patientId) return;
   777→    opts
   778→      .reload(patientId)
   779→      .then((rows) => {
   780→        const fresh = rows.find((r) => r.id === id);
   781→        if (!fresh) return;
   782→        opts.setState((cur) => {
   783→          const exists = cur.some((x) => x.id === id);
   784→          return exists ? cur.map((x) => (x.id === id ? fresh : x)) : [...cur, fresh];
   785→        });
   786→      })
   787→      .catch((e: unknown) => warn(`realtime ${opts.label}`, e));
   788→  };
   789→}

   874→// ─── Provider ──────────────────────────────────────────────────────────────

   876→export function SupabaseSyncProvider({ children }: SupabaseSyncProviderProps) {
   877→  const store = useStore();
   878→  const loadedRef = useRef(false);
   879→  const channelRef = useRef<ReturnType<typeof supabase.channel> | null>(null);
   880→
   881→  useEffect(() => {
   882→    if (loadedRef.current) return;
   883→    loadedRef.current = true;
   884→
   885→    // Snapshot the store once — actions on this snapshot remain stable for
   886→    // the lifetime of the provider, even if the component re-renders.
   887→    const s = useStore.getState;
   888→
   889→    // ── 1. Initial load (fire-and-forget, never blocks UI) ────────────────
   890→    (async () => {
   891→      try {
   892→        const patients = await loadPatients(s());
   893→        // Always run scoped-data load — even when patients is empty — so any
   894→        // stale per-patient arrays (vitals, meds, etc.) are cleared.
   895→        if (Array.isArray(patients)) {
   896→          await loadPatientScopedData(s(), patients as PalliativePatientInfo[]);
   897→        }
   898→        console.log('[SupabaseSync] initial load complete — patients:', patients?.length ?? 0);
   899→      } catch (e) {
   900→        warn('initial load skipped', e);
   901→      }
   902→    })();
   903→
   904→    // ── 2. Realtime subscriptions (best-effort, silently skipped on error) ─
   905→    try {
   906→      const channel = supabase.channel('carelivia-realtime');
   907→      channelRef.current = channel;
   908→
   909→      const tableHandlers: Array<[string, (store: ReturnType<typeof s>, p: RealtimePayload) => void]> = [
   910→        ['patients', handlePatientEvent],
   911→        ['vital_signs', handleVitalEvent],
   912→        ['screenings', handleScreeningEvent],
   913→        ['medications', handleMedicationEvent],
   914→        ['nutrition', handleNutritionEvent],
   915→        ['daily_complaints', handleComplaintEvent],
   916→        ['social_assessments', handleSocialEvent],
   917→        ['acp', handleAcpEvent],
   918→        ['chat_rooms', handleChatRoomEvent],
   919→        ['messages', handleMessageEvent],
   920→        ['clinical_alerts', handleClinicalAlertEvent],
   921→        ['audit_log', handleAuditLogEvent],
   922→        ['notifications', handleNotificationEvent],
   923→        // ── New modules — full realtime sync ──────────────────────────────
   924→        ['caregivers', handleCaregiverEvent],
   925→        ['family_meetings', handleFamilyMeetingEvent],
   926→        ['family_coordination_notes', handleFamilyCoordinationNoteEvent],
   927→        ['emergency_contacts', handleEmergencyContactEvent],
   928→        ['financial_support', handleFinancialSupportEvent],
   929→        ['transport_records', handleTransportRecordEvent],
   930→        ['palliative_resumes', handlePalliativeResumeEvent],
   931→        ['referral_letters', handleReferralLetterEvent],
   932→        ['patient_documents', handlePatientDocumentEvent],
   933→      ];
   934→
   935→      let builder = channel;
   936→      for (const [table, handler] of tableHandlers) {
   937→        try {
   938→          builder = builder.on(
   939→            'postgres_changes',
   940→            { event: '*', schema: 'public', table },
   941→            (payload: any) => {
   942→              try {
   943→                handler(s(), {
   944→                  eventType: payload.eventType,
   945→                  new: payload.new,
   946→                  old: payload.old,
   947→                  errors: payload.errors,
   948→                });
   949→              } catch (e) {
   950→                warn(`realtime handler for ${table}`, e);
   951→              }
   952→            }
   953→          );
   954→        } catch (e) {
   955→          warn(`subscribe to ${table} skipped`, e);
   956→        }
   957→      }
   958→
   959→      try {
   960→        builder.subscribe((status: string) => {
   961→          if (status === 'CHANNEL_ERROR' || status === 'TIMED_OUT') {
   962→            warn(`realtime channel status: ${status}`, null);
   963→          } else {
   964→            // 'SUBSCRIBED' | 'CLOSED' — both are non-fatal
   965→          }
   966→        });
   967→      } catch (e) {
   968→        warn('channel.subscribe skipped', e);
   969→      }
   970→    } catch (e) {
   971→      warn('realtime setup skipped', e);
   972→    }
   973→
   974→    // ── 3. Cleanup on unmount ─────────────────────────────────────────────
   975→    return () => {
   976→      try {
   977→        if (channelRef.current) {
   978→          supabase.removeChannel(channelRef.current);
   979→          channelRef.current = null;
   980→        }
   981→      } catch (e) {
   982→        warn('removeChannel', e);
   983→      }
   984→    };
   985→  }, []);
   986→
   987→  // ALWAYS render children immediately. Supabase loads in the background.
   988→  return <>{children}</>;
   989→}

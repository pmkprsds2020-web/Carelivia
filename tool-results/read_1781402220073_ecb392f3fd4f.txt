   154→   154→// ── Helper: Compute vital sign criticality score ─────────────────────────────
   155→   155→function computeCriticalityScore(v: VitalSignRecord): { score: number; reasons: string[] } {
   156→   156→  let score = 0;
   157→   157→  const reasons: string[] = [];
   158→   158→
   159→   159→  if (v.oxygenSat != null && v.oxygenSat < 90) {
   160→   160→    score += 4;
   161→   161→    reasons.push(`SpO2 ${v.oxygenSat}% (<90%)`);
   162→   162→  }
   163→   163→  if (v.heartRate != null) {
   164→   164→    if (v.heartRate > 120) { score += 3; reasons.push(`Nadi ${v.heartRate} bpm (>120)`); }
   165→   165→    else if (v.heartRate < 50) { score += 3; reasons.push(`Nadi ${v.heartRate} bpm (<50)`); }
   166→   166→  }
   167→   167→  if (v.respiratoryRate != null) {
   168→   168→    if (v.respiratoryRate > 28) { score += 3; reasons.push(`RR ${v.respiratoryRate}/menit (>28)`); }
   169→   169→    else if (v.respiratoryRate < 8) { score += 3; reasons.push(`RR ${v.respiratoryRate}/menit (<8)`); }
   170→   170→  }
   171→   171→  if (v.systolicBP != null) {
   172→   172→    if (v.systolicBP > 180) { score += 3; reasons.push(`SBP ${v.systolicBP} mmHg (>180)`); }
   173→   173→    else if (v.systolicBP < 90) { score += 3; reasons.push(`SBP ${v.systolicBP} mmHg (<90)`); }
   174→   174→  }
   175→   175→  if (v.temperature != null) {
   176→   176→    if (v.temperature > 39) { score += 2; reasons.push(`Suhu ${v.temperature}°C (>39)`); }
   177→   177→    else if (v.temperature < 35) { score += 2; reasons.push(`Suhu ${v.temperature}°C (<35)`); }
   178→   178→  }
   179→   179→
   180→   180→  return { score, reasons };
   181→   181→}

   183→   183→// ── Helper: Categorize medication ────────────────────────────────────────────
   184→   184→function categorizeMedication(med: MedicationWithAdherences): string {
   185→   185→  const name = med.medicineName.toLowerCase();
   186→   186→  const indication = (med.indication || '').toLowerCase();
   187→   187→
   188→   188→  // Analgesik
   189→   189→  const analgesikKeywords = ['nyeri', 'pain', 'morphine', 'fentanyl', 'tramadol', 'ketorolac', 'paracetamol', 'ibuprofen', 'diclofenac', 'morfine', 'kodein', 'gabapentin', 'pregabalin'];
   190→   190→  if (analgesikKeywords.some(k => name.includes(k) || indication.includes(k))) return 'analgesik';
   191→   191→
   192→   192→  // Antiemetik
   193→   193→  const antiemetikKeywords = ['mual', 'nausea', 'vomiting', 'muntah', 'ondansetron', 'metoclopramide', 'domperidone', 'granisetron'];
   194→   194→  if (antiemetikKeywords.some(k => name.includes(k) || indication.includes(k))) return 'antiemetik';
   195→   195→
   196→   196→  // Laksatif
   197→   197→  const laksatifKeywords = ['konstipasi', 'constipation', 'sembelit', 'bisacodyl', 'lactulose', 'senna', 'peg', 'movicol'];
   198→   198→  if (laksatifKeywords.some(k => name.includes(k) || indication.includes(k))) return 'laksatif';
   199→   199→
   200→   200→  // Antidepresan
   201→   201→  const antidepresanKeywords = ['depresi', 'depression', 'sertraline', 'fluoxetine', 'mirtazapine', 'citalopram', 'escitalopram', 'venlafaxine', 'amitriptyline'];
   202→   202→  if (antidepresanKeywords.some(k => name.includes(k) || indication.includes(k))) return 'antidepresan';
   203→   203→
   204→   204→  // Ansiolitik
   205→   205→  const ansiolitikKeywords = ['cemas', 'anxiety', 'diazepam', 'lorazepam', 'alprazolam', 'clonazepam', 'midazolam'];
   206→   206→  if (ansiolitikKeywords.some(k => name.includes(k) || indication.includes(k))) return 'ansiolitik';
   207→   207→
   208→   208→  return 'lainnya';
   209→   209→}

   229→   229→function formatTTVNarrative(v: VitalSignRecord, label: string): string {
   230→   230→  const tgl = new Date(v.recordedAt).toLocaleString('id-ID');
   231→   231→  const parts = [
   232→   232→    `${label} (${tgl}):`,
   233→   233→    `  TD: ${fmtVal(v.systolicBP, '')}/${fmtVal(v.diastolicBP, ' mmHg')}`,
   234→   234→    `  Nadi: ${fmtVal(v.heartRate, ' bpm')}`,
   235→   235→    `  RR: ${fmtVal(v.respiratoryRate, '/menit')}`,
   236→   236→    `  Suhu: ${fmtVal(v.temperature, '°C')}`,
   237→   237→    `  SpO2: ${fmtVal(v.oxygenSat, '%')}`,
   238→   238→  ];
   239→   239→  if (v.weight != null) parts.push(`  BB: ${v.weight} kg`);
   240→   240→  if (v.bmi != null) parts.push(`  BMI: ${v.bmi}`);
   241→   241→  if (v.notes) parts.push(`  Catatan: ${v.notes}`);
   242→   242→  return parts.join('\n');
   243→   243→}

   256→   256→  parts.push(`  Makan/Minum: ${c.makanMinum === 'ya' ? 'Ya' : 'Tidak'}`);
   257→   257→  if (c.makanMinum === 'tidak' && c.alasanMakanMinum) parts.push(`  Alasan: ${c.alasanMakanMinum}`);
   258→   258→  parts.push(`  Tidur: ${c.tidur === 'ya' ? 'Ya' : 'Tidak'}`);
   259→   259→  if (c.tidur === 'tidak' && c.alasanTidur) parts.push(`  Alasan: ${c.alasanTidur}`);

   261→   261→    parts.push(`  Masalah Obat: ${c.deskripsiMasalahObat}`);
   262→   262→  }
   263→   263→  parts.push(`  Tingkat Keparahan: ${c.severityLevel}`);
   264→   264→  return parts.join('\n');

   650→   650→function parseLLMJson(response: string): AIAnalysis | null {
   651→   651→  // Try direct JSON parse
   652→   652→  try {
   653→   653→    const parsed = JSON.parse(response);
   654→   654→    if (parsed && typeof parsed === 'object') return parsed as AIAnalysis;
   655→   655→  } catch {
   656→   656→    // Not direct JSON, continue
   657→   657→  }
   658→   658→
   659→   659→  // Try to extract JSON from markdown code blocks
   660→   660→  const jsonBlockMatch = response.match(/```(?:json)?\s*\n?([\s\S]*?)\n?\s*```/);
   661→   661→  if (jsonBlockMatch) {
   662→   662→    try {
   663→   663→      const parsed = JSON.parse(jsonBlockMatch[1].trim());
   664→   664→      if (parsed && typeof parsed === 'object') return parsed as AIAnalysis;
   665→   665→    } catch {
   666→   666→      // Failed to parse code block JSON
   667→   667→    }
   668→   668→  }
   669→   669→
   670→   670→  // Try to find JSON object in the response
   671→   671→  const jsonMatch = response.match(/\{[\s\S]*\}/);
   672→   672→  if (jsonMatch) {
   673→   673→    try {
   674→   674→      const parsed = JSON.parse(jsonMatch[0]);
   675→   675→      if (parsed && typeof parsed === 'object') return parsed as AIAnalysis;
   676→   676→    } catch {
   677→   677→      // Failed to parse extracted JSON
   678→   678→    }
   679→   679→  }
   680→   680→
   681→   681→  return null;
   682→   682→}

  1000→  1000→  const count = await db.palliativeResume.count({
  1001→  1001→    where: {
  1002→  1002→      createdAt: {
  1003→  1003→        gte: startOfMonth,
  1004→  1004→        lt: startOfNextMonth,
  1005→  1005→      },
  1006→  1006→    },
  1007→  1007→  });
  1008→  1008→
  1009→  1009→  const sequence = String(count + 1).padStart(4, '0');
